# AI 短剧控制台 · P5b 字幕 · 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 成片带字幕:每段对着当前视频把已知台词对齐到时间轴(云机 GPU 上 whisper 出逐字时间,本机算句首句尾、判卷、写 SRT),在视频面板上对齐 / 查看 / 补全,导出时拼成整集字幕烧进两版成片。

**Architecture:** 对齐是一种新的 task(`kind = 'align'`,库里 P1 起就留了这个值),与首帧 / 视频同一张 task 表、同一个 worker:
worker 空闲时轮到它,就在本轮里经 ssh 同步跑完(不进 ComfyUI 队列)。云机只负责「听」(逐字时间戳),**起止时间的计算与判卷全在本机、纯函数、可单测**:
句尾用 whisper,句首吸附到本机算的能量曲线上的发声起点。导出把各段 SRT 按段落起点平移拼成整集,libass 烧录,读回自检核对。

**Tech Stack:** Python 3.14 标准库 · 本机 ffmpeg 4.4(libass)· 云机 whisper-large-v3-turbo(MIT,借 ComfyUI 的 venv:transformers 5.12、torch 2.12)· 现有 FastAPI / Jinja2 / Playwright。**不加依赖、不动库结构版本、云机上不装任何东西。**

```python
# -*- coding: utf-8 -*-
"""字幕:已知台词 + whisper 逐字时间戳 + 本地能量曲线 → 每句的起止时间 → SRT。纯函数,可单测。

2026-09-26 云机实测(`spike-results/2026-09-26-字幕对齐云机实测.md`):
- whisper token 级时间戳的**句尾可靠**(一片两句:句尾误差 0.06 s);
- **句首会被拉进前面的静音**(首字的时间段从上一个 token 的时刻算起):第 2 句被标成 3.80 s,实际 4.89 s 才开口 ——
  所以句首**吸附到能量曲线的发声起点**(0.1 s 窗 RMS 第一次越过峰值的一定比例);
- 识别会有错字(「它是」→「她是」)与字节级 token(「玉佩」),逐字对齐用 difflib,命中率低于 MIN_HIT 就不出这句的字幕。

用数据审:命中率、句首 < 句尾、各句单调不重叠、不超出片长 —— 不过就判失败,不「尽力出一版」。
"""
from __future__ import annotations

import difflib
from dataclasses import dataclass

PUNCT = set("，。！？、,.!?\"'“”「」：:；; …—")
MIN_HIT = 0.7            # 一句台词至少对上 70% 的字才算对齐可信
ONSET_RATIO = 0.15       # 能量越过峰值的 15% 算开口
STEP = 0.1               # 能量曲线的窗长(秒)
MIN_SHOW = 1.0           # 字幕至少显示 1 秒
TAIL = 0.3               # 句尾多留 0.3 秒(不越过下一句句首)


@dataclass(frozen=True)
class Cue:
    text: str
    start: float
    end: float


def _clean(s: str) -> str:
    return "".join(c for c in s if c not in PUNCT and not c.isspace())


def align(lines: list[str], chars: list[tuple[str, float, float]]) -> list[dict]:
    """每句台词按顺序在识别文字里找命中字(用过的字不再给后面的句子)。
    返回 [{line, hit, total, first, second, last}],first/second/last 是命中字在 chars 里的下标(没命中为 None)。"""
    heard = "".join(c for c, _, _ in chars)
    used = 0
    out = []
    for line in lines:
        want = _clean(line)
        sm = difflib.SequenceMatcher(None, want, heard[used:], autojunk=False)
        hit = sorted(used + b + k for _, b, n in sm.get_matching_blocks() for k in range(n))
        out.append({"line": line, "hit": len(hit), "total": len(want),
                    "first": hit[0] if hit else None, "second": hit[1] if len(hit) > 1 else (hit[0] if hit else None),
                    "last": hit[-1] if hit else None})
        if hit:
            used = hit[-1] + 1
    return out


def onset(envelope: list[float], lo: float, hi: float) -> float | None:
    """[lo, hi] 内能量第一个**上升沿**(前一窗低于、这一窗达到峰值的 ONSET_RATIO)的时刻;没有返回 None。
    要上升沿而不是「第一次高于」:lo 是上一句的句尾,上一句的尾音可能还没落下去。"""
    peak = max(envelope) if envelope else 0.0
    if peak <= 0:
        return None
    th = peak * ONSET_RATIO
    for i in range(round(lo / STEP), min(len(envelope), round(hi / STEP) + 1)):
        if envelope[i] >= th and (i == 0 or envelope[i - 1] < th):
            return round(i * STEP, 2)
    return None


def build(lines: list[str], chars: list[tuple[str, float, float]], envelope: list[float],
          duration: float) -> tuple[list[Cue], list[str]]:
    """返回 (字幕, 问题)。有问题就不该出字幕 —— 调用方据问题判失败。"""
    problems = []
    cues = []
    prev_end = 0.0
    for n, a in enumerate(align(lines, chars), 1):
        if a["total"] == 0:
            continue
        if a["hit"] / a["total"] < MIN_HIT:
            problems.append(f"第 {n} 句只对上 {a['hit']}/{a['total']} 字(低于 {int(MIN_HIT * 100)}%)—— 识别与台词对不上,需人工校")
            continue
        end = chars[a["last"]][2]
        guess = chars[a["second"]][2]                      # 第二个命中字的结束时刻:句首一定在它之前
        start = onset(envelope, prev_end, guess)
        if start is None:                                   # 找不到发声起点:退回 whisper 的首字时刻
            start = max(prev_end, chars[a["first"]][1])
        if not prev_end <= start < end:
            problems.append(f"第 {n} 句时间不合理:{start:.2f}–{end:.2f}(上一句到 {prev_end:.2f})")
            continue
        if end > duration + 0.1:
            problems.append(f"第 {n} 句结束 {end:.2f}s 超出片长 {duration:.2f}s")
            continue
        cues.append(Cue(a["line"], round(start, 2), round(end, 2)))
        prev_end = end
    return _pad(cues, duration), problems


def _pad(cues: list[Cue], duration: float) -> list[Cue]:
    """句尾多留一点、太短的拉到 MIN_SHOW,但都不越过下一句句首与片长。"""
    out = []
    for i, c in enumerate(cues):
        limit = cues[i + 1].start if i + 1 < len(cues) else duration
        end = min(max(c.end + TAIL, c.start + MIN_SHOW), limit)
        out.append(Cue(c.text, c.start, round(max(end, c.end), 2)))
    return out


def _ts(t: float) -> str:
    ms = int(round(t * 1000))
    return f"{ms // 3600000:02d}:{ms // 60000 % 60:02d}:{ms // 1000 % 60:02d},{ms % 1000:03d}"


def srt(cues: list[Cue], offset: float = 0.0, first_index: int = 1) -> str:
    return "".join(f"{i}\n{_ts(c.start + offset)} --> {_ts(c.end + offset)}\n{c.text}\n\n"
                   for i, c in enumerate(cues, first_index))


def parse_srt(text: str) -> list[Cue]:
    cues = []
    for block in text.strip().split("\n\n"):
        rows = block.strip().splitlines()
        if len(rows) < 3:
            continue
        a, b = (x.strip() for x in rows[1].split("-->"))
        cues.append(Cue("\n".join(rows[2:]), _secs(a), _secs(b)))
    return cues


def _secs(ts: str) -> float:
    h, m, rest = ts.split(":")
    s, ms = rest.split(",")
    return int(h) * 3600 + int(m) * 60 + int(s) + int(ms) / 1000


def episode_srt(parts: list[tuple[list[Cue], float]]) -> str:
    """整集字幕:parts = [(本段字幕, 本段在成片里的起点秒)],编号连续。"""
    out, n = [], 1
    for cues, offset in parts:
        out.append(srt(cues, offset, n))
        n += len(cues)
    return "".join(out)
```

```python
def rms_envelope(video: Path, step: float = 0.1) -> list[float]:
    """音轨的能量曲线:单声道 16 kHz,每 step 秒一窗的 RMS(16 位样本的原值)。字幕句首吸附用。"""
    p = _run(["ffmpeg", "-v", "error", "-i", str(video), "-vn", "-ac", "1", "-ar", "16000", "-f", "s16le", "-"],
             binary=True)
    samples = array.array("h", p.stdout)
    w = round(16000 * step)
    return [round(math.sqrt(sum(x * x for x in samples[i:i + w]) / w), 1) for i in range(0, len(samples) - w + 1, w)]
```

```python
# -*- coding: utf-8 -*-
"""**在云机上跑**的逐字时间戳脚本:源码经 SSH 的 stdin 送过去,用 ComfyUI 的 venv 执行(不在云机上装任何东西)。

    <venv python> - <whisper 目录> <视频绝对路径>

stdout 最后一行是 JSON:`{"chars": [[字, 起, 止], ...], "asr_sec": 秒}`。失败时退出码非 0、stderr 写原因:
2 = 模型目录 / 视频不在;3 = 显存不够(和 ComfyUI 同卡)。

2026-09-26 云机实测(`spike-results/2026-09-26-字幕对齐云机实测.md`):
- `pipeline(return_timestamps="word")` 对中文退化成整句一块 —— 用 `generate(return_token_timestamps=True)`;
- transformers 5 返回 dict,取 `out["sequences"]` / `out["token_timestamps"]`;
- 一个汉字可能被拆成几个字节级 token,单独解码是 `�` —— 攒到能解出完整字再算一组(`merge_tokens`);
- GPU fp16 每条 8 秒片 5–6 s(CPU 要 290 s,不用)。

本模块**顶层只用标准库**:torch / transformers 在 `main()` 里才导入,本机单测只测 `merge_tokens`。
"""
from __future__ import annotations

import json
import subprocess
import sys
import time
from pathlib import Path

BAD = "�"


def merge_tokens(ids: list[int], times: list[float], decode) -> list[tuple[str, float, float]]:
    """token 序列 → 逐字 (字, 起, 止)。第 k 个 token 的时间段是「上一个 token 的时刻 → 它的时刻」;
    解不出完整字的 token 与后面的攒成一组,组内每个字共用这一组的时间段。空白、特殊 token 丢掉。"""
    out: list[tuple[str, float, float]] = []
    k = 0
    while k < len(ids):
        j = k
        text = decode(ids[k:j + 1])
        while BAD in text and j + 1 < len(ids):
            j += 1
            text = decode(ids[k:j + 1])
        start = times[k - 1] if k > 0 else 0.0
        for c in text:
            if not c.isspace() and c != BAD:
                out.append((c, round(start, 2), round(times[j], 2)))
        k = j + 1
    return out


def _audio(video: str):
    import numpy as np
    p = subprocess.run(["ffmpeg", "-v", "error", "-i", video, "-vn", "-ac", "1", "-ar", "16000", "-f", "s16le", "-"],
                       capture_output=True, check=True)
    return np.frombuffer(p.stdout, dtype=np.int16).astype(np.float32) / 32768


def main(argv: list[str]) -> int:
    model_dir, video = argv
    for p in (model_dir, video):
        if not Path(p).exists():
            print(f"不存在:{p}", file=sys.stderr)
            return 2
    import torch
    from transformers import WhisperForConditionalGeneration, WhisperProcessor

    t0 = time.time()
    try:
        proc = WhisperProcessor.from_pretrained(model_dir)
        model = WhisperForConditionalGeneration.from_pretrained(model_dir, dtype=torch.float16).to("cuda")
        feats = proc(_audio(video), sampling_rate=16000, return_tensors="pt").input_features.to("cuda", model.dtype)
        out = model.generate(feats, language="zh", task="transcribe", return_token_timestamps=True,
                             return_dict_in_generate=True)
    except torch.cuda.OutOfMemoryError as e:
        print(f"显存不够(ComfyUI 可能正占着卡):{e}", file=sys.stderr)
        return 3
    ids = [int(x) for x in out["sequences"][0]]
    times = [float(x) for x in out["token_timestamps"][0]]
    chars = merge_tokens(ids, times, lambda t: proc.tokenizer.decode(t, skip_special_tokens=True))
    print(json.dumps({"chars": chars, "asr_sec": round(time.time() - t0, 1)}, ensure_ascii=False))
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
```

```python
# -*- coding: utf-8 -*-
"""把「这个段落要对字幕」变成一条不可变的对齐任务(与首帧 / 视频同一张 task 表、同一个 worker)。

对齐的输入是**当前视频 + 段落描述里的台词**,规划时定死:之后换了视频或改了台词,旧任务的结果对不上新内容,
worker 执行前会核对视频没换(`aligner`)。whisper 是确定性解码,种子记 0。
"""
from __future__ import annotations

from tools.drama import repo, rules
from tools.drama.db import Store
from tools.drama.tasks import create_task
from tools.gen.capability import WHISPER_ALIGN


def align_problems(store: Store, segment_id: int) -> list[str]:
    seg = repo.get(store, "segment", segment_id)
    problems = []
    if not seg["video_path"]:
        problems.append("还没有视频")
    if not rules.dialogue_lines(seg["description"]):
        problems.append("段落描述里没有台词(「某某说：「……」」),不需要字幕")
    return problems


def plan_align(store: Store, segment_id: int) -> int:
    problems = align_problems(store, segment_id)
    if problems:
        raise ValueError(";".join(problems))
    seg = repo.get(store, "segment", segment_id)
    lines = [line for _, line in rules.dialogue_lines(seg["description"])]
    params = {"workflow": "whisper_align", "video": seg["video_path"], "lines": lines,
              "writes": "subtitle_path", "expect_ext": ".srt"}
    return create_task(store, kind="align", target_type="segment", target_id=segment_id, model=WHISPER_ALIGN.name,
                       params=params, seed=0, prompt="\n".join(lines), refs=[seg["video_path"]])
```

```python
# -*- coding: utf-8 -*-
"""执行一条字幕对齐任务:视频传到云机 → 云机 GPU 上 whisper 出逐字时间 → **本机**算每句起止、判卷、写 SRT。

与 `dispatch` 同一个签名、同一套结局,worker 按 task.kind 选一个调:
- 连不上云机(上传失败 / ssh 连不上,退出码 255)→ **留在 `pending`**,下一轮再试;
- 其余失败(视频不在、规划后视频被换了、云机脚本报错、判卷不过)→ `failed` 写明原因;
- 成功 → 一次走完 `queued → done`(对齐 30 秒内同步跑完,不进 ComfyUI 队列,没有可接管的在途状态)。
  中途进程被杀,任务还在 `pending`,下一轮重跑 —— 对齐是确定性的、便宜的,重跑无害。

**判卷不过不出字幕**(用数据审 LLM 的同一条纪律:识别结果也是模型输出):命中率低、时间不单调、超片长都算失败,
`align.json` 照样落盘(逐字时间、能量曲线、每句判定),人可以查是识别错了还是台词与画面对不上。
写回只填空位(与 collect 一致):段落已有字幕时,新版本留在任务记录里等人挑。
"""
from __future__ import annotations

import json
import sqlite3
from pathlib import Path

from tools.cut import media, subtitles
from tools.drama import repo
from tools.drama.db import Store
from tools.drama.tasks import set_task_state
from tools.gen import preflight
from tools.gen.capability import WHISPER_ALIGN
from tools.gen.comfy_client import ComfyError, ComfyUnreachable
from tools.serve.dispatch import Outcome, sha256_file
from tools.serve.hosts import Host

SCRIPT = Path(__file__).with_name("align_remote.py")
SSH_UNREACHABLE = 255


def parse_output(out: str) -> list[tuple[str, float, float]]:
    """云机脚本的输出(stdout 与 stderr 混在一起):取最后一行以 `{"chars"` 开头的 JSON。"""
    rows = [line for line in out.splitlines() if line.startswith('{"chars"')]
    if not rows:
        raise ValueError(f"云机脚本没有输出结果:{out.strip()[-500:]}")
    return [tuple(c) for c in json.loads(rows[-1])["chars"]]


def run(store: Store, client, host: Host, task: sqlite3.Row, *, data_root: Path, run: preflight.Runner) -> Outcome:
    tid = task["id"]
    root = Path(data_root)
    p = json.loads(task["params_json"])

    def fail(msg: str) -> Outcome:
        set_task_state(store, tid, "failed", error=msg)
        return Outcome("failed", msg)

    video = root / p["video"]
    if not video.is_file():
        return fail(f"视频不存在:{p['video']}")
    seg = repo.get(store, "segment", task["target_id"])
    if seg["video_path"] != p["video"]:
        return fail(f"规划后段落换了视频({p['video']} → {seg['video_path']}),这条对齐作废 —— 请对新视频重新对齐")

    try:
        name = client.upload_image(video, subfolder=f"ai-short-drama/{sha256_file(video)[:16]}")
    except ComfyUnreachable as e:
        return Outcome("pending", f"连不上云机,下轮再试:{e}")
    except ComfyError as e:
        return fail(f"视频传不上云机:{e}")
    code, out = run([WHISPER_ALIGN.model_dir, f"{WHISPER_ALIGN.input_dir}/{name}"], SCRIPT.read_bytes())
    if code == SSH_UNREACHABLE:
        return Outcome("pending", f"ssh 连不上云机,下轮再试:{out.strip()[-300:]}")
    if code != 0:
        return fail(f"云机对齐脚本失败(退出码 {code}):{out.strip()[-500:]}")
    try:
        chars = parse_output(out)
    except ValueError as e:
        return fail(str(e))

    envelope = media.rms_envelope(video)
    duration = media.probe(video).duration
    cues, problems = subtitles.build(p["lines"], chars, envelope, duration)
    out_dir = root / "outputs" / "tasks" / str(tid)
    out_dir.mkdir(parents=True, exist_ok=True)
    (out_dir / "align.json").write_text(json.dumps(
        {"lines": p["lines"], "chars": chars, "envelope": envelope, "duration": duration,
         "cues": [c.__dict__ for c in cues], "problems": problems}, ensure_ascii=False, indent=1), encoding="utf-8")
    if problems:
        return fail("判卷不过:" + ";".join(problems) + f"(逐字时间见 outputs/tasks/{tid}/align.json)")

    srt = out_dir / "subtitles.srt"
    srt.write_text(subtitles.srt(cues), encoding="utf-8")
    rel = srt.relative_to(root).as_posix()
    set_task_state(store, tid, "queued", external_id=f"align-{tid}", host_id=host.id)
    note = ""
    if repo.get(store, "segment", task["target_id"])["subtitle_path"]:
        note = "subtitle_path 已有字幕,新版本留在任务记录里待人挑选"
    else:
        repo.update(store, "segment", task["target_id"], subtitle_path=rel)
    set_task_state(store, tid, "done", result_path=rel, result_sha256=sha256_file(srt))
    return Outcome("done", note or rel)
```

```python
def _parts(root: Path, videos: list[dict]) -> list[tuple[list[subtitles.Cue], float, Path]]:
    """每段 (字幕, 本段在成片里的起点秒, 本段视频)。起点 = 前面各段视频时长之和(拼接就是按这个接的)。"""

        clip = root / v["video_path"]
        sub = v.get("subtitle_path")
        cues = subtitles.parse_srt((root / sub).read_text(encoding="utf-8")) if sub else []
        parts.append((cues, offset, clip))

def episode_subtitles(root: Path, videos: list[dict]) -> str:
    """整集 SRT;本集没有一句字幕时返回空串(不烧录)。"""

def _subtitle_check(root: Path, videos: list[dict], plain: Path, report_rel: str) -> tuple[bool, str]:
    """整集 SRT 与按各段重拼的逐字相同;第一句字幕的中点,成片底部比原片多出字(真烧进去了)。"""
    parts = _parts(root, videos)
    want = subtitles.episode_srt([(c, o) for c, o, _ in parts]) if any(c for c, _, _ in parts) else ""
    if not want:
        return True, "本集没有台词,不烧字幕"
    saved = root / Path(report_rel).parent / "subtitles.srt"
    if not saved.is_file() or saved.read_text(encoding="utf-8") != want:
        return False, "整集字幕文件缺失,或与各段字幕按段落起点重拼的结果不同"
    cues, offset, clip = next(p for p in parts if p[0])
    mid = (cues[0].start + cues[0].end) / 2
    p = media.probe(plain)
    box = media.diff_bbox(media.gray_frame(plain, offset + mid, p.width, p.height),
                          media.gray_frame(clip, mid, p.width, p.height), p.width)
    if box is None or box[1] < p.height * 0.6:
        return False, f"{offset + mid:.2f}s 处成片底部看不到字幕(差异框 {box})"
    return True, f"{sum(len(c) for c, _, _ in parts)} 句;{offset + mid:.2f}s 处底部有字(差异框 {box})"
```
