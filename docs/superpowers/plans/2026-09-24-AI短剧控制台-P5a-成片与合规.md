# AI 短剧控制台 · P5a 成片与合规 · 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把段落视频变成**可上架**的一集:尾帧衔接 → 拼接成集 → 显式标识 + GB 45438-2025 附录 E 隐式元数据 → 溯源报告 → 读回自检,全部在控制台「成片」面板里操作。

**Architecture:** 新增 `tools/cut/`:`media`(本机 ffmpeg / ffprobe 的薄封装)、`aigc`(附录 E 的拼与验)、`tailframe`(抽最后一帧作下一段首帧,记成创建即完成的 task)、`provenance`(沿 task 表往回追生成链,用文件哈希反证)、`export`(两态:begin 当场核对,run 后台出两版 + 报告 + 7 项读回自检)。库升到 v4(`export` / `setting` 表,触发器锁身份与结束态)。控制台新登记「成片」面板,参考帧面板加尾帧入口,自测加三条链路。

**Tech Stack:** Python 3.14 标准库 · 本机 ffmpeg 4.4(libx264 / aac / drawtext / hash 复用器)· Noto Sans SC(SIL OFL)· 现有 FastAPI / Jinja2 / Playwright。**不加依赖**。

## 合规要求 → 代码

| 要求(出处) | 落在哪 |
|---|---|
| 显式标识含「AI」+「生成」(GB 45438 5.4 b) | `media.LABEL_TEXT = "AI 生成"` |
| 位于起始画面、边或角、≥ 2 s(5.4 c d g) | 左上角、**全程**烧录;自检在 0 s、1 s、中点、末尾逐时刻比两版画面 |
| 字高 ≥ 最短边 5%(5.4 f) | 字号 = 最短边 7.5%;自检量差异框高度、**扣掉描边**再比 |
| 附录 E 整串 7 个键,值只能用 0x21、0x23–0x5B、0x5D–0x7E | `aigc.build` / `aigc.check`;服务提供者编码★必答,只能 ASCII,缺了不导出 |
| 首次写入:传播者 = 生产者,传播编号 = 制作编号(E 注 1) | `aigc.build`;`aigc.check` 逐项核对 |
| 一个文件只保留一份隐式标识(6.1 c) | 导出 `-map_metadata -1` 清空后只写一份;自检「含 AIGC 的字段恰好 1 个」「原有元数据已清空」 |
| ReservedCode1 存完整性信息(E f,F.4) | 视频流 SHA-256(只换封装时不变);自检重算比对 |
| ProduceID 与溯源报告作品编号同一个(设计 §6.8) | `export.produce_id` 写进元数据与报告;报告与 task 表按库重算比对 |
| 另提供无显式标识版(设计 §6.8) | 每次导出同时出 `plain.mp4`(隐式标识照写) |
| 导出的成片记录所用模型(MiniMax H3 许可证 III.3.a、设计 §三) | 元数据 `comment=Powered by MiniMax H3`;报告列出全部模型与地域限制 |

```python
# -*- coding: utf-8 -*-
"""真 ffmpeg 跑(本机 4.4)。每条几百毫秒。"""
import subprocess

import pytest

from tools.cut import media


def _ramp(path, n=10):
    """n 帧 64x64,第 i 帧亮度 = 20·i(无损编码)—— 取到哪一帧,看亮度就知道。"""
    subprocess.run(["ffmpeg", "-v", "error", "-y", "-f", "lavfi", "-i", f"color=black:s=64x64:r=10:d={n / 10}",
                    "-vf", "geq=lum='N*20':cb=128:cr=128", "-c:v", "libx264", "-qp", "0", str(path)], check=True)


def test_extract_last_frame_really_takes_the_last_one(tmp_path):
    _ramp(tmp_path / "ramp.mp4")
    assert media.frame_count(tmp_path / "ramp.mp4") == 10
    media.extract_last_frame(tmp_path / "ramp.mp4", tmp_path / "out" / "last.png")
    px = media.gray_frame(tmp_path / "out" / "last.png", 0, 64, 64)
    # 第 10 帧亮度 180(视频是有限范围),转成全范围 PNG 约 191;第 9 帧(160)会是约 168
    assert abs(sum(px) / len(px) - 191) < 3
```

```python
# -*- coding: utf-8 -*-
"""成片用到的 ffmpeg 操作。只调本机 ffmpeg / ffprobe(4.4 实测),不引依赖。

- 找不到 ffmpeg 报 `ValueError`(给人看的原因,控制台渲染成 400),不假装做完;
- ffmpeg 自己失败报 `CutError`,带 stderr 末尾 —— 不吞、不兜空。
- 显式标识的尺寸按**画面最短边**算(GB 45438-2025 5.4 f:字高不低于最短边 5%):
  字号取 7.5%。真 H3 片实测(字号 6.5% 时):量到的字高是字号的 0.82–0.92 倍(亮背景上白字吃掉一部分),
  最差一帧只比下限多 2.6px —— 所以留到 7.5%;描边另算,检查器会扣掉。
"""
from __future__ import annotations

import json
import math
import shutil
import subprocess
from dataclasses import dataclass
from pathlib import Path

FONT = Path("C:/Windows/Fonts/NotoSansSC-VF.ttf")    # SIL OFL,可渲染进视频
LABEL_TEXT = "AI 生成"                                 # 同时含「AI」与「生成」(5.4 b)
FONT_RATIO = 0.075
MARGIN_RATIO = 0.04


class CutError(RuntimeError):
    """ffmpeg / ffprobe 执行失败。"""


@dataclass(frozen=True)
class Probe:
    width: int
    height: int
    duration: float
    has_audio: bool


@dataclass(frozen=True)
class Label:
    fontsize: int
    border: int
    margin: int

    @classmethod
    def for_size(cls, width: int, height: int) -> "Label":
        short = min(width, height)
        return cls(math.ceil(short * FONT_RATIO), max(2, round(short * 0.004)), round(short * MARGIN_RATIO))


def require_ffmpeg() -> None:
    missing = [t for t in ("ffmpeg", "ffprobe") if shutil.which(t) is None]
    if missing:
        raise ValueError(f"本机找不到 {' / '.join(missing)} —— 成片与抽尾帧都要用它,请先安装并加入 PATH")


def _run(args: list[str], *, binary: bool = False) -> subprocess.CompletedProcess:
    p = subprocess.run(args, capture_output=True, **({} if binary else
                                                      {"text": True, "encoding": "utf-8", "errors": "replace"}))
    if p.returncode != 0:
        err = p.stderr.decode("utf-8", "replace") if binary else p.stderr
        raise CutError(f"{args[0]} 失败(退出码 {p.returncode}):{err.strip()[-800:]}")
    return p


def probe(path: Path) -> Probe:
    p = _run(["ffprobe", "-v", "error", "-show_entries", "stream=codec_type,width,height:format=duration",
              "-of", "json", str(path)])
    doc = json.loads(p.stdout)
    video = [s for s in doc["streams"] if s["codec_type"] == "video"]
    if not video:
        raise CutError(f"{path} 里没有视频流")
    return Probe(video[0]["width"], video[0]["height"], float(doc["format"]["duration"]),
                 any(s["codec_type"] == "audio" for s in doc["streams"]))


def frame_count(path: Path) -> int:
    p = _run(["ffprobe", "-v", "error", "-count_frames", "-select_streams", "v:0",
              "-show_entries", "stream=nb_read_frames", "-of", "csv=p=0", str(path)])
    return int(p.stdout.strip())


def extract_last_frame(video: Path, out_png: Path) -> None:
    """先数帧、再按帧号取最后一帧 —— 取到的**确实**是最后一帧,不靠从末尾倒退的近似。"""
    n = frame_count(video)
    if n < 1:
        raise CutError(f"{video} 一帧都解不出来")
    out_png.parent.mkdir(parents=True, exist_ok=True)
    _run(["ffmpeg", "-v", "error", "-y", "-i", str(video), "-vf", f"select=eq(n\\,{n - 1})", "-vsync", "0",
          "-frames:v", "1", str(out_png)])
    if not out_png.is_file():
        raise CutError(f"ffmpeg 没有写出 {out_png}")


def encode_episode(clips: list[Path], out: Path, label: Label | None) -> None:
    """按顺序拼接(重编码),可选烧录显式标识(左上角、全程)。**清空全部原有元数据**:
    ComfyUI 的 mp4 里嵌着完整工作流(提示词、种子、参考图名),不能带进成片。"""
    n = len(clips)
    ins = [a for c in clips for a in ("-i", str(c))]
    graph = "".join(f"[{i}:v:0][{i}:a:0]" for i in range(n)) + f"concat=n={n}:v=1:a=1[v][a]"
    vout = "[v]"
    if label is not None:
        font = FONT.as_posix().replace(":", "\\:")
        graph += (f";[v]drawtext=fontfile='{font}':text='{LABEL_TEXT}':fontsize={label.fontsize}:fontcolor=white"
                  f":borderw={label.border}:bordercolor=black:x={label.margin}:y={label.margin}[vl]")
        vout = "[vl]"
    out.parent.mkdir(parents=True, exist_ok=True)
    _run(["ffmpeg", "-v", "error", "-y", *ins, "-filter_complex", graph, "-map", vout, "-map", "[a]",
          "-c:v", "libx264", "-preset", "medium", "-crf", "18", "-pix_fmt", "yuv420p",
          "-c:a", "aac", "-b:a", "128k", "-ar", "48000", "-map_metadata", "-1", str(out)])


def stream_sha256(path: Path) -> str:
    """视频流的 SHA-256(按压缩后的数据包算)。只换封装、改元数据时不变 —— 写进 ReservedCode1 保护内容完整性。"""
    p = _run(["ffmpeg", "-v", "error", "-i", str(path), "-map", "0:v:0", "-c", "copy", "-f", "hash",
              "-hash", "sha256", "-"])
    return p.stdout.strip().split("=", 1)[1]


def write_tags(src: Path, dst: Path, tags: dict[str, str]) -> None:
    """只换封装写元数据(不重编码):先清空原有元数据,再只写给定的这几个。"""
    meta = [a for k, v in tags.items() for a in ("-metadata", f"{k}={v}")]
    _run(["ffmpeg", "-v", "error", "-y", "-i", str(src), "-map", "0", "-c", "copy", "-map_metadata", "-1",
          "-movflags", "use_metadata_tags", *meta, str(dst)])


def format_tags(path: Path) -> dict[str, str]:
    p = _run(["ffprobe", "-v", "error", "-show_entries", "format_tags", "-of", "json", str(path)])
    return json.loads(p.stdout).get("format", {}).get("tags", {})


def gray_frame(path: Path, t: float, width: int, height: int) -> bytes:
    """t 秒处的一帧,8 位灰度原始像素。"""
    p = _run(["ffmpeg", "-v", "error", "-ss", f"{t:.3f}", "-i", str(path), "-frames:v", "1",
              "-f", "rawvideo", "-pix_fmt", "gray", "-"], binary=True)
    if len(p.stdout) != width * height:
        raise CutError(f"{path} 在 {t:.2f}s 取帧得到 {len(p.stdout)} 字节,应为 {width * height}")
    return p.stdout


def diff_bbox(a: bytes, b: bytes, width: int, threshold: int = 48, min_run: int = 3) -> tuple[int, int, int, int] | None:
    """两帧灰度差超过阈值的区域外接框 (x0, y0, x1, y1),没有差异返回 None。

    一行 / 一列里至少 min_run 个差异像素才算进框 —— 两次独立编码的同一画面会有孤立噪点
    (实测:1 秒测试图样末帧出现过 1 个差 49 的像素);标识是成片的差异,不受影响。
    行首尾若不足 min_run 会让框略小 —— 只会让字高检查更严,不会放过不合格的。
    """
    height = len(a) // width
    rows, cols = [0] * height, [0] * width
    for y in range(height):
        row = y * width
        for x in range(width):
            if abs(a[row + x] - b[row + x]) > threshold:
                rows[y] += 1
                cols[x] += 1
    ys = [y for y, n in enumerate(rows) if n >= min_run]
    xs = [x for x, n in enumerate(cols) if n >= min_run]
    return (xs[0], ys[0], xs[-1], ys[-1]) if xs and ys else None
```

```python
# -*- coding: utf-8 -*-
"""GB 45438-2025 附录 E(规范性)的文件元数据隐式标识:拼、验。

原文核对见 `docs/superpowers/spike-results/2026-09-24-GB45438原文核对与导出实测.md`。要点:
- 字段值为整串 `{"AIGC":{…7 个键…}}`,键名以**规范性附录 E** 为准(资料性附录 F 的示例拼错了三个键名);
- 各要素值只能由码位 0x21、0x23–0x5B、0x5D–0x7E 的字符构成(E.j)—— 即除空格、`"`、`\\` 外的单字节可见字符;
- 首次写入时,传播者 = 生产者、传播编号 = 制作编号(E 注 1);
- 一个文件只保留一份(6.1 c)。
"""
from __future__ import annotations

import json
import re
import uuid
from datetime import datetime, timezone

TAG = "AIGC"
KEYS = ("Label", "ContentProducer", "ProduceID", "ReservedCode1", "ContentPropagator", "PropagateID",
        "ReservedCode2")
_VALUE = re.compile(r"^[\x21\x23-\x5b\x5d-\x7e]*$")
MAX_PRODUCER = 64


def value_problems(name: str, value: str) -> list[str]:
    if not _VALUE.match(value):
        bad = sorted({c for c in value if not _VALUE.match(c)})
        return [f"{name} 含不允许的字符 {bad} —— 只能用单字节可见字符(不含空格、\" 和 \\)"]
    return []


def producer_problems(producer: str) -> list[str]:
    """服务提供者编码(★必答):由平台运营方提供,如统一社会信用代码。缺了就不导出,不用默认值。"""
    if not producer:
        return ["缺少「生成合成服务提供者编码」(★必答,如统一社会信用代码)—— 没有它不能导出"]
    if len(producer) > MAX_PRODUCER:
        return [f"服务提供者编码太长({len(producer)} 字符,上限 {MAX_PRODUCER})"]
    return value_problems("服务提供者编码", producer)


def new_produce_id(now: datetime | None = None) -> str:
    """内容制作编号:时间 + 随机,ASCII。与溯源报告的作品编号是同一个。"""
    ts = (now or datetime.now(timezone.utc)).strftime("%Y%m%d%H%M%S")
    return f"ASD-{ts}-{uuid.uuid4().hex[:12]}"


def build(producer: str, produce_id: str, reserved1: str) -> str:
    problems = producer_problems(producer) + value_problems("ProduceID", produce_id) + value_problems(
        "ReservedCode1", reserved1)
    if problems:
        raise ValueError(";".join(problems))
    body = {"Label": "1", "ContentProducer": producer, "ProduceID": produce_id, "ReservedCode1": reserved1,
            "ContentPropagator": producer, "PropagateID": produce_id, "ReservedCode2": ""}
    return json.dumps({TAG: body}, ensure_ascii=True, separators=(",", ":"))


def check(tags: dict[str, str], *, producer: str, produce_id: str, reserved1: str) -> list[str]:
    """读回的元数据逐条对附录 E。返回问题清单,空 = 合规。"""
    keys = [k for k in tags if TAG.lower() in k.lower()]
    if len(keys) != 1:
        return [f"含「AIGC」的元数据字段应恰好 1 个(6.1 c),实际 {len(keys)} 个:{keys}"]
    try:
        doc = json.loads(tags[keys[0]])
    except json.JSONDecodeError as e:
        return [f"AIGC 字段不是合法 JSON:{e}"]
    if not isinstance(doc, dict) or list(doc) != [TAG] or not isinstance(doc[TAG], dict):
        return ['AIGC 字段应为 {"AIGC":{…}} 的整串(附录 E b)']
    body = doc[TAG]
    problems = []
    if list(body) != list(KEYS):
        problems.append(f"键应依次为 {list(KEYS)},实际 {list(body)}")
    for k, v in body.items():
        if not isinstance(v, str):
            problems.append(f"{k} 应为字符串,实际 {type(v).__name__}")
        else:
            problems += value_problems(k, v)
    expect = {"Label": "1", "ContentProducer": producer, "ProduceID": produce_id, "ReservedCode1": reserved1,
              "ContentPropagator": producer, "PropagateID": produce_id}
    problems += [f"{k} 应为 {v!r},实际 {body.get(k)!r}" for k, v in expect.items() if body.get(k) != v]
    return problems
```

```python
# -*- coding: utf-8 -*-
"""尾帧衔接(设计 §6.6):本段当前视频的最后一帧 → 下一段的首帧。保证**物理连续**,不是「看起来像」。

- 三种前置条件各给可操作的原因:本段没有视频(或文件不在本机)/ 本段是最后一段 / 本机没有 ffmpeg;
- 抽帧是控制面本地的确定性操作,记成一条**创建即完成**的 task(`tasks.record_local`):
  溯源链能从下一段的视频一路追回到上一段的视频;抽出来的帧也进历史,人可以换回原来的首帧;
- 下一段已有视频时,换首帧会让那段视频作废 → plan/confirm 两态,先报代价再落盘(与「用这一版」同一套)。
"""
from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path

from tools.cut import media
from tools.drama import repo
from tools.drama.db import Store
from tools.drama.invalidation import Impact
from tools.drama.tasks import record_local
from tools.gen.capability import MINIMAX_H3
from tools.serve.dispatch import sha256_file

MODEL = "ffmpeg:last-frame"


@dataclass(frozen=True)
class TailPlan:
    source_id: int
    target_id: int
    video_path: str
    impact: Impact

    def summary(self) -> str:
        return self.impact.summary()


def plan_tail(store: Store, data_root: Path, segment_id: int) -> TailPlan:
    seg = repo.get(store, "segment", segment_id)
    if not seg["video_path"]:
        raise ValueError(f"段落 {seg['number']} 还没有视频 —— 先在「视频」面板生成,再抽尾帧")
    if not (Path(data_root) / seg["video_path"]).is_file():
        raise ValueError(f"段落 {seg['number']} 的视频文件不在本机:{seg['video_path']}")
    nxt = store.conn.execute("SELECT * FROM segment WHERE episode_id = ? AND number > ? ORDER BY number LIMIT 1",
                             (seg["episode_id"], seg["number"])).fetchone()
    if nxt is None:
        raise ValueError(f"段落 {seg['number']} 是本集最后一段,没有下一段可以接")
    media.require_ffmpeg()
    if nxt["video_path"]:
        imp = Impact((nxt["id"],), 0, 1, MINIMAX_H3.est_seconds(MINIMAX_H3.frames_for(nxt["duration_sec"])))
    else:
        imp = Impact((), 0, 0, 0.0)
    return TailPlan(segment_id, nxt["id"], seg["video_path"], imp)


def confirm_tail(store: Store, data_root: Path, plan: TailPlan) -> int:
    root = Path(data_root)
    src = root / plan.video_path
    sha = sha256_file(src)
    rel = f"outputs/tailframes/{sha[:16]}.png"          # 按源视频内容寻址:同一个视频抽几次都是同一张
    if not (root / rel).is_file():
        media.extract_last_frame(src, root / rel)
    number = repo.get(store, "segment", plan.source_id)["number"]
    tid = record_local(store, kind="image", target_type="segment", target_id=plan.target_id, model=MODEL,
                       params={"workflow": "last_frame", "source_segment": plan.source_id,
                               "source_sha256": sha, "writes": "first_frame_path", "expect_ext": ".png"},
                       prompt=f"段落 {number} 视频的最后一帧", refs=[plan.video_path],
                       result_path=rel, result_sha256=sha256_file(root / rel))
    if plan.impact.segment_ids:
        with store.write() as c:
            c.execute("UPDATE segment SET video_path = NULL, subtitle_path = NULL WHERE id = ?", (plan.target_id,))
    repo.update(store, "segment", plan.target_id, first_frame_path=rel)
    return tid
```

```python
# -*- coding: utf-8 -*-
"""溯源报告(设计 §6.9):整集的生成链,平台备案的「AI 工具溯源材料」。

从每段的视频出发,按「产物路径 → 产出它的 task → 它的参考素材」一路往回追:
视频 ← 首帧(合成的 / 从上一段视频抽的)← 角色四视图 / 场景空镜 / 道具图 ← …
每个节点记:哪个模型、什么参数、什么种子、什么提示词、何时在哪台机器生成,以及**文件的 sha256**。

**用数据审**:文件的 sha256 必须等于 task 表里记的产物哈希 —— 不等说明文件被替换过,
报告就不是真的,`problems()` 会列出来,导出据此拒绝。没有生成记录的素材(人工提供)照实标出。
"""
from __future__ import annotations

import json
from pathlib import Path

from tools.drama import repo
from tools.drama.db import Store
from tools.serve.dispatch import sha256_file

LICENSE_NOTE = ("视频由 MiniMax H3 生成(Powered by MiniMax H3)。依 MiniMax H3 Community License,"
                "其输出不得在欧盟、英国、韩国、美国使用、分发或展示;使用限制须逐级传递给下游。")


def _task_for(store: Store, path: str):
    return store.conn.execute("SELECT * FROM task WHERE state = 'done' AND result_path = ? ORDER BY id DESC LIMIT 1",
                              (path,)).fetchone()


def _node(store: Store, root: Path, path: str) -> dict:
    f = root / path
    node = {"path": path, "sha256": sha256_file(f) if f.is_file() else None, "task": None, "inputs": []}
    t = _task_for(store, path)
    if t is None:
        return node
    host = repo.get(store, "host", t["host_id"])["label"] if t["host_id"] is not None else "本机"
    node["task"] = {"id": t["id"], "kind": t["kind"], "model": t["model"], "params": json.loads(t["params_json"]),
                    "seed": t["seed"], "prompt": t["prompt"], "host": host, "external_id": t["external_id"],
                    "created_at": t["created_at"], "finished_at": t["finished_at"],
                    "result_sha256": t["result_sha256"]}
    node["inputs"] = [_node(store, root, r) for r in json.loads(t["refs_json"])]   # 参考素材都早于产物,链无环
    return node


def build(store: Store, data_root: Path, episode_id: int, *, produce_id: str, producer: str,
          videos: list[dict]) -> dict:
    """videos:导出时快照的 [{segment_id, number, video_path}],按顺序。"""
    ep = repo.get(store, "episode", episode_id)
    drama = repo.get(store, "drama", ep["drama_id"])
    segs = []
    for v in videos:
        seg = repo.get(store, "segment", v["segment_id"])
        segs.append({"number": v["number"], "duration_sec": seg["duration_sec"], "description": seg["description"],
                     "video": _node(store, Path(data_root), v["video_path"])})
    return {"produce_id": produce_id, "content_producer": producer, "drama": drama["title"],
            "aspect_ratio": drama["aspect_ratio"], "episode": ep["number"], "license": LICENSE_NOTE,
            "models": sorted(_models([s["video"] for s in segs])), "segments": segs}


def _models(nodes: list[dict]) -> set[str]:
    out = set()
    for n in nodes:
        if n["task"]:
            out.add(n["task"]["model"])
        out |= _models(n["inputs"])
    return out


def _walk(node: dict):
    yield node
    for i in node["inputs"]:
        yield from _walk(i)


def problems(report: dict) -> list[str]:
    out = []
    for s in report["segments"]:
        for n in _walk(s["video"]):
            if n["sha256"] is None:
                out.append(f"段落 {s['number']}:文件不在本机 {n['path']}")
            elif n["task"] and n["task"]["result_sha256"] and n["sha256"] != n["task"]["result_sha256"]:
                out.append(f"段落 {s['number']}:{n['path']} 的内容与 task #{n['task']['id']} 记录的产物哈希不一致"
                           "(文件被替换过)—— 溯源不成立")
        if s["video"]["task"] is None:
            out.append(f"段落 {s['number']}:视频 {s['video']['path']} 在 task 表里没有生成记录")
    return out


def to_markdown(report: dict) -> str:
    lines = [f"# 溯源报告 · {report['drama']} 第 {report['episode']} 集", "",
             f"- 内容制作编号(ProduceID):`{report['produce_id']}`",
             f"- 生成合成服务提供者编码:`{report['content_producer']}`",
             f"- 画幅:{report['aspect_ratio']}", f"- 用到的模型:{'、'.join(report['models'])}",
             f"- 许可证:{report['license']}", ""]

    def emit(node: dict, depth: int) -> None:
        pad = "  " * depth
        t = node["task"]
        if t is None:
            lines.append(f"{pad}- `{node['path']}` —— 无生成记录(人工提供)· sha256 `{(node['sha256'] or '缺失')[:16]}`")
        else:
            lines.append(f"{pad}- `{node['path']}` ← task #{t['id']} · {t['model']} · 种子 {t['seed']} · "
                         f"{t['host']} · {t['finished_at']} · sha256 `{(node['sha256'] or '缺失')[:16]}`")
            lines.append(f"{pad}  提示词:{t['prompt']}")
        for i in node["inputs"]:
            emit(i, depth + 1)

    for s in report["segments"]:
        lines += [f"## 段落 {s['number']}({s['duration_sec']} 秒)", "", s["description"], ""]
        emit(s["video"], 0)
        lines.append("")
    return "\n".join(lines)
```

```python
"""成片导出:拼接 → 显式标识 → 隐式元数据 → 溯源报告 → **读回自检**。

一次导出出两个文件(设计 §6.8):
- `labeled.mp4`:带显式标识「AI 生成」(左上角、**全程**;标准下限是起始画面 ≥ 2 s);
- `plain.mp4`:无显式标识版(办法允许应用户申请提供),隐式标识照样写。
两者共用同一个 ProduceID(同一个作品),各自的 ReservedCode1 是各自视频流的 SHA-256。

两态:`begin` 当场核对前提(★必答的服务提供者编码、每段有视频、尺寸一致、有音轨、溯源成立),
不满足就报原因不开工;`run` 在后台跑(一集 1–3 分钟),结束时整行锁定(触发器)。
**自检不过 = 导出失败**:文件留着给人查原因,但状态是 failed,界面标「不可发布」。

from pathlib import Path

from tools.cut import aigc, media, provenance

from tools.serve.dispatch import sha256_file

COMMENT = "Powered by MiniMax H3"                     # MiniMax H3 许可证 III.3.a,随文件走
ALLOWED_TAGS = {"major_brand", "minor_version", "compatible_brands", "encoder", aigc.TAG, "comment"}


def saved_producer(store: Store) -> str:
    row = store.conn.execute("SELECT value FROM setting WHERE key = ?", (SETTING,)).fetchone()
    return row[0] if row else ""


def get(store: Store, export_id: int):
    row = store.conn.execute("SELECT * FROM export WHERE id = ?", (export_id,)).fetchone()
    if row is None:
        raise KeyError(f"导出 {export_id} 不存在")
    return row


def list_for(store: Store, episode_id: int) -> list:
    return store.conn.execute("SELECT * FROM export WHERE episode_id = ? ORDER BY id", (episode_id,)).fetchall()


def _segments(store: Store, episode_id: int) -> list:
    return store.conn.execute("SELECT * FROM segment WHERE episode_id = ? ORDER BY number", (episode_id,)).fetchall()


def readiness(store: Store, data_root: Path, episode_id: int) -> list[str]:
    """面板上的轻量检查(不读视频内容):缺什么。空 = 可以点导出。"""
    segs = _segments(store, episode_id)
    problems = [] if segs else ["这一集还没有段落"]
    for s in segs:
        if not s["video_path"]:
            problems.append(f"段落 {s['number']} 还没有视频")
        elif not (Path(data_root) / s["video_path"]).is_file():
            problems.append(f"段落 {s['number']} 的视频文件不在本机:{s['video_path']}")
    try:
        media.require_ffmpeg()
    except ValueError as e:
        problems.append(str(e))
    if not media.FONT.is_file():
        problems.append(f"找不到显式标识用的字体 {media.FONT}(Noto Sans SC,SIL OFL)")
    return problems


def begin(store: Store, data_root: Path, episode_id: int, producer: str) -> int:
    root = Path(data_root)
    producer = producer.strip()
    problems = aigc.producer_problems(producer) + readiness(store, root, episode_id)
    if problems:
        raise ValueError(";".join(problems))
    segs = _segments(store, episode_id)
    probes = {s["number"]: media.probe(root / s["video_path"]) for s in segs}
    sizes = {(p.width, p.height) for p in probes.values()}
    if len(sizes) != 1:
        problems.append(f"各段视频尺寸不一致:{ {n: (p.width, p.height) for n, p in probes.items()} }")
    problems += [f"段落 {n} 的视频没有音轨" for n, p in probes.items() if not p.has_audio]
    videos = [{"segment_id": s["id"], "number": s["number"], "video_path": s["video_path"],
               "sha256": sha256_file(root / s["video_path"])} for s in segs]
    problems += provenance.problems(provenance.build(store, root, episode_id, produce_id="", producer=producer,
                                                     videos=videos))
    if problems:
        raise ValueError(";".join(problems))
    with store.write() as c:
        c.execute("INSERT INTO setting (key, value) VALUES (?, ?) ON CONFLICT(key) DO UPDATE SET value = excluded.value",
                  (SETTING, producer))
        return c.execute("INSERT INTO export (episode_id, produce_id, content_producer, inputs_json, created_at)"
                         " VALUES (?, ?, ?, ?, ?)",
                         (episode_id, aigc.new_produce_id(), producer, json.dumps(videos, ensure_ascii=False),
                          now())).lastrowid


def _finish(store: Store, export_id: int, state: str, **fields) -> None:
    sets = {"state": state, "finished_at": now(), **fields}
    with store.write() as c:
        c.execute(f"UPDATE export SET {', '.join(f'{k} = ?' for k in sets)} WHERE id = ?",
                  (*sets.values(), export_id))


def fail(store: Store, export_id: int, error: str) -> None:
    if not error.strip():
        raise ValueError("失败必须写明原因")
    _finish(store, export_id, "failed", error=error)


def fail_stale(store: Store) -> int:
    """控制台重启时,上次没跑完的导出不会再有人接着跑:标失败,别让它永远显示「导出中」。"""
    ids = [r[0] for r in store.conn.execute("SELECT id FROM export WHERE state = 'running'")]
    for i in ids:
        fail(store, i, "控制台重启,这次导出中断了(产物不完整,不可发布)—— 请重新导出")
    return len(ids)


def run(store: Store, data_root: Path, export_id: int) -> None:
    row = get(store, export_id)
    root = Path(data_root)
    videos = json.loads(row["inputs_json"])
    clips = [root / v["video_path"] for v in videos]
    changed = [v["number"] for v, c in zip(videos, clips) if sha256_file(c) != v["sha256"]]
    if changed:
        raise media.CutError(f"段落 {changed} 的视频在点导出之后被换掉了 —— 请重新导出")
    out = Path("outputs/exports") / row["produce_id"]
    first = media.probe(clips[0])
    label = media.Label.for_size(first.width, first.height)
    rel = {"labeled": (out / "labeled.mp4").as_posix(), "plain": (out / "plain.mp4").as_posix()}
    for key, lab in (("plain", None), ("labeled", label)):
        raw = root / out / f"{key}.raw.mp4"
        media.encode_episode(clips, raw, lab)
        tag = aigc.build(row["content_producer"], row["produce_id"], media.stream_sha256(raw))
        media.write_tags(raw, root / rel[key], {aigc.TAG: tag, "comment": COMMENT})
        raw.unlink()
    report = provenance.build(store, root, row["episode_id"], produce_id=row["produce_id"],
                              producer=row["content_producer"], videos=videos)
    report_rel = (out / "provenance.json").as_posix()
    (root / report_rel).write_text(json.dumps(report, ensure_ascii=False, indent=2), encoding="utf-8")
    (root / out / "provenance.md").write_text(provenance.to_markdown(report), encoding="utf-8")
    checks = check(store, root, row, rel["labeled"], rel["plain"], report_rel, label)
    bad = [c["name"] for c in checks if not c["ok"]]
    _finish(store, export_id, "failed" if bad else "done", labeled_path=rel["labeled"], plain_path=rel["plain"],
            report_path=report_rel, checks_json=json.dumps(checks, ensure_ascii=False),
            error=f"自检不通过:{'、'.join(bad)}(不可发布)" if bad else None)


def _label_check(labeled: Path, plain: Path, label: media.Label) -> tuple[bool, str]:
    """显式标识:逐时刻比对两版画面,差异框 = 标识。要求:每个时刻都在、在左上角、字高(扣掉描边)≥ 最短边 5%。"""
    p = media.probe(labeled)
    w, h, short = p.width, p.height, min(p.width, p.height)
    times = sorted({0.0, min(1.0, p.duration / 2), p.duration / 2, max(0.0, p.duration - 0.1)})
    notes = []
    for t in times:
        box = media.diff_bbox(media.gray_frame(labeled, t, w, h), media.gray_frame(plain, t, w, h), w)
        if box is None:
            return False, f"{t:.1f}s 处看不到标识"
        x0, y0, x1, y1 = box
        glyph = (y1 - y0 + 1) - 2 * label.border
        if glyph < 0.05 * short:
            return False, f"{t:.1f}s 处字高 {glyph}px < 最短边 5%({0.05 * short:.1f}px)"
        if x1 >= w / 2 or y1 >= h / 4:
            return False, f"{t:.1f}s 处标识不在左上角:差异框 {box}"
        notes.append(f"{t:.1f}s 字高 {glyph}px")
    return True, f"{'、'.join(notes)}(最短边 5% = {0.05 * short:.1f}px)"


def check(store: Store, root: Path, row, labeled_rel: str, plain_rel: str, report_rel: str,
          label: media.Label) -> list[dict]:
    """读回自检。每项 PASS / FAIL 带依据;不看代码怎么写的,只看文件里实际有什么。"""
    out = []

    def add(name: str, ok: bool, detail: str) -> None:
        out.append({"name": name, "ok": ok, "detail": detail})

    for key, rel in (("带标识版", labeled_rel), ("无标识版", plain_rel)):
        f = root / rel
        tags = media.format_tags(f)
        probs = aigc.check(tags, producer=row["content_producer"], produce_id=row["produce_id"],
                           reserved1=media.stream_sha256(f))
        add(f"隐式标识(附录 E)· {key}", not probs, ";".join(probs) or "7 个键齐全,注 1 一致,ReservedCode1 = 视频流哈希")
        extra = sorted(set(tags) - ALLOWED_TAGS)
        add(f"原有元数据已清空 · {key}", not extra, f"多出 {extra}" if extra else f"只有 {sorted(tags)}")
    ok, detail = _label_check(root / labeled_rel, root / plain_rel, label)
    add("显式标识(5.4)", ok, detail)
    videos = json.loads(row["inputs_json"])
    want = sum(media.probe(root / v["video_path"]).duration for v in videos)
    got = media.probe(root / labeled_rel).duration
    tol = 0.1 * len(videos) + 0.1
    add("时长 = 各段之和", abs(got - want) <= tol, f"成片 {got:.2f}s,各段合计 {want:.2f}s(容差 {tol:.1f}s)")
    saved = json.loads((root / report_rel).read_text(encoding="utf-8"))
    fresh = provenance.build(store, root, row["episode_id"], produce_id=row["produce_id"],
                             producer=row["content_producer"], videos=videos)
    probs = provenance.problems(saved)
    same = saved == fresh
    add("溯源报告与 task 表一致", same and not probs,
        ";".join(probs) or ("报告按库重算逐项相同,每个文件哈希与 task 记录一致" if same else "报告与按库重算的结果不同"))
    return out
```
