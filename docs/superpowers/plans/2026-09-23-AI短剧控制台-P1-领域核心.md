```python
# -*- coding: utf-8 -*-
"""SQLite 存储 —— 控制面的唯一真相源。只用标准库 sqlite3。

几条业务硬规则直接写成数据库约束和触发器,而不是靠调用方自觉:
- 画幅建后不可改(改画幅等于全部参考帧和视频作废)
- task 的生成参数不可变(溯源的前提;要重跑就新建一条)
- 外键开启(段落绑到不存在的场景要当场失败)

写操作一律走 Store.write() 的单锁 —— 设计 M-21「落盘走单线程」,从根上消灭并发覆盖。
"""
from __future__ import annotations

import sqlite3
import threading
from contextlib import contextmanager
from datetime import datetime, timezone
from pathlib import Path

SCHEMA_VERSION = 1

SCHEMA = """
CREATE TABLE meta (key TEXT PRIMARY KEY, value TEXT NOT NULL);

CREATE TABLE drama (
  id INTEGER PRIMARY KEY,
  title TEXT NOT NULL,
  aspect_ratio TEXT NOT NULL CHECK (aspect_ratio IN ('9:16', '16:9')),
  style TEXT NOT NULL DEFAULT '',
  created_at TEXT NOT NULL
);
CREATE TRIGGER drama_aspect_immutable BEFORE UPDATE OF aspect_ratio ON drama
BEGIN SELECT RAISE(ABORT, '画幅建后不可改 —— 改画幅等于全部参考帧和视频作废,请新建剧'); END;

CREATE TABLE episode (
  id INTEGER PRIMARY KEY,
  drama_id INTEGER NOT NULL REFERENCES drama(id),
  number INTEGER NOT NULL,
  source_type TEXT NOT NULL CHECK (source_type IN ('novel', 'synopsis', 'script', 'footage')),
  source_text TEXT NOT NULL DEFAULT '',
  outline TEXT NOT NULL DEFAULT '',
  script TEXT NOT NULL DEFAULT '',
  UNIQUE (drama_id, number)
);

CREATE TABLE character (
  id INTEGER PRIMARY KEY,
  drama_id INTEGER NOT NULL REFERENCES drama(id),
  name TEXT NOT NULL,
  appearance TEXT NOT NULL DEFAULT '',
  styling TEXT NOT NULL DEFAULT '',
  voice_desc TEXT NOT NULL DEFAULT '',
  final_prompt TEXT NOT NULL DEFAULT '',
  seed INTEGER,
  turnaround_path TEXT,
  voice_ref_path TEXT,
  library_id INTEGER,
  UNIQUE (drama_id, name)
);

CREATE TABLE scene (
  id INTEGER PRIMARY KEY,
  drama_id INTEGER NOT NULL REFERENCES drama(id),
  episode_id INTEGER REFERENCES episode(id),
  name TEXT NOT NULL,
  location TEXT NOT NULL DEFAULT '',
  time TEXT NOT NULL DEFAULT '',
  lighting TEXT NOT NULL DEFAULT '',
  final_prompt TEXT NOT NULL DEFAULT '',
  image_path TEXT,
  library_id INTEGER,
  UNIQUE (drama_id, name)
);

CREATE TABLE prop (
  id INTEGER PRIMARY KEY,
  drama_id INTEGER NOT NULL REFERENCES drama(id),
  name TEXT NOT NULL,
  description TEXT NOT NULL DEFAULT '',
  final_prompt TEXT NOT NULL DEFAULT '',
  image_path TEXT,
  library_id INTEGER,
  UNIQUE (drama_id, name)
);

CREATE TABLE segment (
  id INTEGER PRIMARY KEY,
  episode_id INTEGER NOT NULL REFERENCES episode(id),
  number INTEGER NOT NULL,
  scene_id INTEGER REFERENCES scene(id),
  duration_sec REAL,
  shot_type TEXT NOT NULL DEFAULT '',
  angle TEXT NOT NULL DEFAULT '',
  movement TEXT NOT NULL DEFAULT '',
  description TEXT NOT NULL DEFAULT '',
  result TEXT NOT NULL DEFAULT '',
  atmosphere TEXT NOT NULL DEFAULT '',
  layout TEXT NOT NULL DEFAULT '',
  image_prompt TEXT NOT NULL DEFAULT '',
  video_prompt TEXT NOT NULL DEFAULT '',
  bgm_prompt TEXT NOT NULL DEFAULT '',
  sfx_prompt TEXT NOT NULL DEFAULT '',
  reference_mode TEXT NOT NULL DEFAULT 'first' CHECK (reference_mode IN
    ('first', 'last', 'key', 'first_last', 'first_last_key', 'text_only', 'anchors')),
  first_frame_path TEXT,
  last_frame_path TEXT,
  video_path TEXT,
  subtitle_path TEXT,
  UNIQUE (episode_id, number)
);

CREATE TABLE segment_character (
  segment_id INTEGER NOT NULL REFERENCES segment(id) ON DELETE CASCADE,
  character_id INTEGER NOT NULL REFERENCES character(id),
  position INTEGER NOT NULL,
  PRIMARY KEY (segment_id, character_id)
);

CREATE TABLE segment_prop (
  segment_id INTEGER NOT NULL REFERENCES segment(id) ON DELETE CASCADE,
  prop_id INTEGER NOT NULL REFERENCES prop(id),
  position INTEGER NOT NULL,
  PRIMARY KEY (segment_id, prop_id)
);

CREATE TABLE host (
  id INTEGER PRIMARY KEY,
  label TEXT NOT NULL,
  ssh_host TEXT NOT NULL,
  ssh_port INTEGER NOT NULL,
  ssh_user TEXT NOT NULL,
  key_ref TEXT NOT NULL,
  local_port INTEGER NOT NULL,
  remote_comfy_port INTEGER NOT NULL DEFAULT 8188,
  vram_gb REAL NOT NULL
);

CREATE TABLE task (
  id INTEGER PRIMARY KEY,
  kind TEXT NOT NULL CHECK (kind IN ('image', 'video', 'align')),
  target_type TEXT NOT NULL CHECK (target_type IN ('segment', 'character', 'scene', 'prop')),
  target_id INTEGER NOT NULL,
  model TEXT NOT NULL,
  params_json TEXT NOT NULL,
  seed INTEGER NOT NULL,
  prompt TEXT NOT NULL,
  refs_json TEXT NOT NULL,
  created_at TEXT NOT NULL,
  host_id INTEGER REFERENCES host(id),
  external_id TEXT,
  state TEXT NOT NULL DEFAULT 'pending' CHECK (state IN
    ('pending', 'queued', 'running', 'waiting_host', 'done', 'failed')),
  result_path TEXT,
  result_sha256 TEXT,
  error TEXT,
  started_at TEXT,
  finished_at TEXT
);
CREATE TRIGGER task_params_immutable
BEFORE UPDATE OF kind, target_type, target_id, model, params_json, seed, prompt, refs_json, created_at ON task
BEGIN SELECT RAISE(ABORT, 'task 的生成参数不可变 —— 这是溯源的前提;要重跑请新建一条 task'); END;
"""


def now() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds")


class Store:
    def __init__(self, path: str | Path = ":memory:"):
        self.conn = sqlite3.connect(str(path), check_same_thread=False, isolation_level=None)
        self.conn.row_factory = sqlite3.Row
        self.conn.execute("PRAGMA foreign_keys = ON")
        if str(path) != ":memory:":
            self.conn.execute("PRAGMA journal_mode = WAL")
        self._lock = threading.Lock()
        try:
            self._migrate()
        except BaseException:
            self.conn.close()  # 版本不符时别留着文件句柄(Windows 上会锁住数据库文件)
            raise

    def _migrate(self) -> None:
        has_meta = self.conn.execute(
            "SELECT 1 FROM sqlite_master WHERE type = 'table' AND name = 'meta'").fetchone()
        if not has_meta:
            self.conn.executescript(SCHEMA)
            self.conn.execute("INSERT INTO meta (key, value) VALUES ('schema_version', ?)",
                              (str(SCHEMA_VERSION),))
            return
        row = self.conn.execute("SELECT value FROM meta WHERE key = 'schema_version'").fetchone()
        if row is None or int(row[0]) != SCHEMA_VERSION:
            found = row[0] if row else "缺失"
            raise RuntimeError(
                f"数据库结构版本 {found} 与代码 {SCHEMA_VERSION} 不符 —— 不许在不匹配的结构上继续写")

    @contextmanager
    def write(self):
        """单写锁 + 事务。块内抛异常整体回滚。"""
        with self._lock:
            self.conn.execute("BEGIN IMMEDIATE")
            try:
                yield self.conn
            except BaseException:
                self.conn.execute("ROLLBACK")
                raise
            else:
                self.conn.execute("COMMIT")

    def close(self) -> None:
        self.conn.close()
```

```python
# -*- coding: utf-8 -*-
"""剧 / 集 / 资产 / 段落的读写。

**写入受控**(CLAUDE.md LLM 纪律「白名单写入范围」):每张表只允许写声明过的字段,
写别的字段直接抛错 —— 将来 LLM 产出的结构化数据经这里落盘,越界要当场失败。

跨剧绑定当场拒绝:外键只能保证「资产存在」,保证不了「资产属于这部剧」。
"""
from __future__ import annotations

import sqlite3

from tools.drama.db import Store, now

# 各表允许调用方写入的字段。主键、外键归属、创建时间由本模块自己管。
_COLUMNS: dict[str, set[str]] = {
    "episode": {"source_text", "outline", "script"},
    "character": {"appearance", "styling", "voice_desc", "final_prompt", "seed",
                  "turnaround_path", "voice_ref_path"},
    "scene": {"episode_id", "location", "time", "lighting", "final_prompt", "image_path"},
    "prop": {"description", "final_prompt", "image_path"},
    # scene_id 不在这里:改场景要走 set_scene(),它会核对归属
    "segment": {"duration_sec", "shot_type", "angle", "movement", "description", "result",
                "atmosphere", "layout", "image_prompt", "video_prompt", "bgm_prompt", "sfx_prompt",
                "reference_mode", "first_frame_path", "last_frame_path", "video_path", "subtitle_path"},
}
_READABLE = {"drama", "episode", "character", "scene", "prop", "segment", "task", "host"}


def _check_cols(table: str, fields: dict) -> None:
    bad = set(fields) - _COLUMNS[table]
    if bad:
        raise ValueError(f"{table} 不允许写入字段:{sorted(bad)}")


def _insert(store: Store, table: str, fields: dict) -> int:
    cols = ", ".join(fields)
    marks = ", ".join("?" * len(fields))
    with store.write() as c:
        return c.execute(f"INSERT INTO {table} ({cols}) VALUES ({marks})", tuple(fields.values())).lastrowid


def get(store: Store, table: str, row_id: int) -> sqlite3.Row:
    if table not in _READABLE:
        raise ValueError(f"未知表 {table}")
    row = store.conn.execute(f"SELECT * FROM {table} WHERE id = ?", (row_id,)).fetchone()
    if row is None:
        raise KeyError(f"{table} {row_id} 不存在")
    return row


def update(store: Store, table: str, row_id: int, **fields) -> None:
    if table not in _COLUMNS:
        raise ValueError(f"表 {table} 不走通用更新")
    _check_cols(table, fields)
    if not fields:
        return
    sets = ", ".join(f"{k} = ?" for k in fields)
    with store.write() as c:
        n = c.execute(f"UPDATE {table} SET {sets} WHERE id = ?", (*fields.values(), row_id)).rowcount
    if n == 0:
        raise KeyError(f"{table} {row_id} 不存在")


# ── 创建 ──────────────────────────────────────────────────────

def create_drama(store: Store, title: str, aspect_ratio: str, style: str = "") -> int:
    return _insert(store, "drama", {"title": title, "aspect_ratio": aspect_ratio,
                                    "style": style, "created_at": now()})


def add_episode(store: Store, drama_id: int, number: int, source_type: str, **fields) -> int:
    _check_cols("episode", fields)
    return _insert(store, "episode", {"drama_id": drama_id, "number": number,
                                      "source_type": source_type, **fields})


def add_character(store: Store, drama_id: int, name: str, **fields) -> int:
    _check_cols("character", fields)
    return _insert(store, "character", {"drama_id": drama_id, "name": name, **fields})


def add_scene(store: Store, drama_id: int, name: str, **fields) -> int:
    _check_cols("scene", fields)
    return _insert(store, "scene", {"drama_id": drama_id, "name": name, **fields})


def add_prop(store: Store, drama_id: int, name: str, **fields) -> int:
    _check_cols("prop", fields)
    return _insert(store, "prop", {"drama_id": drama_id, "name": name, **fields})


def add_segment(store: Store, episode_id: int, number: int, scene_id: int | None = None, **fields) -> int:
    _check_cols("segment", fields)
    if scene_id is not None:
        _check_same_drama(store.conn, _drama_of_episode(store.conn, episode_id), "scene", scene_id)
    return _insert(store, "segment", {"episode_id": episode_id, "number": number,
                                      "scene_id": scene_id, **fields})


# ── 归属与绑定 ────────────────────────────────────────────────

def _drama_of_episode(c: sqlite3.Connection, episode_id: int) -> int:
    row = c.execute("SELECT drama_id FROM episode WHERE id = ?", (episode_id,)).fetchone()
    if row is None:
        raise KeyError(f"episode {episode_id} 不存在")
    return row[0]


def _drama_of_segment(c: sqlite3.Connection, segment_id: int) -> int:
    row = c.execute("SELECT e.drama_id FROM segment s JOIN episode e ON e.id = s.episode_id "
                    "WHERE s.id = ?", (segment_id,)).fetchone()
    if row is None:
        raise KeyError(f"segment {segment_id} 不存在")
    return row[0]


def _check_same_drama(c: sqlite3.Connection, drama_id: int, table: str, row_id: int) -> None:
    row = c.execute(f"SELECT drama_id FROM {table} WHERE id = ?", (row_id,)).fetchone()
    if row is None:
        raise KeyError(f"{table} {row_id} 不存在")
    if row[0] != drama_id:
        raise ValueError(f"{table} {row_id} 属于另一部剧,不能绑到这里")


def set_scene(store: Store, segment_id: int, scene_id: int | None) -> None:
    with store.write() as c:
        if scene_id is not None:
            _check_same_drama(c, _drama_of_segment(c, segment_id), "scene", scene_id)
        c.execute("UPDATE segment SET scene_id = ? WHERE id = ?", (scene_id, segment_id))


def _bind(store: Store, segment_id: int, table: str, link: str, col: str, ids: list[int]) -> None:
    with store.write() as c:
        drama = _drama_of_segment(c, segment_id)
        for i in ids:
            _check_same_drama(c, drama, table, i)
        c.execute(f"DELETE FROM {link} WHERE segment_id = ?", (segment_id,))
        c.executemany(f"INSERT INTO {link} (segment_id, {col}, position) VALUES (?, ?, ?)",
                      [(segment_id, i, pos) for pos, i in enumerate(ids)])


def bind_characters(store: Store, segment_id: int, character_ids: list[int]) -> None:
    """整体替换本段落的角色绑定。列表顺序即绑定顺序。"""
    _bind(store, segment_id, "character", "segment_character", "character_id", character_ids)


def bind_props(store: Store, segment_id: int, prop_ids: list[int]) -> None:
    """整体替换本段落的道具绑定。列表顺序即绑定顺序。"""
    _bind(store, segment_id, "prop", "segment_prop", "prop_id", prop_ids)


def segment_assets(store: Store, segment_id: int) -> dict:
    """本段落绑定的资产:{"characters": [...], "scene": Row|None, "props": [...]},按绑定顺序。"""
    c = store.conn
    seg = get(store, "segment", segment_id)
    chars = c.execute("SELECT ch.* FROM segment_character sc JOIN character ch ON ch.id = sc.character_id "
                      "WHERE sc.segment_id = ? ORDER BY sc.position", (segment_id,)).fetchall()
    props = c.execute("SELECT p.* FROM segment_prop sp JOIN prop p ON p.id = sp.prop_id "
                      "WHERE sp.segment_id = ? ORDER BY sp.position", (segment_id,)).fetchall()
    scene = get(store, "scene", seg["scene_id"]) if seg["scene_id"] is not None else None
    return {"characters": chars, "scene": scene, "props": props}
```

```python
# -*- coding: utf-8 -*-
"""生成任务的生命周期。

- **生成参数不可变**(数据库触发器兜底):模型、参数、种子、提示词、参考素材一经创建就定死。
  重跑就新建一条 —— 这同时白送了「历史版本不覆盖,人可以挑」。
- **种子必须显式**:不能复现的生成等于没做。
- **状态只能按表迁移**;失败必须写原因,完成必须有产物 —— 不许静默失败、不许空手宣布完成。
- `waiting_host` 不是失败:机器不可达时任务还在,机器回来就接着走。
"""
from __future__ import annotations

import json
import sqlite3

from tools.drama.db import Store, now

TRANSITIONS: dict[str, set[str]] = {
    "pending": {"queued", "failed"},
    "queued": {"running", "waiting_host", "done", "failed"},
    "running": {"done", "failed", "waiting_host"},
    "waiting_host": {"queued", "running", "failed"},
    "done": set(),
    "failed": set(),
}


def create_task(store: Store, *, kind: str, target_type: str, target_id: int, model: str,
                params: dict, seed: int, prompt: str, refs: list[str]) -> int:
    if not isinstance(seed, int) or isinstance(seed, bool):
        raise TypeError("seed 必须是显式整数 —— 不能复现的生成等于没做")
    with store.write() as c:
        return c.execute(
            "INSERT INTO task (kind, target_type, target_id, model, params_json, seed, prompt,"
            " refs_json, created_at) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)",
            (kind, target_type, target_id, model,
             json.dumps(params, ensure_ascii=False, sort_keys=True), seed, prompt,
             json.dumps(list(refs), ensure_ascii=False), now())).lastrowid


def set_task_state(store: Store, task_id: int, state: str, *, external_id: str | None = None,
                   host_id: int | None = None, result_path: str | None = None,
                   result_sha256: str | None = None, error: str | None = None) -> None:
    with store.write() as c:
        row = c.execute("SELECT state, started_at FROM task WHERE id = ?", (task_id,)).fetchone()
        if row is None:
            raise KeyError(f"task {task_id} 不存在")
        cur = row["state"]
        if state not in TRANSITIONS[cur]:
            raise ValueError(f"非法状态迁移:{cur} → {state}")
        if state == "failed" and not (error or "").strip():
            raise ValueError("失败必须写明原因(error),不许静默失败")
        if state == "done" and not result_path:
            raise ValueError("完成必须有产物路径(result_path)")

        sets: dict = {"state": state}
        for key, val in (("external_id", external_id), ("host_id", host_id),
                         ("result_path", result_path), ("result_sha256", result_sha256),
                         ("error", error)):
            if val is not None:
                sets[key] = val
        if state == "running" and row["started_at"] is None:
            sets["started_at"] = now()
        if state in ("done", "failed"):
            sets["finished_at"] = now()
        c.execute(f"UPDATE task SET {', '.join(f'{k} = ?' for k in sets)} WHERE id = ?",
                  (*sets.values(), task_id))


def tasks_for(store: Store, target_type: str, target_id: int) -> list[sqlite3.Row]:
    """某个目标的全部生成记录,旧的在前 —— 历史版本从这里挑。"""
    return store.conn.execute("SELECT * FROM task WHERE target_type = ? AND target_id = ? ORDER BY id",
                              (target_type, target_id)).fetchall()
```

```python
# -*- coding: utf-8 -*-
"""分镜的可量化规则 —— 由代码判卷,不交给 LLM 自觉。

规则来自行业实践(huobao-drama 分镜规范),但 huobao 只把它们写在提示词里,
代码侧 `duration = LLM给的 || 10` 静默填空。这里把它们落成断言(设计 §6.3):

  总时长       = 剧本字数 ÷ 500 字/分钟
  段落数       ≈ 总时长 ÷ 12 秒, ±20%
  段落时长     ∈ 视频后端 Capability 声明的训练区间(H3: 124–362 帧)
  段落时长下限 = 台词字数 ÷ 4.5 字/秒 + 2 秒表演余量
  子镜头数     ∈ [2, 4](按 description 里的【镜头N】计)

description 的写法约定:按【镜头1】【镜头2】…逐子镜头写;
台词写成「角色名说：「台词」」,旁白写成「旁白：内容」。
"""
from __future__ import annotations

import math
import re

from tools.gen.capability import Capability

CHARS_PER_MINUTE = 500
SEGMENT_TARGET_SEC = 12
SEGMENT_COUNT_TOLERANCE = 0.2
SPEECH_CHARS_PER_SEC = 4.5
PERFORMANCE_PAD_SEC = 2.0
SUBSHOTS_MIN, SUBSHOTS_MAX = 2, 4

_PUNCT = set("，。！？、；：,.!?;:\"'“”‘’「」『』（）()【】《》…—- \t\r\n")
_SPEECH = re.compile(r"说[：:]\s*「([^」]*)」")
_SPEAKER = re.compile(r"([^\s【】：:「」，。！？]+?)说[：:]\s*「([^」]*)」")
_NARRATION = re.compile(r"旁白[：:]\s*([^【\n]*)")
_SUBSHOT = re.compile(r"【镜头\s*\d+】")


def count_chars(text: str) -> int:
    """计字数:去掉标点和空白。"""
    return sum(1 for ch in text if ch not in _PUNCT)


def dialogue_chars(description: str) -> int:
    speech = sum(count_chars(s) for s in _SPEECH.findall(description))
    narration = sum(count_chars(s) for s in _NARRATION.findall(description))
    return speech + narration


def dialogue_by_speaker(description: str) -> dict[str, int]:
    """每个说话人的台词字数。说话人名取「说：」前面的那串字,调用方按已知角色名前缀匹配。"""
    out: dict[str, int] = {}
    for name, line in _SPEAKER.findall(description):
        out[name] = out.get(name, 0) + count_chars(line)
    return out


def min_duration_sec(description: str) -> float:
    return dialogue_chars(description) / SPEECH_CHARS_PER_SEC + PERFORMANCE_PAD_SEC


def subshot_count(description: str) -> int:
    return len(_SUBSHOT.findall(description))


def check_segment(description: str, duration_sec: float | None, cap: Capability) -> list[str]:
    """返回问题清单,空表示通过。每条都是给人看的话。"""
    problems: list[str] = []
    if duration_sec is None:
        problems.append("缺少时长 —— 不许用默认值填空")
    else:
        problems += cap.check_duration(duration_sec)
        need = min_duration_sec(description)
        if duration_sec < need:
            problems.append(f"台词 {dialogue_chars(description)} 字需 ≥{need:.1f} 秒,当前 {duration_sec:g} 秒")
    n = subshot_count(description)
    if not SUBSHOTS_MIN <= n <= SUBSHOTS_MAX:
        problems.append(f"子镜头 {n} 个,应为 {SUBSHOTS_MIN}–{SUBSHOTS_MAX} 个(按【镜头N】计)")
    return problems


def expected_segment_range(script: str) -> tuple[int, int]:
    total_sec = count_chars(script) / CHARS_PER_MINUTE * 60
    n = total_sec / SEGMENT_TARGET_SEC
    lo = max(1, math.floor(n * (1 - SEGMENT_COUNT_TOLERANCE)))
    hi = max(1, math.ceil(n * (1 + SEGMENT_COUNT_TOLERANCE)))
    return lo, hi


def check_episode(script: str, n_segments: int) -> list[str]:
    lo, hi = expected_segment_range(script)
    if lo <= n_segments <= hi:
        return []
    total_sec = count_chars(script) / CHARS_PER_MINUTE * 60
    return [f"剧本 {count_chars(script)} 字(约 {total_sec:.0f} 秒),段落数应在 {lo}–{hi},当前 {n_segments}"]
```

```python
    def est_seconds(self, frames: int, steps: int | None = None) -> float:
        """按实测点估算一条视频的生成秒数。

        每步耗时随帧数**超线性**增长(注意力随序列长度),所以两点之间按幂律
        s = s0 · (f/f0)^k 插值/外推,k 由两个实测点反解。只有一个点时按线性。
        这是估算 —— 界面上必须标「估算」。
        """
        if not self.timing_points:
            raise ValueError(f"{self.name} 没有实测耗时点,不能估算")
        pts = sorted(self.timing_points)
        steps = steps or self.default_steps
        f0, s0 = pts[0]
        if len(pts) == 1:
            k = 1.0
        else:
            f1, s1 = pts[-1]
            k = math.log(s1 / s0) / math.log(f1 / f0)
        return steps * s0 * (frames / f0) ** k
```

```python
# -*- coding: utf-8 -*-
"""提示词里的 `@名字` 引用,以及参考图槽位分配。

「模型不会自己知道哪张图管什么」(设计 §6.1):人和 LLM 在提示词里写 `@林小雨`,
提交生成前再按槽位展开成编号标签(见 tools/drama/tags.py)。

槽位是硬约束:出首帧用的 Qwen-Image-Edit 只有 3 个图槽。
「角色四视图 + 场景空镜 + 道具白底」正好占满,两个角色同框就放不下。
闸门 0-3 实测:出槽的道具只靠文字描述,**设计必然丢失**(雕龙淡青玉 → 通用绿色团块)。
所以出槽的素材必须报出来,不许静默丢弃。
"""
from __future__ import annotations

import re
from dataclasses import dataclass

_UNKNOWN_FRAGMENT = re.compile(r"[^\s@，。,.!?！？：:、「」]{1,12}")


@dataclass(frozen=True)
class Ref:
    kind: str          # character / scene / prop
    name: str
    path: str | None   # 参考图路径;缺图时为 None


def find_mentions(prompt: str, names: list[str]) -> tuple[list[str], list[str]]:
    """返回 (按出现顺序去重的已知名字, 对不上的片段)。

    名字取「@ 之后最长的已知名字前缀」:`@林小雨推开门` 要识别成「林小雨」,
    不能把「推开门」吞进名字;同时有「林小」和「林小雨」时取后者。
    """
    by_len = sorted(set(names), key=len, reverse=True)
    found: list[str] = []
    unknown: list[str] = []
    i = 0
    while (j := prompt.find("@", i)) >= 0:
        rest = prompt[j + 1:]
        hit = next((n for n in by_len if rest.startswith(n)), None)
        if hit is None:
            m = _UNKNOWN_FRAGMENT.match(rest)
            unknown.append(m.group(0) if m else "")
            i = j + 1
            continue
        if hit not in found:
            found.append(hit)
        i = j + 1 + len(hit)
    return found, unknown


def allocate_slots(characters: list[tuple[Ref, int]], scene: Ref | None, props: list[Ref],
                   slots: int) -> tuple[list[Ref], list[Ref]]:
    """按优先级填参考图槽位,返回 (进槽的, 出槽的)。

    优先级(设计 §6.1):台词最多的角色 > 场景 > 第二个角色 > 道具 > 其余角色。
    characters 是 (角色, 本镜台词字数);台词一样多时保持绑定顺序。
    """
    chars = [r for r, _ in sorted(characters, key=lambda x: -x[1])]
    order = chars[:1] + ([scene] if scene else []) + chars[1:2] + list(props) + chars[2:]
    return order[:slots], order[slots:]
```

```python
# -*- coding: utf-8 -*-
"""把 `@名字` 展开成模型认的编号标签,并做标签双向校验。

两种方言:
- `qwen`:Qwen-Image-Edit 的节点自动在提示词前拼「Picture 1: <vision…>」,正文写 `Picture N`
- `h3`:MiniMax H3 写 `<Picture i>` / `<Audio j>`,每类各自从 1 编号

🔴 ComfyUI 节点对标签**零校验**,写错编号静默错位。云机上一个已出片的工作流,
提示词写着「…and <Audio 1> exactly as it is」,而工作流里根本没接音频。

校验规则(闸门 0 实测后定):
1. 越界引用恒报;
2. 「每张接入的图都要被引用」只在 ≥2 张图时成立 —— 单图没有歧义;
3. 接了参考音频就必须引用 `<Audio N>`,否则模型不知道这段音频做什么用。
"""
from __future__ import annotations

import re

from tools.drama.refs import Ref

_FORMAT = {"qwen": "Picture {}", "h3": "<Picture {}>"}
_IMAGE_TAG = {"qwen": re.compile(r"(?<!<)\bPicture\s*(\d+)"), "h3": re.compile(r"<Picture\s+(\d+)>")}
_AUDIO_TAG = re.compile(r"<Audio\s+(\d+)>")


def expand(prompt: str, kept: list[Ref], dropped: list[Ref], dialect: str) -> str:
    """`@名字` → 编号标签(后面保留名字,人还读得懂);出槽的素材退化成纯名字。

    提示词里出现未绑定的 `@名字` 直接抛错 —— 不许静默跳过。
    """
    fmt = _FORMAT[dialect]
    index = {r.name: i + 1 for i, r in enumerate(kept)}
    by_len = sorted([r.name for r in kept] + [r.name for r in dropped], key=len, reverse=True)
    out: list[str] = []
    i = 0
    while (j := prompt.find("@", i)) >= 0:
        out.append(prompt[i:j])
        rest = prompt[j + 1:]
        hit = next((n for n in by_len if rest.startswith(n)), None)
        if hit is None:
            raise ValueError(f"提示词里的 @{rest[:12]} 不是本镜绑定的素材 —— 不许静默跳过")
        out.append(f"{fmt.format(index[hit])}({hit})" if hit in index else hit)
        i = j + 1 + len(hit)
    out.append(prompt[i:])
    return "".join(out)


def check_tags(prompt: str, n_images: int, n_audios: int = 0, dialect: str = "h3") -> list[str]:
    problems: list[str] = []
    images = {int(n) for n in _IMAGE_TAG[dialect].findall(prompt)}
    audios = {int(n) for n in _AUDIO_TAG.findall(prompt)} if dialect == "h3" else set()
    for n in sorted(images):
        if not 1 <= n <= n_images:
            problems.append(f"提示词引用了第 {n} 张参考图,但只接入了 {n_images} 张")
    for n in sorted(audios):
        if not 1 <= n <= n_audios:
            problems.append(f"提示词引用了 <Audio {n}>,但只接入了 {n_audios} 段参考音频")
    if n_images >= 2:
        for n in range(1, n_images + 1):
            if n not in images:
                problems.append(f"接了 {n_images} 张参考图,但提示词从未引用第 {n} 张 —— 多图必须逐一指代")
    for n in range(1, n_audios + 1):
        if n not in audios:
            problems.append(f"接了参考音频,但提示词从未引用 <Audio {n}> —— 模型不知道这段音频做什么用")
    return problems
```

```python
# -*- coding: utf-8 -*-
"""段落 readiness:一串带人话的检查项,不是一个布尔值(设计 §6.4)。

界面上直接显示「还差什么」。包含外部依赖(云机通不通、模型配没配)—— 那往往是最常见的没准备好。
readiness 是**可计算的**,不是人手工点的;它和任务状态(task.state)是两个独立维度。

`warn=True` 的项不阻塞生成,但必须让人看见(例如素材因槽位不足退化成文字描述)。
"""
from __future__ import annotations

from dataclasses import dataclass

from tools.drama import refs, repo, rules
from tools.drama.db import Store
from tools.gen.capability import Capability

# 各参考模式需要哪些参考帧。key / first_last_key / anchors 尚未实现 —— 报出来,不假装支持。
FRAMES_NEEDED: dict[str, tuple[str, ...]] = {
    "text_only": (),
    "first": ("first_frame_path",),
    "last": ("last_frame_path",),
    "first_last": ("first_frame_path", "last_frame_path"),
}
FRAME_LABEL = {"first_frame_path": "首帧", "last_frame_path": "尾帧"}
KIND_LABEL = {"character": "角色", "scene": "场景", "prop": "道具"}


@dataclass(frozen=True)
class Check:
    key: str
    ok: bool
    msg: str
    warn: bool = False


def is_ready(checks: list[Check]) -> bool:
    return all(c.ok for c in checks)


def segment_readiness(store: Store, segment_id: int, *, video_cap: Capability, image_cap: Capability,
                      host_ok: bool, host_msg: str, model_ok: bool, model_msg: str) -> list[Check]:
    seg = repo.get(store, "segment", segment_id)
    assets = repo.segment_assets(store, segment_id)
    chars, scene, props = assets["characters"], assets["scene"], assets["props"]
    mode = seg["reference_mode"]
    needs_images = mode != "text_only"
    checks: list[Check] = []

    problems = rules.check_segment(seg["description"], seg["duration_sec"], video_cap)
    checks.append(Check("rules", not problems, ";".join(problems) or "时长与子镜头结构合规"))

    missing = []
    if not seg["video_prompt"].strip():
        missing.append("视频提示词")
    if needs_images and not seg["image_prompt"].strip():
        missing.append("首帧图提示词")
    checks.append(Check("prompt_ready", not missing,
                        ("缺少" + "、".join(missing)) if missing else "提示词齐备"))

    if needs_images:
        lack = [] if scene is not None else ["未绑定场景"]
        lack += [f"角色「{c['name']}」缺四视图" for c in chars if not c["turnaround_path"]]
        if scene is not None and not scene["image_path"]:
            lack.append(f"场景「{scene['name']}」缺空镜图")
        lack += [f"道具「{p['name']}」缺白底图" for p in props if not p["image_path"]]
        checks.append(Check("assets_bound", not lack, ";".join(lack) or "参考素材齐备"))

    bound = ([c["name"] for c in chars] + ([scene["name"]] if scene is not None else [])
             + [p["name"] for p in props])
    found, unknown = refs.find_mentions(seg["video_prompt"], bound)
    bad = [f"@{u} 不是本镜绑定的素材" for u in unknown]
    if len(bound) >= 2:
        bad += [f"绑定了「{n}」但视频提示词从未 @ 它" for n in bound if n not in found]
    checks.append(Check("mentions", not bad, ";".join(bad) or "提示词引用与绑定一致"))

    need = FRAMES_NEEDED.get(mode)
    if need is None:
        checks.append(Check("refs_ready", False, f"参考模式 {mode} 尚未实现"))
    else:
        lack_frames = [FRAME_LABEL[f] for f in need if not seg[f]]
        if lack_frames:
            checks.append(Check("refs_ready", False, f"参考模式 {mode},缺少" + "、".join(lack_frames)))
        else:
            checks.append(Check("refs_ready", True, f"参考模式 {mode},参考帧齐备"))

    if needs_images:
        spoken = rules.dialogue_by_speaker(seg["description"])
        weighted = [(refs.Ref("character", c["name"], c["turnaround_path"]),
                     sum(n for who, n in spoken.items() if who.startswith(c["name"]))) for c in chars]
        scene_ref = refs.Ref("scene", scene["name"], scene["image_path"]) if scene is not None else None
        prop_refs = [refs.Ref("prop", p["name"], p["image_path"]) for p in props]
        _, dropped = refs.allocate_slots(weighted, scene_ref, prop_refs, image_cap.max_ref_images)
        if dropped:
            msg = ";".join(f"{KIND_LABEL[r.kind]}「{r.name}」因槽位不足未作为参考图,仅以文字描述参与"
                           for r in dropped)
            checks.append(Check("slots", True, msg, warn=True))

    checks.append(Check("host_ready", host_ok, host_msg))
    checks.append(Check("video_model_ready", model_ok, model_msg))
    return checks
```

```python
# -*- coding: utf-8 -*-
"""失效传播:改了什么 → 哪些已生成的产物作废 → 重跑要多久(设计 §6.5)。

两态(CLAUDE.md「plan/confirm」):`impact()` 只算不改,给人看「这会重跑 9 条视频,约 2.2 小时」;
人确认后才 `apply()`。**不许默默全量重跑** —— 单条 8 秒片约 15 分钟,代价不允许。

`apply()` 只清掉段落上的产物指针,不删文件:每次生成都在 task 表里有不可变记录,旧版本可以挑回来。
"""
from __future__ import annotations

from dataclasses import dataclass

from tools.drama import repo
from tools.drama.db import Store
from tools.gen.capability import Capability

# 事件 → 受影响段落的查询。参数是事件目标的 id。
_AFFECTED = {
    "script": "SELECT id FROM segment WHERE episode_id = ?",
    "style": "SELECT s.id FROM segment s JOIN episode e ON e.id = s.episode_id WHERE e.drama_id = ?",
    "character": "SELECT segment_id FROM segment_character WHERE character_id = ?",
    "scene": "SELECT id FROM segment WHERE scene_id = ?",
    "prop": "SELECT segment_id FROM segment_prop WHERE prop_id = ?",
    "layout": "SELECT id FROM segment WHERE id = ?",
    "segment_prompt": "SELECT id FROM segment WHERE id = ?",
}
_FRAME_FIELDS = ("first_frame_path", "last_frame_path")


@dataclass(frozen=True)
class Impact:
    segment_ids: tuple[int, ...]
    frames: int          # 要重出的参考帧张数(只算已经出过的)
    videos: int          # 要重跑的视频条数(只算已经出过的)
    est_seconds: float

    def summary(self) -> str:
        if self.videos == 0 and self.frames == 0:
            return "不影响任何已生成的产物"
        return (f"这会重跑 {self.videos} 条视频、{self.frames} 张参考帧,"
                f"约 {self.est_seconds / 3600:.1f} 小时(估算)")


def impact(store: Store, event: str, target_id: int, *, video_cap: Capability,
           image_cap: Capability) -> Impact:
    if event not in _AFFECTED:
        raise ValueError(f"未知的变更事件 {event},可选:{sorted(_AFFECTED)}")
    ids = sorted({r[0] for r in store.conn.execute(_AFFECTED[event], (target_id,))})
    frames = videos = 0
    est = 0.0
    for sid in ids:
        seg = repo.get(store, "segment", sid)
        if seg["video_path"]:
            if seg["duration_sec"] is None:
                raise ValueError(f"段落 {sid} 有视频却没有时长 —— 数据不一致,不能估算")
            videos += 1
            est += video_cap.est_seconds(video_cap.frames_for(seg["duration_sec"]))
        n = sum(1 for f in _FRAME_FIELDS if seg[f])
        frames += n
        est += n * image_cap.sec_per_item
    return Impact(tuple(ids), frames, videos, est)


def apply(store: Store, imp: Impact) -> None:
    """人确认后才调用:清掉受影响段落的产物指针。"""
    with store.write() as c:
        c.executemany("UPDATE segment SET first_frame_path = NULL, last_frame_path = NULL,"
                      " video_path = NULL, subtitle_path = NULL WHERE id = ?",
                      [(i,) for i in imp.segment_ids])
```
