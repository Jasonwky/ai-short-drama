```python
# -*- coding: utf-8 -*-
"""竖屏可拍性:有些构图在 9:16 里画不出来,分镜阶段就拦(spec 2026-10-23-不穿帮验收-design.md 第 2 节)。

为什么:竖屏全景里「躺着的全身 + 站着的人 + 整张床」放不下,模型保住人的大小,就让人伸出床外。
2026-10-23 第 6 段对照(spike-results/2026-10-23-躺姿穿帮归因与规避方案.md):原样全景 4/4 出床、
床尾纵深 2/2 出床、近景(只拍胸口以上)2/2 合理。所以规则只放行近景和特写。
三人以上同框只提示:真人竖屏剧「三个人以上的群戏比较少,即便有群戏也是会分批次」(界面新闻采访主创)。
"""
from __future__ import annotations

SHOT_TYPES = ("特写", "近景", "中景", "全景")
CLOSE = ("特写", "近景")
LYING = ("仰躺", "侧躺", "趴")
VERTICAL = "9:16"
# 与 10-23 对照 a1 组同一写法(那组 2/2 合理)
CLOSE_PHRASE = "近景,画面只拍到躺着的人胸口以上,站着的人在画面上方"


def _lying(entries: list[dict]) -> list[str]:
    return [e["character"] for e in entries if e["pose"] in LYING]


def problems(aspect: str, entries: list[dict], shot_type: str) -> list[str]:
    """拦:竖屏、有人躺着、同框 ≥ 2 人,景别却不是近景 / 特写。"""
    lying = _lying(entries)
    if aspect != VERTICAL or len(entries) < 2 or not lying or shot_type in CLOSE:
        return []
    return [f"竖屏里「{'、'.join(lying)}」躺着、同框 {len(entries)} 人,景别是「{shot_type or '没填'}」:"
            f"画面放不下,人会伸出床外 —— 景别只能选近景或特写(实测全景 4/4 出床、近景 2/2 合理)"]


def hints(aspect: str, entries: list[dict]) -> list[str]:
    """只提示:竖屏三人以上同框。"""
    if aspect == VERTICAL and len(entries) >= 3:
        return [f"竖屏同框 {len(entries)} 人:真人竖屏剧三人以上的群戏少,多半分批拍 —— 考虑拆成两段"]
    return []


def phrase(aspect: str, entries: list[dict], shot_type: str) -> str:
    """拼进首帧提示词的那句(由代码写,LLM 不写);规则不适用时为空。"""
    if aspect == VERTICAL and len(entries) >= 2 and _lying(entries) and shot_type in CLOSE:
        return CLOSE_PHRASE
    return ""
```

```python
def _aspect(store: Store, seg) -> str:
    return store.conn.execute("SELECT d.aspect_ratio FROM drama d JOIN episode e ON e.drama_id = d.id WHERE e.id = ?",
                              (seg["episode_id"],)).fetchone()[0]


def frame_text(store: Store, segment_id: int, field: str) -> str:
    """参考帧实际用的图提示词。首帧在人 / LLM 写的那句后面拼上场记表(姿态、朝向、站位、手持,由代码写 ——

    场记表只描述段落开头,尾帧不拼。"""
    seg = repo.get(store, "segment", segment_id)
    text = seg[FRAMES[field][1]]
    entries = blocking.parse(seg["blocking_json"]) if field == "first_frame_path" else []
    if not (entries and text.strip()):
        return text
    tail = "。".join(x for x in (blocking.phrase(entries),
                                 framing.phrase(_aspect(store, seg), entries, seg["shot_type"])) if x)
    return f"{text.rstrip('。;; ')}。{tail}"
```

```python
def confirm(store: Store, root: Path, segment_id: int, checks: list[str], note: str = "") -> int:
    """人逐项勾过清单才算确认:人勾项缺一项就拒(确认的意思就是「这些我都看过了」)。"""
    seg = repo.get(store, "segment", segment_id)
    if not seg["video_path"] or not (Path(root) / seg["video_path"]).is_file():
        raise ValueError(f"段落 {seg['number']} 还没有视频,没什么可看")
    gone = checklist.missing(checks)
    if gone:
        raise ValueError(f"段落 {seg['number']} 还有没勾的检查项:{'、'.join(gone)}")
    return repo.add_review(store, segment_id, video_path=seg["video_path"],
                           sha256=sha256_file(Path(root) / seg["video_path"]), checks=checks, note=note)


def complete(r: dict) -> bool:
    """按现行清单勾全了。v13 时代的确认(checks_json 为空)不算 —— 清单变了,要按新清单重看。"""
    return bool(r["checks_json"]) and not checklist.missing(json.loads(r["checks_json"]))


def problem(number: int, r: dict | None = None) -> str:
    if r is not None:
        return f"段落 {number} 是按旧清单确认的,需要按新清单重看(在「视频」面板逐项勾选)"
    return f"段落 {number} 的当前视频还没看片确认(在「视频」面板看切镜帧与接缝,逐项勾选后确认)"


def snapshot(review: dict | None) -> dict | None:
    if review is None:
        return None
    return {"id": review["id"], "sha256": review["sha256"], "note": review["note"], "created_at": review["created_at"],
            "checks": json.loads(review["checks_json"]) if review["checks_json"] else []}


def note(number: int, snap: dict) -> str:
    """溯源报告里的一行。"""
    extra = f",备注:{snap['note']}" if snap["note"] else ""
    return f"看片确认:段落 {number},{snap['created_at']},勾过 {len(snap.get('checks', []))} 项{extra}"
```
