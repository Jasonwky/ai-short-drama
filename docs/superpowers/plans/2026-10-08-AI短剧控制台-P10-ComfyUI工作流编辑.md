`tools/gen/graphs.py`:

```python
# -*- coding: utf-8 -*-
"""ComfyUI API 格式工作流(节点 id → {class_type, inputs})上的纯函数。可单测,不碰网络、不碰库。

工作流编辑(设计 `specs/2026-10-08-ComfyUI工作流编辑-design.md`)的两种用法都靠这里:
- **模板预设**:画布改后的图与「标准例子」比,提取三类改动(改参数 / 加节点 / 删节点);用的时候叠到控制台拼好的图上;
- **改参重跑**:比对归控制台的字段有没有被改(锁定)。

**哪些字段归控制台**:按模板家族(qwen / h3)写死在 OWNED 里 —— 提示词、参考图节点、种子、宽高 / 帧数、锚点、输出文件名前缀。
这些每次由控制台按段落填,画布里改了也不算数(预设)或直接拒绝(改参重跑)。
"""
from __future__ import annotations

import copy

# 模板类别 → 家族(决定哪些字段归控制台)
FAMILY = {"frame": "qwen", "fourview": "qwen", "asset": "qwen", "video": "h3"}

# 整个节点归控制台(参考图槽、尺寸 latent 等)
_OWNED_NODES = {"qwen": {"18", "19", "20", "25", "26", "30"}, "h3": {"8", "9", "10", "11"}}
# 节点的部分输入归控制台
_OWNED_INPUTS = {
    "qwen": {"21": {"prompt", "image1", "image2", "image3"}, "27": {"seed", "latent_image"},
             "29": {"filename_prefix"}},
    "h3": {"12": {"prompt", "width", "height", "length", "ref_images.ref_image_0", "ref_images.ref_image_1",
                  "ref_images.ref_image_2", "ref_audios.ref_audio_0"},
           "13": {"conditioning"}, "19": {"noise_seed"}, "24": {"filename_prefix"}},
}
_H3_DYNAMIC_FROM = 40          # build_h3 的锚点节点从 40 起编号,整段归控制台

MODEL_EXT = (".safetensors", ".gguf", ".ckpt", ".pt", ".pth", ".bin", ".sft")
SEED_INPUTS = ("seed", "noise_seed")


def owned_node(family: str, node_id: str) -> bool:
    if node_id in _OWNED_NODES[family]:
        return True
    return family == "h3" and node_id.isdigit() and int(node_id) >= _H3_DYNAMIC_FROM


def owned_input(family: str, node_id: str, name: str) -> bool:
    return owned_node(family, node_id) or name in _OWNED_INPUTS[family].get(node_id, set())


def _is_link(v) -> bool:
    return isinstance(v, list) and len(v) == 2 and isinstance(v[0], str) and isinstance(v[1], int)


def _sort_key(node_id: str):
    return (0, int(node_id)) if node_id.isdigit() else (1, node_id)


def diff(base: dict, edited: dict, family: str, default_of=lambda cls, name: None) -> tuple[dict, list[str]]:
    """edited 相对 base 的改动,返回 (改动, 问题)。

    改动 = {"set": {节点: {输入: 值}}, "unset": {节点: [输入]}, "add": {u1: 节点}, "remove": [节点]}。
    - 归控制台的字段一律忽略;删了归控制台的节点 → 问题
    - 新加节点改名 u1、u2……(避开控制台动态生成的编号),连到它们的连线一并改名
    - 前端会给模板没写的可选输入补默认值(实测 `CreateVideo.color_space = sRGB`):等于 default_of 给的默认值就不算改动
    """
    problems = []
    rename = {nid: f"u{i}" for i, nid in enumerate(sorted((n for n in edited if n not in base), key=_sort_key), 1)}

    def ren(v):
        return [rename.get(v[0], v[0]), v[1]] if _is_link(v) else v

    out = {"set": {}, "unset": {}, "add": {}, "remove": []}
    for nid in sorted(base, key=_sort_key):
        if nid not in edited:
            if owned_node(family, nid) or nid in _OWNED_INPUTS[family]:
                problems.append(f"节点 {nid}({base[nid]['class_type']})归控制台,不能删")
            else:
                out["remove"].append(nid)
            continue
        if edited[nid]["class_type"] != base[nid]["class_type"]:
            problems.append(f"节点 {nid} 的类型从 {base[nid]['class_type']} 变成了 {edited[nid]['class_type']}")
            continue
        if owned_node(family, nid):
            continue
        old, new = base[nid]["inputs"], edited[nid]["inputs"]
        cls = base[nid]["class_type"]
        for name, val in new.items():
            if owned_input(family, nid, name):
                continue
            val = ren(val)
            if name in old:
                if val != old[name]:
                    out["set"].setdefault(nid, {})[name] = val
            elif val != default_of(cls, name):
                out["set"].setdefault(nid, {})[name] = val
        for name in old:
            if name not in new and not owned_input(family, nid, name):
                out["unset"].setdefault(nid, []).append(name)
    for nid, new_id in rename.items():
        node = copy.deepcopy(edited[nid])
        node["inputs"] = {k: ren(v) for k, v in node["inputs"].items()}
        node.pop("_meta", None)
        out["add"][new_id] = node
    return out, problems


def is_empty(d: dict) -> bool:
    return not (d["set"] or d["unset"] or d["add"] or d["remove"])


def apply(wf: dict, d: dict) -> dict:
    """把改动叠到控制台拼好的图上,返回新图。叠不上(改的节点这次不存在、连线断了)直接抛 ValueError,不静默跳过。"""
    wf = copy.deepcopy(wf)
    problems = []
    for nid in d["remove"]:
        if nid not in wf:
            problems.append(f"预设要删的节点 {nid} 这次不存在")
        wf.pop(nid, None)
    for nid, node in d["add"].items():
        wf[nid] = copy.deepcopy(node)
    for nid, inputs in d["set"].items():
        if nid not in wf:
            problems.append(f"预设要改的节点 {nid} 这次不存在")
            continue
        wf[nid]["inputs"].update(copy.deepcopy(inputs))
    for nid, names in d["unset"].items():
        if nid in wf:
            for name in names:
                wf[nid]["inputs"].pop(name, None)
    problems += [f"连线断了:{p}" for p in dangling(wf)]
    if problems:
        raise ValueError(";".join(problems))
    return wf


def dangling(wf: dict) -> list[str]:
    return [f"{nid}.{k} → {v[0]}" for nid, n in wf.items() for k, v in n["inputs"].items()
            if _is_link(v) and v[0] not in wf]


def summary(d: dict, base: dict) -> list[str]:
    """给人看的改动清单(确认页、工作流面板)。"""
    lines = []
    for nid, inputs in d["set"].items():
        cls = base.get(nid, {}).get("class_type", "?")
        for name, val in inputs.items():
            old = base.get(nid, {}).get("inputs", {}).get(name, "(无)")
            lines.append(f"节点 {nid} {cls}.{name}:{_show(old)} → {_show(val)}")
    for nid, names in d["unset"].items():
        lines += [f"节点 {nid} {base.get(nid, {}).get('class_type', '?')}.{n}:去掉" for n in names]
    lines += [f"新增节点 {nid} {n['class_type']}" for nid, n in d["add"].items()]
    lines += [f"删掉节点 {nid} {base.get(nid, {}).get('class_type', '?')}" for nid in d["remove"]]
    return lines


def _show(v) -> str:
    return f"连到 {v[0]}" if _is_link(v) else repr(v)


def lock_problems(source: dict, edited: dict, family: str) -> list[str]:
    """改参重跑:归控制台的字段(提示词、参考图、尺寸、帧数、锚点、种子以外的)必须和原任务一模一样。"""
    problems = []
    for nid in sorted(set(source) | set(edited), key=_sort_key):
        a, b = source.get(nid), edited.get(nid)
        if owned_node(family, nid):
            if (a and a["inputs"]) != (b and b["inputs"]) or (a and a["class_type"]) != (b and b["class_type"]):
                problems.append(f"节点 {nid}({(a or b)['class_type']})归控制台,不能在画布里改")
            continue
        for name in _OWNED_INPUTS[family].get(nid, set()):
            if name in SEED_INPUTS:
                continue                     # 种子允许改(改参重跑最常见的就是换种子)
            va = a["inputs"].get(name) if a else None
            vb = b["inputs"].get(name) if b else None
            if va != vb:
                problems.append(f"{nid}.{name} 归控制台,不能在画布里改 —— 提示词 / 尺寸 / 帧数请去段落或资产里改")
    return problems


def model_files(wf: dict) -> list[tuple[str, str, str]]:
    """图里用到的模型文件 (节点, 输入, 文件名):按扩展名认。"""
    return [(nid, k, v) for nid, n in wf.items() for k, v in n["inputs"].items()
            if isinstance(v, str) and v.lower().endswith(MODEL_EXT)]


def output_nodes(wf: dict, object_info: dict) -> list[str]:
    return [nid for nid, n in wf.items() if object_info.get(n["class_type"], {}).get("output_node")]


def seeds(wf: dict) -> list[tuple[str, str, int]]:
    return [(nid, k, v) for nid, n in wf.items() for k, v in n["inputs"].items()
            if k in SEED_INPUTS and isinstance(v, int) and not isinstance(v, bool)]


def load_images(wf: dict) -> dict[str, str]:
    return {nid: n["inputs"]["image"] for nid, n in wf.items()
            if n["class_type"] == "LoadImage" and isinstance(n["inputs"].get("image"), str)}


def with_images(wf: dict, names: dict[str, str]) -> dict:
    """把「加载图像」节点里的路径按 names 换掉(相对路径 ↔ 上传名)。"""
    wf = copy.deepcopy(wf)
    for nid in load_images(wf):
        wf[nid]["inputs"]["image"] = names.get(wf[nid]["inputs"]["image"], wf[nid]["inputs"]["image"])
    return wf


def steps(wf: dict) -> int | None:
    """H3 视频的采样步数(BasicScheduler.steps),估算耗时用;找不到返回 None(调用方用默认步数)。"""
    for n in wf.values():
        if n["class_type"] == "BasicScheduler" and isinstance(n["inputs"].get("steps"), int):
            return n["inputs"]["steps"]
    return None
```

```python
# -*- coding: utf-8 -*-
"""工作流的提交把关(设计 §四):画布改出来的图和 LLM 输出一样,是要被审的输入。

`check()` 只做确定性检查,返回问题清单(空 = 通过),不改任何东西:
连线、节点类型与模型在节点清单里、模型白名单、恰好一个输出、尺寸 / 帧数、种子唯一、参考图来自素材库。
控制台字段锁定(改参重跑)见 `graphs.lock_problems`;种子锁定见 `rerun`。
"""
from __future__ import annotations

from pathlib import Path, PurePosixPath

from tools.gen import graphs
from tools.gen.capability import ALLOWED_MODELS

OUTPUT_CLASSES = {".png": {"SaveImage"}, ".mp4": {"SaveVideo"}}
_SIZE_NODE = {"qwen": "30", "h3": "12"}       # 宽高在哪个节点(与 graphs.OWNED 一致)


def check(wf: dict, *, object_info: dict, family: str, expect_ext: str, size: tuple[int, int] | None,
          frames: int | None = None, data_root: Path | None = None) -> list[str]:
    problems = [f"连线断了:{d}" for d in graphs.dangling(wf)]

    unknown = sorted({n["class_type"] for n in wf.values() if n["class_type"] not in object_info})
    problems += [f"节点类型 {c} 不在节点清单里(云机没装?)" for c in unknown]

    allowed = {m.name for m in ALLOWED_MODELS}
    for nid, name, value in graphs.model_files(wf):
        if value not in allowed:
            problems.append(f"模型 {value}(节点 {nid}.{name})没有登记许可证 —— 白名单见 capability.ALLOWED_MODELS")

    for nid, n in wf.items():
        if n["class_type"] == "LoadImage" or n["class_type"] not in object_info:
            continue
        spec = object_info[n["class_type"]]["input"]
        spec = {**spec.get("required", {}), **spec.get("optional", {})}
        for k, v in n["inputs"].items():
            options = spec.get(k, [None])[0]
            if isinstance(options, list) and isinstance(v, str) and v not in options:
                problems.append(f"{nid}.{k} = {v!r} 不在云机的可选项里")

    outs = graphs.output_nodes(wf, object_info)
    want = OUTPUT_CLASSES[expect_ext]
    if len(outs) != 1 or wf[outs[0]]["class_type"] not in want:
        found = [f"{o}({wf[o]['class_type']})" for o in outs]
        problems.append(f"要恰好 1 个输出节点({' / '.join(sorted(want))}),实际 {found or '没有'}")

    node = wf.get(_SIZE_NODE[family], {}).get("inputs", {})
    if size is not None and (node.get("width"), node.get("height")) != tuple(size):
        problems.append(f"尺寸应为 {size[0]}x{size[1]}(剧的画幅),实际 {node.get('width')}x{node.get('height')}")
    if frames is not None and node.get("length") != frames:
        problems.append(f"帧数应为 {frames}(由段落时长换算),实际 {node.get('length')}")

    if len(graphs.seeds(wf)) != 1:
        problems.append(f"要恰好 1 个种子,实际 {[f'{n}.{k}' for n, k, _ in graphs.seeds(wf)] or '没有'} —— 溯源记不下来")

    if data_root is not None:
        for nid, rel in graphs.load_images(wf).items():
            p = PurePosixPath(rel)
            if p.is_absolute() or ".." in p.parts or not (Path(data_root) / rel).is_file():
                problems.append(f"参考图 {rel}(节点 {nid})不在素材库里 —— 只认素材库的相对路径,新图请去资产面板上传")
    return problems
```

```python
# -*- coding: utf-8 -*-
"""云机节点清单(`/object_info`)的本机缓存 —— 离线编辑工作流的命门(设计 §2.3)。

它决定画布里每个节点长什么样、有哪些参数、模型下拉里有哪些文件。云机开着时抓一份存在
`<data_root>/comfy_cache/`,关机后编辑页用它。缓存会过期(云机装 / 卸节点或模型后),所以:
- 记下抓取时间、来源云机、云机要求的前端版本,编辑页上常驻显示;
- worker 发现云机连通、缓存超过一天就自动刷一次;
- 从没抓过 → 明确报错,不拿空清单糊弄(空清单下我们的节点全会显示成「未知节点」)。
"""
from __future__ import annotations

import hashlib
import json
from datetime import datetime, timedelta, timezone
from pathlib import Path

DIR = "comfy_cache"
MAX_AGE = timedelta(days=1)


def _dir(data_root: Path) -> Path:
    return Path(data_root) / DIR


def meta(data_root: Path) -> dict | None:
    f = _dir(data_root) / "meta.json"
    return json.loads(f.read_text(encoding="utf-8")) if f.is_file() else None


def load(data_root: Path) -> dict:
    f = _dir(data_root) / "object_info.json"
    if not f.is_file():
        raise ValueError("还没有节点清单缓存 —— 先开云机,在「主机」面板点「刷新节点清单」")
    return json.loads(f.read_text(encoding="utf-8"))


def refresh(data_root: Path, client, host_label: str) -> dict:
    """从云机抓节点清单与版本信息,写缓存,返回元数据。抓不到由 client 抛(连不上就是连不上,不写半份)。"""
    info = client.object_info()
    stats = client.system_stats()["system"]
    body = json.dumps(info, ensure_ascii=False, sort_keys=True).encode("utf-8")
    m = {"fetched_at": datetime.now(timezone.utc).isoformat(timespec="seconds"), "host": host_label,
         "comfyui_version": stats.get("comfyui_version"), "required_frontend_version": stats.get("required_frontend_version"),
         "node_types": len(info), "sha256": hashlib.sha256(body).hexdigest()}
    d = _dir(data_root)
    d.mkdir(parents=True, exist_ok=True)
    (d / "object_info.json.tmp").write_bytes(body)
    (d / "object_info.json.tmp").replace(d / "object_info.json")
    (d / "meta.json").write_text(json.dumps(m, ensure_ascii=False, indent=1), encoding="utf-8")
    return m


def stale(m: dict | None, now: datetime | None = None) -> bool:
    if m is None:
        return True
    now = now or datetime.now(timezone.utc)
    return now - datetime.fromisoformat(m["fetched_at"]) > MAX_AGE


def default_of(object_info: dict):
    """给 graphs.diff 用:某个节点某个输入在清单里的默认值(没有就 None)。"""
    def get(cls: str, name: str):
        spec = object_info.get(cls, {}).get("input", {})
        spec = {**spec.get("required", {}), **spec.get("optional", {})}.get(name)
        return spec[1].get("default") if spec and len(spec) > 1 and isinstance(spec[1], dict) else None
    return get
```

```python
# -*- coding: utf-8 -*-
"""ComfyUI 前端包(本机离线编辑工作流用,设计 §2.1)。

- **版本钉成云机要求的版本**(云机 `requirements.txt` 与 `/system_stats.required_frontend_version` 都是 1.51.9);
- **核对 PyPI 公布的 sha256**,对不上拒绝使用(防下到被换过的包);
- **GPL-3.0,不进仓库**:下载到 gitignore 的 `thirdparty/comfyui-frontend/<版本>/`,只在本机使用。
  wheel 里不带 LICENSE 文件,许可证原文见 github.com/Comfy-Org/ComfyUI_frontend(v1.51.9 标签,已核)。
"""
from __future__ import annotations

import hashlib
import io
import shutil
import urllib.request
import zipfile
from pathlib import Path

VERSION = "1.51.9"
URL = ("https://files.pythonhosted.org/packages/48/30/b7d9c1df063e56588ab11011fbf190ce7eac313437f48db5a937986c3b71/"
       "comfyui_frontend_package-1.51.9-py3-none-any.whl")
SHA256 = "979aa59bb1c3f972596d540b4efa437941d37cca3646b46018c96c01698ff085"   # PyPI 公布值,2026-10-08 核对
ROOT = Path(__file__).resolve().parents[2] / "thirdparty" / "comfyui-frontend"
_PREFIX = "comfyui_frontend_package/static/"


def static_dir(root: Path = ROOT, version: str = VERSION) -> Path:
    return root / version / "static"


def installed(root: Path = ROOT, version: str = VERSION) -> Path | None:
    d = static_dir(root, version)
    return d if (d / "index.html").is_file() else None


def require(root: Path = ROOT) -> Path:
    d = installed(root)
    if d is None:
        raise ValueError(f"本机还没有 ComfyUI 前端 {VERSION}(约 22 MB,GPL-3.0,不进仓库)—— "
                         "在「工作流」面板点「下载前端」,或运行 python -X utf8 -m tools.gen.frontend")
    return d


def _fetch(url: str) -> bytes:
    with urllib.request.urlopen(url, timeout=300) as r:
        return r.read()


def install(root: Path = ROOT, *, url: str = URL, sha256: str = SHA256, version: str = VERSION, fetch=_fetch) -> Path:
    """下载 → 核哈希 → 解出 static/ → 原子换上。哈希不对就什么都不写。"""
    data = fetch(url)
    got = hashlib.sha256(data).hexdigest()
    if got != sha256:
        raise ValueError(f"前端包 sha256 不对:期望 {sha256},实际 {got} —— 拒绝使用(可能被换过包)")
    tmp = root / f"{version}.tmp"
    shutil.rmtree(tmp, ignore_errors=True)
    with zipfile.ZipFile(io.BytesIO(data)) as z:
        for name in z.namelist():
            if name.startswith(_PREFIX) and not name.endswith("/"):
                dest = tmp / "static" / name[len(_PREFIX):]
                dest.parent.mkdir(parents=True, exist_ok=True)
                dest.write_bytes(z.read(name))
    if not (tmp / "static" / "index.html").is_file():
        shutil.rmtree(tmp, ignore_errors=True)
        raise ValueError("前端包里没有 static/index.html —— 包的结构变了,拒绝使用")
    final = root / version
    shutil.rmtree(final, ignore_errors=True)
    tmp.replace(final)
    return static_dir(root, version)


if __name__ == "__main__":
    print(f"已安装到 {install()}")
```

```python
# -*- coding: utf-8 -*-
"""ComfyUI 前端的本机轻量后端垫片(设计 §2.1–2.2;实测见 spike-results/2026-10-08-ComfyUI前端本机离线编辑实测.md)。

前端挂在 `/comfy/`(与控制台同源,编辑页用 iframe 嵌入、经 `contentWindow.app` 载入 / 导出工作流)。
它要的接口在这里就地回答 —— **不连云机**,所以云机关着也能编辑:

- 节点清单用本机缓存(`nodecache`);设置、用户数据存 `<data_root>/comfy_cache/`;
- 强制设置:不恢复上次打开的工作流(`Comfy.Workflow.Persist`,配合编辑令牌)、不显示版本更新;
- **数据不外发**:前端的「远程配置」(`/api/features`)把 comfy.org 的地址指回本机 `/comfy/offline-api/*`;
  `index.html` 另带 CSP 兜底(只许连本机)—— 实测前端否则会请求 api.comfy.org;
- 画布里「运行」(`/api/prompt`)、上传图片(`/api/upload/image`)一律拒绝:提交只走控制台任务表。
"""
from __future__ import annotations

import json
import re

from fastapi import APIRouter, Request, WebSocket
from fastapi.responses import FileResponse, JSONResponse, Response

from tools.gen import frontend, nodecache
from tools.gen.assets import _png

router = APIRouter()
FRONTEND_ROOT = frontend.ROOT          # 测试里换成临时目录
FORCED_SETTINGS = {"Comfy.Workflow.Persist": False, "Comfy.Notification.ShowVersionUpdates": False,
                   "Comfy.TutorialCompleted": True}
# 这些接口前端要的是**数组**:回对象会在初始化时崩(实测 extensions 回 {} → loadExtensions 里 r.filter is not a function)
_LISTS = {"extensions", "workflow_templates", "global_subgraphs", "experiment/models", "userdata", "embeddings"}
_NO_JOBS = {"jobs": [], "pagination": {"total": 0, "offset": 0, "limit": 0, "has_more": False}}   # 前端用 zod 校验:要对象
# 文件不存在时真 ComfyUI 回 404;这里回空内容 —— 404 会进有头自测的 console 报错
_EMPTY_USERDATA = {"user.css": ("", "text/css"), "comfy.templates.json": ("[]", "application/json")}
_SAFE_NAME = re.compile(r"^[\w.\-]+$")
_GRAY = _png(64, 64, 128)
REFUSE_RUN = "画布里不能直接运行 —— 回控制台点「提交为新版本」/「保存为预设」(提交要过把关、进任务表才有溯源)"
REFUSE_UPLOAD = "画布里不能上传图片 —— 参考图只认素材库里的,请去资产面板上传"


def _cache_dir(request: Request):
    d = request.app.state.runtime.data_root / nodecache.DIR
    d.mkdir(parents=True, exist_ok=True)
    return d


def _settings(request: Request) -> dict:
    f = _cache_dir(request) / "settings.json"
    stored = json.loads(f.read_text(encoding="utf-8")) if f.is_file() else {}
    return {**stored, **FORCED_SETTINGS}


def _refuse(message: str) -> JSONResponse:
    # ComfyUI 后端的报错格式:前端会把 message 原样弹给人看
    return JSONResponse({"error": {"type": "refused", "message": message, "details": ""}, "node_errors": {}},
                        status_code=400)


def _csp(request: Request) -> str:
    port = request.url.port or 80
    return ("default-src 'self' data: blob:; script-src 'self' 'unsafe-inline' 'unsafe-eval' blob:; "
            "style-src 'self' 'unsafe-inline'; img-src 'self' data: blob:; font-src 'self' data:; "
            f"connect-src 'self' data: blob: ws://127.0.0.1:{port} ws://localhost:{port}")


@router.websocket("/comfy/ws")
async def ws(sock: WebSocket):
    """空连接:先报一条「队列为 0」再挂着 —— 前端就不会一直重连刷报错。"""
    await sock.accept()
    await sock.send_json({"type": "status", "data": {"status": {"exec_info": {"queue_remaining": 0}}, "sid": "local"}})
    try:
        while True:
            await sock.receive_text()
    except Exception:          # 对端关掉页面:正常结束,没有要处理的
        return


@router.api_route("/comfy/offline-api/{rest:path}", methods=["GET", "POST"])
def offline_api(rest: str):
    """前端以为在访问 comfy.org(版本更新、节点注册表……),一律空数组。"""
    return JSONResponse([])


@router.api_route("/comfy/api/{rest:path}", methods=["GET", "POST"])
async def api(rest: str, request: Request):
    method = request.method
    if rest == "prompt" and method == "POST":
        return _refuse(REFUSE_RUN)
    if rest.startswith("upload/"):
        return _refuse(REFUSE_UPLOAD)
    if rest == "object_info":
        return JSONResponse(nodecache.load(request.app.state.runtime.data_root))
    if rest == "features":
        local = str(request.base_url).rstrip("/") + "/comfy/offline-api"
        return JSONResponse({"supports_preview_metadata": True, "comfy_api_base_url": local,
                             "comfy_platform_base_url": local})
    if rest == "users":
        return JSONResponse({"storage": "server", "migrated": True})
    if rest == "system_stats":
        m = nodecache.meta(request.app.state.runtime.data_root) or {}
        return JSONResponse({"system": {"os": "local", "comfyui_version": m.get("comfyui_version", ""),
                                        "required_frontend_version": frontend.VERSION, "argv": []}, "devices": []})
    if rest == "settings":
        if method == "POST":
            _save_settings(request, await request.json())
        return JSONResponse(_settings(request))
    if rest.startswith("settings/"):
        if method == "POST":
            _save_settings(request, {rest[len("settings/"):]: await request.json()})
            return JSONResponse({})
        return JSONResponse(_settings(request).get(rest[len("settings/"):]))
    if rest.startswith("userdata/"):
        name = rest[len("userdata/"):]
        if not _SAFE_NAME.match(name):
            return JSONResponse({"error": "文件名不合法"}, status_code=400)       # 防路径穿越:只许单层普通文件名
        f = _cache_dir(request) / "userdata" / name
        if method == "POST":
            f.parent.mkdir(parents=True, exist_ok=True)
            f.write_bytes(await request.body())
            return JSONResponse({})
        if f.is_file():
            return Response(f.read_bytes())
        if name in _EMPTY_USERDATA:
            content, media = _EMPTY_USERDATA[name]
            return Response(content=content, media_type=media)
        return Response(status_code=404)
    if rest in _LISTS:
        return JSONResponse([])
    if rest == "jobs":
        return JSONResponse(_NO_JOBS)
    if rest == "view":
        return _view(request)
    return JSONResponse({})


def _view(request: Request) -> Response:
    """「加载图像」节点的缩略图:素材库里有这张图就给真图(改参重跑时看得见用的是哪张),
    没有(模板编辑的占位图)就给一张灰图 —— 404 会进有头自测的 console 报错。"""
    root = request.app.state.runtime.data_root.resolve()
    q = request.query_params
    f = (root / q.get("subfolder", "") / q.get("filename", "")).resolve()
    if root in f.parents and f.is_file():
        return FileResponse(f)
    return Response(content=_GRAY, media_type="image/png")


def _save_settings(request: Request, values: dict) -> None:
    f = _cache_dir(request) / "settings.json"
    stored = json.loads(f.read_text(encoding="utf-8")) if f.is_file() else {}
    stored.update({k: v for k, v in values.items() if k not in FORCED_SETTINGS})
    f.write_text(json.dumps(stored, ensure_ascii=False, indent=1), encoding="utf-8")


@router.get("/assets/{rest:path}")
def root_assets(rest: str):
    """前端挂在子路径下时,只有两个图标仍请求根路径 /assets/(实测)—— 接住,别让 404 进有头自测。"""
    d = frontend.installed(FRONTEND_ROOT)
    f = d / "assets" / rest if d else None
    return FileResponse(f) if f and f.is_file() else Response(status_code=204)


@router.get("/comfy/user.css")
def user_css():
    return Response("", media_type="text/css")


@router.get("/comfy/{rest:path}")
def static(rest: str, request: Request):
    d = frontend.require(FRONTEND_ROOT)
    f = (d / (rest or "index.html")).resolve()
    if d.resolve() not in f.parents or not f.is_file():
        return Response(status_code=404)
    r = FileResponse(f)
    if f.name == "index.html":
        r.headers["Content-Security-Policy"] = _csp(request)
    return r
```

```python
def _save_submitted(root: Path, task_id: int, wf: dict, names: dict[str, str]) -> None:
    """存下实际提交给云机的工作流(上传名 ↔ 素材库路径的对照一并存):「在 ComfyUI 中编辑」打开的就是它。

    out.mkdir(parents=True, exist_ok=True)
    body = json.dumps({"submitted": wf, "names": names}, ensure_ascii=False, indent=1, sort_keys=True)
```

```python
# -*- coding: utf-8 -*-
"""工作流模板预设(设计 §三)。

- 每类模板(首帧 / 尾帧、四视图、资产图、视频)各有一串版本;预设存**相对内置模板的改动**(`graphs.diff` 的三类)。
- 编辑模板:用「标准例子」拼出完整工作流(叠上当前预设)放进画布;保存时与内置的标准例子比,提取改动。
- 保存前把关(两态):列出差异;用各种典型情况(参考图张数 × 锚点个数 × 画幅)**逐一试拼**并过提交把关,
  任一种拼不出或过不了 → 拒绝保存。确认后才落盘(不可改、不可删,再改 = 新版本)。
- 用:**规划建任务时绑定** —— 参数里记预设编号与**改动内容的副本**(任务自带全部信息,离开本库也能复现);
  切换「当前」不影响已经排着的任务。dispatch 时照常拼,再叠上参数里那份改动。
"""
from __future__ import annotations

import json
from dataclasses import dataclass

from tools.drama.db import Store, now
from tools.gen import graph_checks, graphs, nodecache
from tools.gen.workflows import FRAME_SIZE, VIDEO_SIZE, build_h3, build_qwen

CATEGORIES = {"frame": "首帧 / 尾帧", "fourview": "四视图", "asset": "资产图", "video": "视频"}
EXPECT_EXT = {"frame": ".png", "fourview": ".png", "asset": ".png", "video": ".mp4"}
_PH = "placeholder/"                 # 标准例子里参考图的占位名(画布里会标红 —— 参考图归控制台,用时才填)


def _qwen(n: int, size, preset: str = "fast") -> dict:
    prompt = " ".join(f"Picture {i + 1}" for i in range(n)) if n > 1 else "Picture 1"
    return build_qwen(prompt, [f"{_PH}ref{i + 1}.png" for i in range(n)], 1, preset=preset, size=size)


def _h3(n: int, anchors: int, size) -> dict:
    prompt = "".join(f"<Picture {i + 1}> 参考图 {i + 1}。" for i in range(n))
    pins = [(f"{_PH}first.png", 0), (f"{_PH}last.png", -1)][:anchors]
    return build_h3(prompt, [f"{_PH}ref{i + 1}.png" for i in range(n)], None, 192, 1, size=size, anchors=pins)


def _sizes():
    from tools.gen.assets import FOURVIEW_SIZE       # assets 依赖本模块(规划时绑定预设),这里延迟导入免成环
    return FOURVIEW_SIZE


def example(category: str) -> tuple[dict, dict]:
    """编辑模板时放进画布的标准例子(内置模板)与它的把关参数。"""
    if category == "frame":
        return _qwen(2, FRAME_SIZE["9:16"]), {"size": FRAME_SIZE["9:16"]}
    if category == "fourview":
        return _qwen(1, _sizes()), {"size": _sizes()}
    if category == "asset":
        return _qwen(1, (928, 1664), "baseline"), {"size": (928, 1664)}
    if category == "video":
        return _h3(2, 1, VIDEO_SIZE["9:16"]), {"size": VIDEO_SIZE["9:16"], "frames": 192}
    raise ValueError(f"未知的模板类别 {category},可选:{sorted(CATEGORIES)}")


def trials(category: str) -> list[tuple[str, dict, dict]]:
    """保存前要逐一试拼的典型情况:(说明, 内置拼法, 把关参数)。"""
    out = []
    if category == "frame":
        for aspect, size in FRAME_SIZE.items():
            out += [(f"{n} 张参考图 · {aspect}", _qwen(n, size), {"size": size}) for n in (1, 2, 3)]
    elif category == "fourview":
        out.append(("正面图 → 四视图", _qwen(1, _sizes()), {"size": _sizes()}))
    elif category == "asset":
        for label, size in (("角色正面", (928, 1664)), ("场景 9:16", FRAME_SIZE["9:16"]), ("场景 16:9", FRAME_SIZE["16:9"]),
                            ("道具", (1024, 1024))):
            out.append((label, _qwen(1, size, "baseline"), {"size": size}))
    elif category == "video":
        for aspect, size in VIDEO_SIZE.items():
            out += [(f"{n} 张参考图 · {a} 个锚点 · {aspect}", _h3(n, a, size), {"size": size, "frames": 192})
                    for n in (1, 2, 3) for a in (0, 1, 2)]
    else:
        raise ValueError(f"未知的模板类别 {category}")
    return out


def get(store: Store, preset_id: int):
    row = store.conn.execute("SELECT * FROM workflow_preset WHERE id = ?", (preset_id,)).fetchone()
    if row is None:
        raise KeyError(f"预设 {preset_id} 不存在")
    return row


def versions(store: Store, category: str) -> list:
    return store.conn.execute("SELECT * FROM workflow_preset WHERE category = ? ORDER BY version",
                              (category,)).fetchall()


def active(store: Store, category: str):
    row = store.conn.execute("SELECT p.* FROM workflow_preset_active a JOIN workflow_preset p ON p.id = a.preset_id"
                             " WHERE a.category = ?", (category,)).fetchone()
    return row


def diff_of(row) -> dict:
    return json.loads(row["diff_json"])


def editing_graph(store: Store, category: str) -> dict:
    """打开模板编辑时放进画布的图:标准例子叠上当前预设(在当前版的基础上接着改)。"""
    base, _ = example(category)
    cur = active(store, category)
    return graphs.apply(base, diff_of(cur)) if cur else base


@dataclass(frozen=True)
class SavePlan:
    category: str
    diff: dict
    lines: tuple[str, ...]
    problems: tuple[str, ...]


def plan_save(store: Store, category: str, edited: dict, object_info: dict) -> SavePlan:
    """只算不改:改动清单 + 问题。问题非空就不许保存。"""
    base, _ = example(category)
    family = graphs.FAMILY[category]
    d, problems = graphs.diff(base, edited, family, default_of=nodecache.default_of(object_info))
    cur = active(store, category)
    # 已经有更具体的问题(如删了归控制台的节点)时,不再叠「没有不同」
    if not problems and graphs.is_empty(d):
        problems.append("和内置模板没有任何不同 —— 想回到内置,在「工作流」面板把当前设为「内置」")
    elif not problems and cur is not None and d == diff_of(cur):
        problems.append(f"和当前版本 v{cur['version']} 一样,不用另存")
    if not problems:
        for label, wf, ctx in trials(category):
            try:
                built = graphs.apply(wf, d)
            except ValueError as e:
                problems.append(f"[{label}] 拼不出:{e}")
                continue
            problems += [f"[{label}] {p}" for p in graph_checks.check(
                built, object_info=object_info, family=family, expect_ext=EXPECT_EXT[category], **ctx)]
    return SavePlan(category, d, tuple(graphs.summary(d, base)), tuple(problems))


def confirm_save(store: Store, plan: SavePlan, note: str = "") -> int:
    if plan.problems:
        raise ValueError(";".join(plan.problems))
    cur = active(store, plan.category)
    with store.write() as c:
        version = c.execute("SELECT COALESCE(MAX(version), 0) + 1 FROM workflow_preset WHERE category = ?",
                            (plan.category,)).fetchone()[0]
        return c.execute("INSERT INTO workflow_preset (category, version, diff_json, based_on, note, created_at)"
                         " VALUES (?, ?, ?, ?, ?, ?)",
                         (plan.category, version, json.dumps(plan.diff, ensure_ascii=False, sort_keys=True),
                          cur["id"] if cur else None, note.strip(), now())).lastrowid


def activate(store: Store, category: str, preset_id: int | None) -> None:
    """设为当前(None = 回到内置)。只影响之后新规划的任务。"""
    if category not in CATEGORIES:
        raise ValueError(f"未知的模板类别 {category}")
    if preset_id is not None and get(store, preset_id)["category"] != category:
        raise ValueError(f"预设 {preset_id} 不是「{CATEGORIES[category]}」类的")
    with store.write() as c:
        c.execute("INSERT INTO workflow_preset_active (category, preset_id) VALUES (?, ?)"
                  " ON CONFLICT(category) DO UPDATE SET preset_id = excluded.preset_id", (category, preset_id))


def stamp(store: Store, category: str, params: dict) -> dict:
    """规划建任务时绑定当前预设:参数里记编号与改动副本。没有当前预设 = 内置,参数不变。"""
    cur = active(store, category)
    if cur is None:
        return params
    return {**params, "template": category, "preset_id": cur["id"], "preset_version": cur["version"],
            "preset_diff": diff_of(cur)}


def apply_params(wf: dict, params: dict) -> dict:
    """dispatch 时:内置拼法叠上任务参数里那份改动(没有就原样)。"""
    return graphs.apply(wf, params["preset_diff"]) if "preset_diff" in params else wf


def video_steps(store: Store) -> int | None:
    """当前视频预设改过的采样步数(估算耗时用);没改过 = None(用默认步数)。"""
    cur = active(store, "video")
    if cur is None:
        return None
    base, _ = example("video")
    return graphs.steps(graphs.apply(base, diff_of(cur)))
```

```python
# -*- coding: utf-8 -*-
"""改参重跑(设计 §2.4 / §四):在 ComfyUI 画布里打开某条任务**当时实际提交的工作流**,改完作为同一目标的新版本提交。

- 打开:dispatch 时存下的 `outputs/tasks/<id>/workflow.json`(上传名换回素材库路径);
  老任务没有这个文件 → 按当前代码重建,并标明「可能与当时不同」;文件与它的 sha256 对不上 → 拒绝(被改过);
- 提交前把关:`graph_checks.check`(模型白名单、节点清单、单一输出、尺寸 / 帧数、种子唯一、参考图来自素材库)
  + `graphs.lock_problems`(提示词 / 参考图 / 尺寸 / 帧数 / 锚点归控制台,不许在画布里改)+ 角色种子锁定;
- 任务表的种子从改后的工作流里**提取**(不信表单);提示词与参考图就是原任务的(锁定保证了它们没变)。
"""
from __future__ import annotations

import json
import sqlite3
from pathlib import Path

from tools.drama import repo
from tools.drama.db import Store
from tools.drama.tasks import create_task
from tools.gen import graph_checks, graphs
from tools.gen.assets import locked_seed
from tools.serve.dispatch import SUBMITTED, _uploads, build_workflow, sha256_file

_FAMILY = {"qwen_image": "qwen", "minimax_h3": "h3"}


def editable(task) -> bool:
    """走 ComfyUI 的生成任务才有工作流可编辑;字幕对齐、本机抽尾帧没有。"""
    if task["kind"] not in ("image", "video"):
        return False
    return json.loads(task["params_json"]).get("workflow") in ("qwen_image", "minimax_h3", "custom")


def _params(task) -> dict:
    return json.loads(task["params_json"])


def family(task) -> str:
    p = _params(task)
    return p["family"] if p["workflow"] == "custom" else _FAMILY[p["workflow"]]


def context(task) -> dict:
    """把关参数:输出类型、尺寸、帧数 —— 都取原任务的(改参重跑不许改它们)。"""
    p = _params(task)
    return {"expect_ext": p["expect_ext"], "size": tuple(p["size"]) if p.get("size") else None, "frames": p.get("frames")}


def source_graph(data_root: Path, task) -> tuple[dict, bool]:
    """(这条任务的工作流,参考图写成素材库相对路径;是否按当前代码重建的)。"""
    if not editable(task):
        raise ValueError(f"task {task['id']} 不是走 ComfyUI 的生成任务,没有工作流可编辑")
    f = Path(data_root) / "outputs" / "tasks" / str(task["id"]) / SUBMITTED
    if f.is_file():
        sha = f.with_name(SUBMITTED + ".sha256")
        if not sha.is_file() or sha.read_text(encoding="utf-8").strip() != sha256_file(f):
            raise ValueError(f"{f.name} 和它记下的 sha256 对不上 —— 文件被改过,不能当作「当时提交的工作流」")
        data = json.loads(f.read_text(encoding="utf-8"))
        return graphs.with_images(data["submitted"], {name: rel for rel, name in data["names"].items()}), False
    p = _params(task)
    if p["workflow"] == "custom":
        return p["graph"], False
    return build_workflow(task, {r: r for r in _uploads(p)}), True


def plan_rerun(store: Store, data_root: Path, task_id: int, graph: dict, object_info: dict) -> tuple[int | None, list[str]]:
    """把关通过就建一条同目标的新任务,返回 (新任务 id, []);不过返回 (None, 问题)。"""
    src = repo.get(store, "task", task_id)
    base, _ = source_graph(data_root, src)
    fam = family(src)
    problems = graph_checks.check(graph, object_info=object_info, family=fam, data_root=Path(data_root), **context(src))
    problems += graphs.lock_problems(base, graph, fam)
    found = graphs.seeds(graph)
    if src["target_type"] == "character" and found:
        locked = locked_seed(store, "character", src["target_id"])
        if locked is not None and found[0][2] != locked:
            problems.append(f"种子已锁定为 {locked} —— 要换种子先在资产面板解锁")
    if problems:
        return None, problems
    p = _params(src)
    refs = list(dict.fromkeys(graphs.load_images(graph).values()))
    params = {"workflow": "custom", "family": fam, "graph": graph, "from_task": task_id, "image_refs": refs,
              "writes": p["writes"], "expect_ext": p["expect_ext"], "size": p.get("size"), "frames": p.get("frames")}
    tid = create_task(store, kind=src["kind"], target_type=src["target_type"], target_id=src["target_id"],
                      model=src["model"], params=params, seed=found[0][2], prompt=src["prompt"], refs=refs)
    return tid, []


def history_link(task: sqlite3.Row) -> str | None:
    return f"/comfy-edit/task/{task['id']}" if editable(task) else None
```

```python
def _ready(request: Request) -> dict:
    """编辑页的前提:前端包已下载、节点清单抓过。不满足就 400,写明怎么办。"""
    frontend.require(_fe_root())
    m = nodecache.meta(rt(request).data_root)
    if m is None:
        nodecache.load(rt(request).data_root)          # 抛「先开云机刷新一次节点清单」
    return m


def _live_object_info(request: Request) -> tuple[dict, list[str]]:
    """云机在线:刷新节点清单再核(设计 §四 #11),并核对前端版本(#10,在线时硬拦);离线:用缓存。"""
    runtime = rt(request)
    online = [hid for hid, st in runtime.status.items() if st.comfy_ok]
    problems = []
    if online:
        try:
            m = runtime.refresh_object_info(online[0])
        except ValueError as e:
            return nodecache.load(runtime.data_root), [f"云机在线,但实时节点清单没取到:{e}"]
        if m["required_frontend_version"] != frontend.VERSION:
            problems.append(f"云机要求前端 {m['required_frontend_version']},本机是 {frontend.VERSION} —— 先升级前端再提交")
```

```javascript
// 工作流编辑页:同源 iframe 里的 ComfyUI 画布 ↔ 控制台。
// 载入:app.loadApiJson(工作流, "edit-<令牌>");提交:核对当前标签 → app.graphToPrompt().output → POST 给控制台把关。
// 实测见 spike-results/2026-10-08-ComfyUI前端本机离线编辑实测.md §五。
"use strict";

document.addEventListener("DOMContentLoaded", () => {
  const box = document.querySelector("[data-comfy-edit]");
  if (!box) return;
  const frame = document.querySelector("[data-comfy-frame]");
  const out = box.querySelector("[data-edit-result]");
  const tab = `edit-${box.dataset.token}`;
  const graph = JSON.parse(document.querySelector("[data-graph]").textContent);
  const esc = (s) => String(s).replace(/&/g, "&amp;").replace(/</g, "&lt;");
  const app = () => frame.contentWindow && frame.contentWindow.app;

  const active = (a) => {
    const w = a.extensionManager.workflow.activeWorkflow;
    return w && (w.filename || w.path);
  };
  // 前端启动时会自己开一个默认的「Unsaved Workflow」并切过去(实测比 app.graph 就绪晚)。
  // 所以:等它开完再载入我们的;载入后核对当前标签确实是我们的,不是就再载一次 —— 确认了才算「已载入」。
  const loaded = new Promise((resolve) => {
    let tries = 0;
    const tick = setInterval(async () => {
      const a = app();
      if (!(a && a.graph && a.extensionManager && a.extensionManager.workflow && active(a))) return;
      if (active(a) === tab) {
        clearInterval(tick);
        box.dataset.loaded = "1";
        resolve();
      } else if (tries++ % 5 === 0) {
        // 必须在 iframe 自己的 JS 环境里造这份对象:父页面造的数组在 iframe 里 Array.isArray 认不出,
        // 前端会把连线 ["18", 0] 当普通值扔掉(实测:连线全丢、步数变 NaN)
        await a.loadApiJson(frame.contentWindow.JSON.parse(JSON.stringify(graph)), tab);
      }
    }, 400);
  });

  const show = (html) => { out.innerHTML = html; };

  async function send(confirm) {
    await loaded;
    const a = app();
    const current = a.extensionManager.workflow.activeWorkflow;
    const body = {
      graph: (await a.graphToPrompt()).output,
      token: box.dataset.token,
      tab: current && (current.filename || current.path),
      note: (box.querySelector("[data-note]") || {}).value || "",
      confirm,
    };
    const r = await fetch(box.dataset.action, {
      method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(body),
    });
    const data = await r.json();
    if (data.ok) {
      show(`<p class="status-ok" data-edit-ok>已保存。<a href="${esc(data.back)}">回去看</a></p>`);
    } else if (data.confirm) {
      show(`<div class="confirm card" data-edit-confirm><p class="cost">相对内置模板的改动:</p>` +
           data.lines.map((l) => `<div><code>${esc(l)}</code></div>`).join("") +
           `<button class="btn primary" type="button" data-confirm-save>确认保存为新版本</button></div>`);
      out.querySelector("[data-confirm-save]").addEventListener("click", () => send(true));
    } else {
      show(`<ul class="checklist" data-edit-problems>` +
           data.problems.map((p) => `<li class="bad">${esc(p)}</li>`).join("") + `</ul>`);
    }
  }

  box.querySelector("[data-submit]").addEventListener("click", () => send(false));
});
```
