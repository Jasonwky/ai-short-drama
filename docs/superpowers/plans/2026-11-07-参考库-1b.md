### Task 3: 本机媒体检查与关键帧(`avcheck.py`)

**Files:** Create `tools/refs/avcheck.py`;Test `tests/refs/test_avcheck.py`

只调本机 ffmpeg / ffprobe(找不到抛 `RuntimeError("找不到 ffmpeg …")`,和 `tools/cut` 一样不吞)。阈值标出处(spec 3.1 录屏检查;调研录屏实测 mean −17.9 dB)。

```python
# -*- coding: utf-8 -*-
"""录屏的本机检查与加工(spec 1.1 第 3、4 条、3.1):时长、音量、画面;重封装 + 裁到视频区域;关键帧;转 16 kHz PCM。"""
from __future__ import annotations

import json
import re
import shutil
import subprocess
from pathlib import Path

DURATION_TOL = 2.0        # 录出的时长与片长相差不超过 2 秒(spec 3.1)
SILENT_DB = -50.0         # volumedetect mean_volume 低于它 = 静音(调研实测正常片 −17.9 dB)
LUMA_DARK, LUMA_BRIGHT, LUMA_FLAT = 16.0, 235.0, 6.0     # 抽帧平均亮度在 (16, 235) 之间、标准差 > 6 才算有画面
HEAD_SEC, HEAD_STEP, BODY_STEP = 3.0, 0.5, 10.0          # 关键帧:开头 0–3 秒每 0.5 秒,之后每 10 秒(spec 1.1 第 4 条)


def _bin(name: str) -> str:
    p = shutil.which(name)
    if not p:
        raise RuntimeError(f"找不到 {name} —— 本机要装 ffmpeg(录屏检查与关键帧都要它)")
    return p


def _run(args: list[str]) -> str:
    r = subprocess.run(args, capture_output=True, text=True, encoding="utf-8", errors="replace")
    if r.returncode != 0:
        raise RuntimeError(f"{Path(args[0]).name} 失败:{r.stderr.strip()[-400:]}")
    return r.stdout + r.stderr


def duration(path: Path) -> float:
    out = _run([_bin("ffprobe"), "-v", "error", "-show_entries", "format=duration", "-of", "json", str(path)])
    return float(json.loads(out[out.index("{"):])["format"]["duration"])


def mean_volume(path: Path) -> float | None:
    out = _run([_bin("ffmpeg"), "-hide_banner", "-i", str(path), "-vn", "-af", "volumedetect", "-f", "null", "-"])
    m = re.search(r"mean_volume:\s*(-?[\d.]+|-inf) dB", out)
    return None if not m or m.group(1) == "-inf" else float(m.group(1))


def luma_at(path: Path, sec: float) -> tuple[float, float]:
    """第 sec 秒那一帧的平均亮度与标准差(signalstats YAVG / 用 64×64 灰度像素算标准差)。"""
    raw = subprocess.run([_bin("ffmpeg"), "-v", "error", "-ss", f"{sec:.2f}", "-i", str(path), "-frames:v", "1",
                          "-vf", "scale=64:64,format=gray", "-f", "rawvideo", "-"], capture_output=True)
    if raw.returncode != 0 or len(raw.stdout) < 64 * 64:
        raise RuntimeError(f"第 {sec:.1f} 秒抽不出帧:{raw.stderr.decode('utf-8', 'replace')[-300:]}")
    px = raw.stdout[:64 * 64]
    mean = sum(px) / len(px)
    sd = (sum((p - mean) ** 2 for p in px) / len(px)) ** 0.5
    return mean, sd


def recording_problems(path: Path, expected_sec: float) -> tuple[list[str], dict]:
    """录屏三项检查(spec 3.1)。返回 (问题列表, 量到的数)。空列表 = 通过。"""
    got = duration(path)
    vol = mean_volume(path)
    mid = min(got / 2, max(got - 1, 0))
    mean, sd = luma_at(path, mid)
    meta = {"expected_sec": expected_sec, "duration": round(got, 2), "mean_volume_db": vol, "luma_mean": round(mean, 1), "luma_sd": round(sd, 1)}
    p = []
    if abs(got - expected_sec) > DURATION_TOL:
        p.append(f"时长对不上:录到 {got:.1f} 秒,片长 {expected_sec:.1f} 秒(允许 ±{DURATION_TOL:g} 秒)")
    if vol is None or vol < SILENT_DB:
        p.append(f"没有声音:平均音量 {vol if vol is not None else '-inf'} dB(低于 {SILENT_DB:g} dB 算静音)")
    if not (LUMA_DARK < mean < LUMA_BRIGHT) or sd <= LUMA_FLAT:
        p.append(f"画面不对:第 {mid:.0f} 秒那一帧平均亮度 {mean:.0f}、起伏 {sd:.1f}(黑屏 / 白屏 / 纯色)")
    return p, meta


def remux_crop(src: Path, dst: Path, box: tuple[int, int, int, int] | None) -> None:
    """MediaRecorder 的 webm 没写时长:重封装;给了视频区域 (x, y, w, h)(像素,偶数)就裁到它(要重编码 vp9)。"""
    if box is None:
        _run([_bin("ffmpeg"), "-v", "error", "-y", "-i", str(src), "-c", "copy", str(dst)])
        return
    x, y, w, h = box
    _run([_bin("ffmpeg"), "-v", "error", "-y", "-i", str(src), "-vf", f"crop={w}:{h}:{x}:{y}",
          "-c:v", "libvpx-vp9", "-b:v", "2M", "-c:a", "copy", str(dst)])


def keyframe_times(total: float) -> list[float]:
    head = [round(i * HEAD_STEP, 1) for i in range(int(HEAD_SEC / HEAD_STEP) + 1) if i * HEAD_STEP < total]
    body = [float(t) for t in range(int(BODY_STEP), int(total), int(BODY_STEP))]
    return head + body


def extract_frame(src: Path, sec: float, dst: Path) -> None:
    _run([_bin("ffmpeg"), "-v", "error", "-y", "-ss", f"{sec:.2f}", "-i", str(src), "-frames:v", "1", "-q:v", "3", str(dst)])


def pcm16k(src: Path) -> bytes:
    r = subprocess.run([_bin("ffmpeg"), "-v", "error", "-i", str(src), "-vn", "-ac", "1", "-ar", "16000", "-f", "s16le", "-"],
                       capture_output=True)
    if r.returncode != 0:
        raise RuntimeError(f"转音频失败:{r.stderr.decode('utf-8', 'replace')[-300:]}")
    return r.stdout
```

`asr_remote.py`(**在云机上跑**,顶层只用标准库,和 `serve/align_remote.py` 同一套 whisper 用法;whisper 一次只认 30 秒 → 每 25 秒一块,时间加偏移):
```python
# -*- coding: utf-8 -*-
"""**在云机上跑**:参考库录屏的整集转写。音频(16 kHz 单声道 s16le)由本机嵌在脚本末尾经 ssh stdin 送来,只在内存里用,云机不落盘。

    <venv python> - <whisper 目录>        # 脚本末尾由本机追加:_AUDIO_B64 = "..." 与 sys.exit(main(...))

stdout 最后一行 JSON:{"text": 全文, "chars": [[字, 起, 止], ...], "chunks": 块数, "asr_sec": 秒}。退出码 2 = 模型目录不在;3 = 显存不够。
whisper 用法与 align_remote.py 相同(generate + return_token_timestamps;中文 token 攒成完整字)。"""
from __future__ import annotations

import base64
import json
import sys
import time
from pathlib import Path

BAD = "�"
CHUNK_SEC = 25


def merge_tokens(ids, times, decode):
    out = []
    k = 0
    while k < len(ids):
        j = k
        text = decode(ids[k:j + 1])
        while BAD in text and j + 1 < len(ids):
            j += 1
            text = decode(ids[k:j + 1])
        start = times[k - 1] if k > 0 else 0.0
        for c in text:
            if not c.isspace() and c != BAD:
                out.append((c, round(start, 2), round(times[j], 2)))
        k = j + 1
    return out


def chunks(n_samples: int, sr: int = 16000, sec: int = CHUNK_SEC) -> list[tuple[int, int]]:
    step = sr * sec
    return [(a, min(a + step, n_samples)) for a in range(0, n_samples, step)]


def main(model_dir: str, audio_b64: str) -> int:
    if not Path(model_dir).exists():
        print(f"不存在:{model_dir}", file=sys.stderr)
        return 2
    import numpy as np
    import torch
    from transformers import WhisperForConditionalGeneration, WhisperProcessor
    t0 = time.time()
    audio = np.frombuffer(base64.b64decode(audio_b64), dtype=np.int16).astype(np.float32) / 32768
    try:
        proc = WhisperProcessor.from_pretrained(model_dir)
        model = WhisperForConditionalGeneration.from_pretrained(model_dir, dtype=torch.float16).to("cuda")
        chars = []
        spans = chunks(len(audio))
        for a, b in spans:
            feats = proc(audio[a:b], sampling_rate=16000, return_tensors="pt").input_features.to("cuda", model.dtype)
            out = model.generate(feats, language="zh", task="transcribe", return_token_timestamps=True, return_dict_in_generate=True)
            ids = [int(x) for x in out["sequences"][0]]
            ts = [float(x) for x in out["token_timestamps"][0]]
            off = a / 16000
            chars += [(c, round(s + off, 2), round(e + off, 2))
                      for c, s, e in merge_tokens(ids, ts, lambda t: proc.tokenizer.decode(t, skip_special_tokens=True))]
    except torch.cuda.OutOfMemoryError as e:
        print(f"显存不够(ComfyUI 可能正占着卡):{e}", file=sys.stderr)
        return 3
    print(json.dumps({"text": "".join(c for c, _, _ in chars), "chars": chars, "chunks": len(spans),
                      "asr_sec": round(time.time() - t0, 1)}, ensure_ascii=False))
    return 0
```

"""参考库转写(spec 1.1 第 4 条):worker 机器空闲时调一次 `run_one`,取一条 transcribe 任务,在云机 GPU 上识别整集台词。

- 结果存 ref_content(kind='transcript'),全文进指纹;几乎没字(< 20 个汉字)判「不收」(纯配乐或识别失败)。"""
from __future__ import annotations

import base64
import json
from pathlib import Path

from tools.gen.capability import WHISPER_ALIGN
from tools.gen.comfy_client import ComfyUnreachable
from tools.refs import clock, fingerprint, jobs, rjournal, settings
from tools.refs.avcheck import pcm16k
from tools.refs.sources.base import hanzi
from tools.serve.aligner import CUDA_OOM, SSH_UNREACHABLE, gpu_ready

SCRIPT = Path(__file__).with_name("asr_remote.py")
MIN_HANZI = 20


def payload(pcm: bytes) -> bytes:
    tail = f'\n_AUDIO_B64 = "{base64.b64encode(pcm).decode("ascii")}"\nsys.exit(main(sys.argv[1], _AUDIO_B64))\n'
    return SCRIPT.read_bytes() + tail.encode("ascii")


def parse(out: str) -> dict:
    rows = [l for l in out.splitlines() if l.startswith('{"text"')]
    if not rows:
        raise ValueError(f"云机转写脚本没有输出结果:{out.strip()[-400:]}")
    return json.loads(rows[-1])


def run_one(refs, root: Path, client, run) -> str | None:
    """跑一条转写;没有就返回 None。返回给 worker 日志的一行。"""
    job = jobs.next_waiting(refs, clock.today(), kinds=("transcribe",))
    if job is None:
        return None
    rec = refs.conn.execute("SELECT * FROM ref_content WHERE id = ?", (jobs.params(job)["recording_id"],)).fetchone()
    if rec is None or rec["purged_at"] or not (Path(root) / rec["path"]).is_file():
        jobs.start(refs, job["id"]); jobs.finish(refs, job["id"], error="录屏文件不在了(到期清理或被删),转写作废")
        return f"参考库转写 {job['id']}:录屏不在,作废"
    try:
        wait = gpu_ready(client, WHISPER_ALIGN.vram_gb, "参考库转写")
    except ComfyUnreachable as e:
        return f"参考库转写 {job['id']}:连不上云机,下轮再试({e})"
    if wait is not None:
        if wait.state == "failed":
            jobs.start(refs, job["id"]); jobs.finish(refs, job["id"], error=wait.message)
        return f"参考库转写 {job['id']}:{wait.message}"
    jobs.start(refs, job["id"])
    code, out = run([WHISPER_ALIGN.model_dir], payload(pcm16k(Path(root) / rec["path"])))
    if code == SSH_UNREACHABLE or (code != 0 and CUDA_OOM in out):
        jobs.retry(refs, job["id"], "ssh 连不上 / 显存被抢,下个空档再跑")
        return f"参考库转写 {job['id']}:下个空档再跑"
    if code != 0:
        jobs.finish(refs, job["id"], error=f"云机转写脚本失败(退出码 {code}):{out.strip()[-400:]}")
        return f"参考库转写 {job['id']}:失败"
    try:
        r = parse(out)
    except ValueError as e:
        jobs.finish(refs, job["id"], error=str(e))
        return f"参考库转写 {job['id']}:失败"
    ok = hanzi(r["text"]) >= MIN_HANZI
    src = settings.get(refs, "hongguo")
    rel = Path(rec["path"]).with_suffix(".transcript.json").as_posix()
    body = json.dumps({"text": r["text"], "chars": r["chars"]}, ensure_ascii=False)
    (Path(root) / rel).write_text(body, encoding="utf-8")
    import hashlib
    at = clock.now_iso()
    with refs.write() as c:
        cid = c.execute("INSERT INTO ref_content (work_id, kind, unit, title, path, sha256, origin, verdict, reason, fetched_at, due_at, hard_due_at, meta_json)"
                        " VALUES (?, 'transcript', ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)",
                        (rec["work_id"], rec["unit"], f"第 {rec['unit']} 集转写", rel, hashlib.sha256(body.encode()).hexdigest(),
                         f"whisper 转写录屏 #{rec['id']}", "ok" if ok else "rejected", "" if ok else f"几乎没有台词({hanzi(r['text'])} 个汉字)",
                         at, rec["due_at"], rec["hard_due_at"], json.dumps({"chunks": r["chunks"], "asr_sec": r["asr_sec"]}))).lastrowid
    if ok:
        fingerprint.add(refs, cid, r["text"])
    jobs.finish(refs, job["id"])
    rjournal.append(root, "transcribe", source="hongguo", work_id=rec["work_id"], job=job["id"], chars=len(r["chars"]),
                    cloud="音频经 ssh stdin 送达,云机不落盘")
    return f"参考库转写 {job['id']}:{hanzi(r['text'])} 个汉字"

class Driver(Protocol):
    def open(self, series_id: str) -> None: ...                 # 只打开 /player/<剧号>(robots 允许);不打开 /player/<剧>/<集>
