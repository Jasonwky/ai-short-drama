# -*- coding: utf-8 -*-
"""韵律量法(spike-results/2026-10-31-降低AI感调研.md §12 同一套):出声占比、音高起伏(p10–p90 半音)。
念法模板入库前代码判(spec 2026-10-31 §9 V2):阈值来自参考句实测(出声 99 %、起伏 18.8 半音)与用户选定版(86 %、16.8 半音),**暂定**。"""
from __future__ import annotations

import numpy as np

MIN_OCCUPANCY = 0.85
MIN_RANGE_ST = 15.0
_FRAME, _HOP = 0.04, 0.01


def _frames(x: np.ndarray, sr: int, win: float, hop: float):
    n, h = int(win * sr), int(hop * sr)
    for i in range(0, max(len(x) - n, 0), h):
        yield x[i:i + n]


def occupancy(x: np.ndarray, sr: int, thr_db: float = -35.0) -> float:
    e = [20 * np.log10(np.sqrt((f ** 2).mean()) + 1e-9) for f in _frames(x, sr, 0.02, 0.02)]
    return float(np.mean(np.array(e) > thr_db)) if e else 0.0


def pitch_range_st(x: np.ndarray, sr: int) -> float:
    lo, hi = int(sr / 450), int(sr / 65)
    f0 = []
    for f in _frames(x, sr, _FRAME, _HOP):
        f = f - f.mean()
        if np.sqrt((f ** 2).mean()) < 0.02:
            continue
        ac = np.correlate(f, f, "full")[len(f) - 1:]
        k = lo + int(np.argmax(ac[lo:hi]))
        if ac[k] / (ac[0] + 1e-9) > 0.45:
            f0.append(sr / k)
    if len(f0) < 5:
        return 0.0
    st = 12 * np.log2(np.array(f0) / np.median(f0))
    return float(np.percentile(st, 90) - np.percentile(st, 10))


def template_problems(x: np.ndarray, sr: int) -> list[str]:
    out = []
    occ, rng = occupancy(x, sr), pitch_range_st(x, sr)
    if occ < MIN_OCCUPANCY:
        out.append(f"出声占比 {occ:.0%} < {MIN_OCCUPANCY:.0%}(停顿太多,不像一口气连着说)")
    if rng < MIN_RANGE_ST:
        out.append(f"音高起伏 {rng:.1f} 半音 < {MIN_RANGE_ST:g}(抑扬顿挫不够)")
    return out


def load_wav(path) -> tuple[np.ndarray, int]:
    """单声道 float32;多声道取平均。只读 16 位 PCM wav(我们自己的产物都是)。"""
    import wave
    with wave.open(str(path), "rb") as w:
        sr, ch = w.getframerate(), w.getnchannels()
        x = np.frombuffer(w.readframes(w.getnframes()), np.int16).astype(np.float32) / 32768
    return (x.reshape(-1, ch).mean(1) if ch > 1 else x), sr
```

# -*- coding: utf-8 -*-
"""**在云机上跑**:IndexTTS2 按定稿音色逐句克隆、念法模板引导、去停顿、音高起伏放大,拼成一条。源码经 SSH 的 stdin 送过去:

    <indextts venv python> - <checkpoints 目录> <任务 JSON>

任务 JSON:`{"lines": [[说话人, 台词]], "voices": {说话人: {"audio": 云机路径}}, "templates": [{"audio": 云机路径}](逐句),
"seed", "gap", "lead", "tail", "recipe": {emo_alpha, pitch_widen, squeeze_gap_sec, squeeze_db}, "sr_out"}`。
第 k 句用种子 seed + k(V4)。stdout 最后一行 `{"wav_b64","sr","sec","spans","gen_sec"}`(与 tts_remote 同格式);
退出码 2 = 文件不在,3 = 显存不够。配方见 spec 2026-10-31-IndexTTS2港片配音-design.md §2。
两处兼容垫片(云机实测必需):torchaudio 2.11 的 save 要 torchcodec → 换 soundfile;HF 走 hf-mirror。
本模块**顶层只用标准库**。
"""
from __future__ import annotations

import base64
import io
import json
import os
import re
import sys
import time
from pathlib import Path

INNER_PAUSE = re.compile(r"[,,、;;::]")


def strip_inner_pauses(text: str) -> str:
    """去掉句内逗号 / 顿号 / 分号 / 冒号(句末 。!? 保留),让模型连着念。"""
    return INNER_PAUSE.sub("", text)


def main(argv: list[str]) -> int:
    ckpt_dir, job_json = argv
    job = json.loads(job_json)
    files = [ckpt_dir, *(v["audio"] for v in job["voices"].values()), *(t["audio"] for t in job["templates"])]
    for p in files:
        if not Path(p).exists():
            print(f"不存在:{p}", file=sys.stderr)
            return 2
    os.environ.setdefault("HF_ENDPOINT", "https://hf-mirror.com")
    os.environ.setdefault("HF_HOME", "/root/autodl-tmp/hf")
    repo_dir = str(Path(ckpt_dir).parent)
    sys.path.insert(0, repo_dir)
    os.chdir(repo_dir)
    import numpy as np
    import pyworld as pw
    import soundfile as sf
    import torch
    import torchaudio

    def _save(path, wav, sr, *a, **k):
        sf.write(str(path), wav.detach().cpu().float().numpy().T, sr)
    torchaudio.save = _save
    from indextts.infer_v2 import IndexTTS2

    r = job["recipe"]

    def squeeze(x, sr):
        fr = int(0.01 * sr)
        keep, run = [], 0
        for k in range(0, len(x), fr):
            seg = x[k:k + fr]
            run = run + 1 if 20 * np.log10(np.sqrt((seg ** 2).mean()) + 1e-9) < r["squeeze_db"] else 0
            if run * 0.01 <= r["squeeze_gap_sec"]:
                keep.append(seg)
        return np.concatenate(keep) if keep else x

    def widen(x, sr):
        x = x.astype(np.float64)
        f0, t = pw.harvest(x, sr)
        sp, ap = pw.cheaptrick(x, f0, t, sr), pw.d4c(x, f0, t, sr)
        v = f0 > 0
        if v.sum() < 5:
            return x.astype(np.float32)
        med = np.median(np.log(f0[v]))
        f1 = f0.copy()
        f1[v] = np.exp(med + (np.log(f0[v]) - med) * r["pitch_widen"])
        return pw.synthesize(f1, sp, ap, sr).astype(np.float32)

    t0 = time.time()
    tmp = Path("/tmp/idx_remote")
    tmp.mkdir(exist_ok=True)
    try:
        tts = IndexTTS2(cfg_path=f"{ckpt_dir}/config.yaml", model_dir=ckpt_dir, use_fp16=True)
        clips = []
        for k, ((who, line), tpl) in enumerate(zip(job["lines"], job["templates"])):
            torch.manual_seed(int(job["seed"]) + k)
            out = tmp / f"{k}.wav"
            tts.infer(spk_audio_prompt=job["voices"][who]["audio"], text=strip_inner_pauses(line), output_path=str(out),
                      emo_audio_prompt=tpl["audio"], emo_alpha=r["emo_alpha"], interval_silence=0)
            x, sr = sf.read(str(out))
            x = x.mean(1) if x.ndim > 1 else x
            x = widen(squeeze(x.astype(np.float32), sr), sr)
            n = int(round(len(x) * job["sr_out"] / sr))                  # 重采样到 sr_out(线性插值,V5)
            clips.append(np.interp(np.linspace(0, len(x) - 1, n), np.arange(len(x)), x).astype(np.float32))
    except torch.cuda.OutOfMemoryError as e:
        print(f"显存不够(ComfyUI 可能正占着卡):CUDA out of memory {e}", file=sys.stderr)
        return 3
    sr = job["sr_out"]
    silence = lambda sec: np.zeros(int(round(sec * sr)), dtype=np.float32)   # noqa: E731
    parts, spans, t = [silence(job["lead"])], [], job["lead"]
    for k, clip in enumerate(clips):
        if k:
            parts.append(silence(job["gap"]))
            t += job["gap"]
        parts.append(clip)
        spans.append([round(t, 3), round(t + len(clip) / sr, 3)])
        t += len(clip) / sr
    parts.append(silence(job["tail"]))
    audio = np.clip(np.concatenate(parts), -1, 1)
    buf = io.BytesIO()
    sf.write(buf, audio, sr, format="WAV", subtype="PCM_16")
    print(json.dumps({"wav_b64": base64.b64encode(buf.getvalue()).decode(), "sr": sr, "sec": round(len(audio) / sr, 3),
                      "spans": spans, "gen_sec": round(time.time() - t0, 1)}))
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
```
