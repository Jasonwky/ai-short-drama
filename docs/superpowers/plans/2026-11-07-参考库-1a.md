```python
# -*- coding: utf-8 -*-
"""参考库的「当天」一律按北京时间零点切换(spec 1.1 第 6 条):每日上限、熔断、补拉都用它。"""
from __future__ import annotations

from datetime import date, datetime, timedelta, timezone

BJ = timezone(timedelta(hours=8))


def now_iso() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds")


def today(at: str | None = None) -> str:
    """at 是带时区的 ISO 时间(默认现在),返回北京日期 YYYY-MM-DD。"""
    t = datetime.fromisoformat(at) if at else datetime.now(timezone.utc)
    return t.astimezone(BJ).date().isoformat()


def days_between(a: str, b: str) -> int:
    return (date.fromisoformat(b) - date.fromisoformat(a)).days


def add_days(at: str, days: int) -> str:
    return (datetime.fromisoformat(at) + timedelta(days=days)).isoformat(timespec="seconds")
```

```python
# -*- coding: utf-8 -*-
"""参考库的库(spec 1.2):单独一个文件 `refs.db`,结构版本从 v1 起,与剧库的版本号无关。

连接、写锁、`cached_statements=0` 沿用 `tools.drama.db.Store`(网页线程与抓取线程共用连接,见那里的说明);
只换结构与迁移。不可改的规矩由触发器强制:设置改动记录、热度、指纹只许追加;任务参数不可改;内容行不许删(清理只置 purged_at)。
1c 加 ref_unit / ref_card 时走 REFS_MIGRATIONS[1]。
"""
from __future__ import annotations

from tools.drama.db import Store

REFS_VERSION = 1

REFS_SCHEMA = """
CREATE TABLE meta (key TEXT PRIMARY KEY, value TEXT NOT NULL);

CREATE TABLE ref_source (
  key TEXT PRIMARY KEY,
  mode TEXT NOT NULL DEFAULT 'C' CHECK (mode IN ('A', 'B', 'C')),
  cap_fetch INTEGER CHECK (cap_fetch IS NULL OR cap_fetch >= 0),
  cap_breakdown INTEGER CHECK (cap_breakdown IS NULL OR cap_breakdown >= 0),
  retention_days INTEGER CHECK (retention_days IS NULL OR retention_days >= 1),
  chapters_n INTEGER CHECK (chapters_n IS NULL OR chapters_n >= 1),
  top_k INTEGER CHECK (top_k IS NULL OR top_k >= 1),
  state TEXT NOT NULL DEFAULT 'ok' CHECK (state IN ('ok', 'tripped', 'paused')),
  state_reason TEXT NOT NULL DEFAULT '',
  state_until TEXT NOT NULL DEFAULT '',
  fail_streak INTEGER NOT NULL DEFAULT 0,
  last_ranks_day TEXT NOT NULL DEFAULT '',
  last_sitemap_at TEXT NOT NULL DEFAULT ''
);

CREATE TABLE ref_setting_log (
  id INTEGER PRIMARY KEY,
  source TEXT NOT NULL REFERENCES ref_source(key),
  field TEXT NOT NULL,
  old TEXT,
  new TEXT,
  at TEXT NOT NULL
);
CREATE TRIGGER ref_setting_log_no_update BEFORE UPDATE ON ref_setting_log
BEGIN SELECT RAISE(ABORT, '设置改动记录写下不可改'); END;
CREATE TRIGGER ref_setting_log_no_delete BEFORE DELETE ON ref_setting_log
BEGIN SELECT RAISE(ABORT, '设置改动记录写下不可删'); END;

CREATE TABLE ref_work (
  id INTEGER PRIMARY KEY,
  source TEXT NOT NULL REFERENCES ref_source(key),
  external_id TEXT NOT NULL,
  title TEXT NOT NULL DEFAULT '',
  author TEXT NOT NULL DEFAULT '',
  url TEXT NOT NULL DEFAULT '',
  tags_json TEXT NOT NULL DEFAULT '[]',
  intro TEXT NOT NULL DEFAULT '',
  roles_json TEXT NOT NULL DEFAULT '[]',
  followed INTEGER NOT NULL DEFAULT 0,
  blocked INTEGER NOT NULL DEFAULT 0,
  candidate_reason TEXT NOT NULL DEFAULT '',
  candidate_day TEXT NOT NULL DEFAULT '',
  first_seen TEXT NOT NULL,
  UNIQUE (source, external_id)
);

CREATE TABLE ref_heat (
  id INTEGER PRIMARY KEY,
  work_id INTEGER NOT NULL REFERENCES ref_work(id),
  board TEXT NOT NULL,
  day TEXT NOT NULL,
  at TEXT NOT NULL,
  rank INTEGER,
  heat REAL,
  raw_json TEXT NOT NULL DEFAULT '{}',
  UNIQUE (work_id, board, day)
);
CREATE TRIGGER ref_heat_no_update BEFORE UPDATE ON ref_heat
BEGIN SELECT RAISE(ABORT, '热度写入后不改'); END;
CREATE TRIGGER ref_heat_no_delete BEFORE DELETE ON ref_heat
BEGIN SELECT RAISE(ABORT, '热度写入后不删'); END;

CREATE TABLE ref_job (
  id INTEGER PRIMARY KEY,
  source TEXT NOT NULL REFERENCES ref_source(key),
  kind TEXT NOT NULL CHECK (kind IN ('ranks', 'meta', 'content', 'sitemap')),
  work_id INTEGER REFERENCES ref_work(id),
  params_json TEXT NOT NULL DEFAULT '{}',
  auto INTEGER NOT NULL DEFAULT 0,
  day TEXT NOT NULL,
  state TEXT NOT NULL DEFAULT 'waiting' CHECK (state IN ('waiting', 'running', 'done', 'failed')),
  attempts INTEGER NOT NULL DEFAULT 0,
  error TEXT NOT NULL DEFAULT '',
  created_at TEXT NOT NULL,
  started_at TEXT NOT NULL DEFAULT '',
  finished_at TEXT NOT NULL DEFAULT ''
);
CREATE TRIGGER ref_job_params_immutable BEFORE UPDATE OF source, kind, work_id, params_json, auto, day ON ref_job
BEGIN SELECT RAISE(ABORT, '抓取任务的参数不可改 —— 要改就新排一条'); END;

CREATE TABLE ref_content (
  id INTEGER PRIMARY KEY,
  work_id INTEGER NOT NULL REFERENCES ref_work(id),
  kind TEXT NOT NULL CHECK (kind IN ('chapter', 'review', 'manual')),
  unit INTEGER,
  title TEXT NOT NULL DEFAULT '',
  path TEXT NOT NULL,
  sha256 TEXT NOT NULL,
  origin TEXT NOT NULL,
  verdict TEXT NOT NULL CHECK (verdict IN ('ok', 'rejected')),
  reason TEXT NOT NULL DEFAULT '',
  fetched_at TEXT NOT NULL,
  due_at TEXT NOT NULL,
  hard_due_at TEXT NOT NULL,
  purged_at TEXT NOT NULL DEFAULT ''
);
CREATE TRIGGER ref_content_no_delete BEFORE DELETE ON ref_content
BEGIN SELECT RAISE(ABORT, '内容行不删 —— 到期清理只删文件、置 purged_at(指纹还引用它)'); END;

CREATE TABLE ref_fingerprint (
  content_id INTEGER NOT NULL REFERENCES ref_content(id),
  pos INTEGER NOT NULL,
  h INTEGER NOT NULL,
  PRIMARY KEY (content_id, pos)
);
CREATE INDEX ref_fingerprint_h ON ref_fingerprint(h);
CREATE TRIGGER ref_fingerprint_no_update BEFORE UPDATE ON ref_fingerprint
BEGIN SELECT RAISE(ABORT, '指纹永久保留、不可改'); END;
CREATE TRIGGER ref_fingerprint_no_delete BEFORE DELETE ON ref_fingerprint
BEGIN SELECT RAISE(ABORT, '指纹永久保留、不可删'); END;
"""

REFS_MIGRATIONS: dict[int, list[str]] = {}


class RefStore(Store):
    def _migrate(self) -> None:
        has_meta = self.conn.execute(
            "SELECT 1 FROM sqlite_master WHERE type = 'table' AND name = 'meta'").fetchone()
        if not has_meta:
            self.conn.executescript(REFS_SCHEMA)
            self.conn.execute("INSERT INTO meta (key, value) VALUES ('schema_version', ?)", (str(REFS_VERSION),))
            return
        row = self.conn.execute("SELECT value FROM meta WHERE key = 'schema_version'").fetchone()
        found = int(row[0]) if row else None
        if found is None or found > REFS_VERSION or any(v not in REFS_MIGRATIONS for v in range(found, REFS_VERSION)):
            raise RuntimeError(f"参考库结构版本 {row[0] if row else '缺失'} 与代码 {REFS_VERSION} 不符 —— 不许在不匹配的结构上继续写")
        for v in range(found, REFS_VERSION):
            with self.write() as c:
                for stmt in REFS_MIGRATIONS[v]:
                    c.execute(stmt)
                c.execute("UPDATE meta SET value = ? WHERE key = 'schema_version'", (str(v + 1),))
```

```python
# -*- coding: utf-8 -*-
"""参考库的操作留痕(三支柱):每次抓取的 URL / 状态码、当时的 robots 原文、设置改动、熔断 / 暂停、清理,
按北京日期一天一个 `journal/refs/<日>.jsonl`。**不存页面正文**(spec 1.1)。查看器在来源设置与作品详情里(只读)。"""
from __future__ import annotations

import json
from pathlib import Path

from tools.refs import clock

DIR = Path("journal") / "refs"


def append(root: Path, event: str, **fields) -> dict:
    rec = {"at": clock.now_iso(), "event": event, **fields}
    path = Path(root) / DIR / f"{clock.today(rec['at'])}.jsonl"
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("a", encoding="utf-8") as f:
        f.write(json.dumps(rec, ensure_ascii=False) + "\n")
    return rec


def read(root: Path, *, source: str | None = None, work_id: int | None = None, event: str | None = None,
         limit: int = 100) -> list[dict]:
    """新的在前。逐文件从新到旧读,够 limit 条就停。"""
    d = Path(root) / DIR
    if not d.is_dir():
        return []
    out: list[dict] = []
    for path in sorted(d.glob("*.jsonl"), reverse=True):
        for line in reversed(path.read_text(encoding="utf-8").splitlines()):
            rec = json.loads(line)
            if source and rec.get("source") != source:
                continue
            if work_id is not None and rec.get("work_id") != work_id:
                continue
            if event and rec.get("event") != event:
                continue
            out.append(rec)
            if len(out) >= limit:
                return out
    return out
```

```python
    p = []

    p = fetch_problems(src)
    for f in ("cap_fetch", "cap_breakdown", "top_k"):
        if src[f] is None:
            p.append(f"{INT_FIELDS[f][0]}没填(★ 必填)—— 切不到 A 全自动")

def _parse(field: str, raw: str):
    if field == "mode":
        if raw not in MODES:
            raise ValueError(f"模式只能是 A / B / C(收到「{raw}」)")
        return raw
    label, low = INT_FIELDS[field]
    raw = raw.strip()
    if raw == "":
        return None
    try:
        v = int(raw)
    except ValueError:
        raise ValueError(f"{label}应为整数(收到「{raw}」)") from None
    if v < low:
        raise ValueError(f"{label}不能小于 {low}(收到 {v})")
    return v


def update(store, root: Path, key: str, form: dict[str, str]) -> list[str]:
    """改设置;返回真正改了的字段。不合法整组拒绝(ValueError),一个字段也不写。"""
    cur = get(store, key)
    parsed = {f: _parse(f, v) for f, v in form.items() if f in INT_FIELDS or f == "mode"}
    new = {**cur, **parsed}
    if key == "manual" and new["mode"] != "C":
        raise ValueError("人工投递只能是 C(全手动)")
    if new["mode"] == "A":
        p = auto_problems(new)
        if p:
            raise ValueError(";".join(p))
    changed = [f for f in parsed if parsed[f] != cur[f]]
    if not changed:
        return []
    at = clock.now_iso()
    with store.write() as c:
        for f in changed:
            c.execute(f"UPDATE ref_source SET {f} = ? WHERE key = ?", (parsed[f], key))
            old, nv = cur[f], parsed[f]
            c.execute("INSERT INTO ref_setting_log (source, field, old, new, at) VALUES (?, ?, ?, ?, ?)",
                      (key, f, None if old is None else str(old), None if nv is None else str(nv), at))
    for f in changed:
        rjournal.append(root, "setting", source=key, field=f,
                        old=None if cur[f] is None else str(cur[f]), new=None if parsed[f] is None else str(parsed[f]))
    return changed


def _set_state(store, root: Path, key: str, state: str, reason: str, until: str = "") -> None:
    with store.write() as c:
        c.execute("UPDATE ref_source SET state = ?, state_reason = ?, state_until = ? WHERE key = ?",
                  (state, reason, until, key))
    rjournal.append(root, "state", source=key, state=state, reason=reason, until=until)


def trip(store, root: Path, key: str, reason: str, day: str) -> None:
    _set_state(store, root, key, "tripped", reason, day)


def pause(store, root: Path, key: str, reason: str) -> None:
    _set_state(store, root, key, "paused", reason)


def reset_state(store, root: Path, key: str) -> None:
    with store.write() as c:
        c.execute("UPDATE ref_source SET fail_streak = 0 WHERE key = ?", (key,))
    _set_state(store, root, key, "ok", "")


def blocked_reason(src: dict, today: str) -> str | None:
    if src["state"] == "paused":

    if src["state"] == "tripped" and src["state_until"] >= today:

    return None
```

```python
# -*- coding: utf-8 -*-
"""参考库取页(spec 1.1 第 1、2 条、4.1)。

- UA 写明身份;每个 Fetcher 实例(= 一轮任务)**重读一次 robots**,原文进 journal;robots 禁止 / 读不到 → disallowed(来源暂停);
- 同一域名两次请求之间至少 interval 秒;
- 403 / 412 / 429 或页面里有验证码字样 → blocked(当天熔断,不重试);连不上 / 超时 → unreachable(下一轮重试);
- `browser_get`:普通 HTTP 抓不到的页面用 Playwright **无头**取(用户 2026-11-06 定的唯一无头例外,只限参考库后台抓取);
  用哪种由适配器写死,这里不做运行时切换。
"""
from __future__ import annotations

import time
import urllib.error
import urllib.request
from collections.abc import Callable
from pathlib import Path
from urllib.parse import urlsplit
from urllib.robotparser import RobotFileParser

from tools.refs import rjournal

UA = "Mozilla/5.0 (compatible; ai-short-drama-refs/1.0; reference research, low rate)"
BLOCK_STATUS = {403, 412, 429}
CAPTCHA = ("安全验证", "拖动下方滑块", "访问被拒绝", "captcha")
TIMEOUT = 20


class FetchError(Exception):
    def __init__(self, kind: str, message: str, status: int | None = None):
        super().__init__(message)
        self.kind, self.status = kind, status


def urllib_opener(url: str, headers: dict, timeout: float):
    req = urllib.request.Request(url, headers=headers)
    try:
        with urllib.request.urlopen(req, timeout=timeout) as r:
            return r.status, r.read(), r.geturl()
    except urllib.error.HTTPError as e:
        return e.code, e.read() or b"", url


def headless_html(url: str) -> str:
    from playwright.sync_api import sync_playwright
    with sync_playwright() as p:
        b = p.chromium.launch(headless=True)      # 唯一无头例外:参考库后台抓取(CLAUDE.md、spec 1.1)
        try:
            page = b.new_page(user_agent=UA)
            page.goto(url, wait_until="domcontentloaded", timeout=30000)
            page.wait_for_timeout(3000)
            return page.content()
        finally:
            b.close()


class Fetcher:
    def __init__(self, root: Path, source: str, *, opener: Callable = urllib_opener,
                 browser: Callable[[str], str] = headless_html, interval: float = 3.0,
                 sleep: Callable[[float], None] = time.sleep, clock: Callable[[], float] = time.monotonic):
        self.root, self.source, self.opener, self.browser = Path(root), source, opener, browser
        self.interval, self.sleep, self.clock = interval, sleep, clock
        self._robots: dict[str, RobotFileParser] = {}
        self._last: dict[str, float] = {}

    def _host(self, url: str) -> str:
        u = urlsplit(url)
        return f"{u.scheme}://{u.netloc}"

    def _wait(self, host: str) -> None:
        last = self._last.get(host)
        if last is not None:
            gap = self.interval - (self.clock() - last)
            if gap > 0:
                self.sleep(gap)
        self._last[host] = self.clock()

    def _open(self, url: str):
        try:
            return self.opener(url, {"User-Agent": UA}, TIMEOUT)
        except (urllib.error.URLError, TimeoutError, OSError) as e:
            rjournal.append(self.root, "fetch", source=self.source, url=url, status=None, error=str(e))
            raise FetchError("unreachable", f"连不上 {url}:{e}") from e

    def _check_robots(self, url: str) -> None:
        host = self._host(url)
        rp = self._robots.get(host)
        if rp is None:
            self._wait(host)
            status, body, _ = self._open(host + "/robots.txt")
            text = body.decode("utf-8", "replace")
            rjournal.append(self.root, "robots", source=self.source, url=host + "/robots.txt", status=status, text=text[:4000])
            rp = RobotFileParser()
            if status == 404:
                rp.parse([])
            elif status == 200:
                rp.parse(text.splitlines())
            else:
                raise FetchError("disallowed", f"robots.txt 读不到(HTTP {status})—— 不知道让不让抓,先不抓", status)
            self._robots[host] = rp
        if not rp.can_fetch(UA, url):
            raise FetchError("disallowed", f"robots.txt 禁止抓 {url}")

    def get(self, url: str, *, encoding: str = "utf-8") -> str:
        self._check_robots(url)
        self._wait(self._host(url))
        status, body, final = self._open(url)
        text = body.decode(encoding, "replace")
        rjournal.append(self.root, "fetch", source=self.source, url=url, status=status, bytes=len(body))
        if status in BLOCK_STATUS or any(m in text[:50000] for m in CAPTCHA):
            raise FetchError("blocked", f"被风控 / 验证码挡住(HTTP {status}):{url}", status)
        if status != 200:
            raise FetchError("http", f"HTTP {status}:{url}", status)
        return text

    def browser_get(self, url: str) -> str:
        self._check_robots(url)
        self._wait(self._host(url))
        html = self.browser(url)
        rjournal.append(self.root, "fetch", source=self.source, url=url, status=200, bytes=len(html), via="headless")
        if any(m in html[:50000] for m in CAPTCHA):
            raise FetchError("blocked", f"被验证码挡住:{url}")
        return html
```

```python
"""单测快照脱敏(spec 4.2):页面结构留下,版权正文与个人信息不进 git。

- 汉字逐个映射成假字(固定映射、长度不变);KEEP 里的标记词原样留(解析器 / 判定靠它们);私用区字符原样留;
- 剧评昵称、头像清掉(JSON 与 HTML 转义两种写法)。

开发时抓一份快照(**不是产品功能**,只给写适配器的人用):
    python -X utf8 -m tools.refs.snapshot <url> <输出文件> [--encoding gb18030]
"""
from __future__ import annotations

import argparse
import re
import sys
from pathlib import Path

KEEP = ("本章共", "字", "热度", "万", "亿", "评分", "收藏", "点赞", "人次读过", "第", "章", "集", "免费",
        "下载", "开通会员", "订阅", "请至", "阅读", "积分", "热播", "榜", "全", "饰", "安全验证", "正文", "完结", "连载")
FAKE = "甲乙丙丁戊己庚辛壬癸子丑寅卯辰巳午未申酉戌亥"
_KEEP_RE = re.compile("|".join(sorted(map(re.escape, KEEP), key=len, reverse=True)))
_NICK = [(re.compile(r'("user_name"\s*:\s*")[^"]*(")'), r"\1已去除\2"),
         (re.compile(r'("user_avatar"\s*:\s*")[^"]*(")'), r"\1\2"),
         (re.compile(r"(&quot;user_name&quot;:&quot;).*?(&quot;)"), r"\1已去除\2"),
         (re.compile(r"(&quot;user_avatar&quot;:&quot;).*?(&quot;)"), r"\1\2")]


def _fake(ch: str) -> str:
    return FAKE[ord(ch) % len(FAKE)] if ch not in FAKE else ch


def sanitize(text: str) -> str:
    for pat, rep in _NICK:
        text = pat.sub(rep, text)
    out, i = [], 0
    for m in _KEEP_RE.finditer(text):
        out.append(re.sub(r"[一-鿿]", lambda c: _fake(c.group()), text[i:m.start()]))
        out.append(m.group())
        i = m.end()
    out.append(re.sub(r"[一-鿿]", lambda c: _fake(c.group()), text[i:]))
    return "".join(out)


def main(argv: list[str] | None = None) -> int:
    from tools.refs.net import Fetcher
    ap = argparse.ArgumentParser(description="抓一份脱敏快照给适配器单测用")
    ap.add_argument("url")
    ap.add_argument("out")
    ap.add_argument("--encoding", default="utf-8")
    a = ap.parse_args(argv)
    html = Fetcher(Path("tmp/snapshot"), "dev").get(a.url, encoding=a.encoding)
    Path(a.out).parent.mkdir(parents=True, exist_ok=True)
    Path(a.out).write_text(sanitize(html), encoding="utf-8")
    print(f"已存 {a.out}({len(html)} 字符,已脱敏)")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
# -*- coding: utf-8 -*-
"""适配器统一契约(spec 1.1 第 1 条)。

解析函数是**纯函数**(HTML 进、数据类出),单测吃脱敏快照;取页函数只负责拼地址、调 Fetcher。
解析不出该有的东西 → AdapterError(网站返回了 200,是我们的适配器坏了,多半网页改版)—— 不当作「今天没有新作品」。
"""
from __future__ import annotations

import re
from collections.abc import Callable
from dataclasses import dataclass, field


class AdapterError(Exception):
    """网站返回了 200,但没解析出该有的东西。"""


def need(cond, message: str) -> None:
    if not cond:
        raise AdapterError(message)


@dataclass(frozen=True)
class RankItem:
    external_id: str
    title: str
    rank: int
    heat: float | None
    raw: dict


@dataclass(frozen=True)
class UnitRef:
    number: int
    title: str
    url: str
    free: bool


@dataclass
class WorkMeta:
    external_id: str
    title: str
    author: str
    url: str
    tags: list[str]
    intro: str
    heat: float | None                       # 本平台详情页的总热度口径(只在同平台内比)
    stats: dict = field(default_factory=dict)
    units: list[UnitRef] = field(default_factory=list)
    roles: list[str] = field(default_factory=list)     # 角色名(1c 换名用);不存演员昵称


@dataclass(frozen=True)
class Unit:
    number: int
    title: str
    text: str
    self_words: int | None                   # 页面自报字数(有才填)


@dataclass(frozen=True)
class Review:
    text: str
    rating: int | None
    likes: float | None
    at: str


@dataclass(frozen=True)
class Adapter:
    key: str
    boards: dict[str, str]                                   # 榜单键 → 中文名
    content: str                                             # "chapters" / "reviews" / "none"
    list_ranks: Callable                                      # (fetcher, board) -> list[RankItem]
    fetch_meta: Callable                                      # (fetcher, external_id) -> WorkMeta
    fetch_unit: Callable | None = None                        # (fetcher, meta, UnitRef) -> Unit
    fetch_reviews: Callable | None = None                     # (fetcher, external_id) -> list[Review]
    match: Callable[[str], str | None] = lambda url: None    # 作品链接 → external_id
    sitemap: Callable | None = None                           # (fetcher) -> list[external_id]


_UNIT = {"万": 1e4, "亿": 1e8}


def num(text) -> float | None:
    """「8339万热度」「314.3万」「22,274,893,824」→ 数;没有数字返回 None。"""
    if text is None:
        return None
    m = re.search(r"(\d[\d,]*(?:\.\d+)?)\s*([万亿])?", str(text))
    if not m:
        return None
    v = float(m.group(1).replace(",", "")) * _UNIT.get(m.group(2) or "", 1)
    return int(v) if v == int(v) else v


def hanzi(text: str) -> int:
    return len(re.findall(r"[一-鿿]", text))


def check_ranks(items: list[RankItem]) -> list[RankItem]:
    need(items, "榜单解析出 0 条 —— 网页可能改版了")
    for it in items:
        need(it.external_id and it.title and it.rank > 0, f"榜单条目缺编号 / 标题 / 名次:{it}")
    ranks = [it.rank for it in items]
    need(len(set(ranks)) == len(ranks), f"榜单名次重复:{ranks}")
    return items
```

```python
"""红果短剧(spec 第 0 节;调研 2026-11-07-红果短剧内容获取实测.md)。

能拿:4 个榜单、详情(热度 / 点赞 / 收藏 / 标签 / 角色 / 简介 / 集数)、剧评(详情页自带的那几条)、sitemap。
不碰:`/player/*/*`、`/series/`、`/query/`(robots 禁止)。正片录屏是 1b,而且永远要人点。
剧评**不取昵称与头像**(spec 1.1 第 4 条)。
"""
from __future__ import annotations

import html as htmllib
import json
import re

from tools.refs.sources.base import Adapter, RankItem, Review, UnitRef, WorkMeta, check_ranks, need, num

BASE = "https://hongguoduanju.com"
BOARDS = {"hot-drama": "热播榜", "hot-real-drama": "真人剧榜", "hot-ai-drama": "AI 剧榜", "hot-comic-drama": "漫剧榜"}
WEB_EPISODES = 5      # 登录后网页能看的集数(调研实测:第 6 集起带锁)
RAW = ("heatText", "scoreText", "favoriteText", "likeText", "tags", "episodeCount", "openEpisodeCount", "statusTags")


def _fn_args(page: str, name: str) -> dict:
    for m in re.finditer(r"data-fn-args='([^']*)'", page):
        args = json.loads(htmllib.unescape(m.group(1)))
        if len(args) >= 3 and args[1] == name:
            return args[2]
    raise AdapterError(f"页面里没有 {name} 数据块 —— 网页可能改版了")


def parse_rank(page: str) -> list[RankItem]:
    c = _fn_args(page, "content")
    need(c.get("isSuccess") and isinstance(c.get("rankList"), list), "榜单数据块不完整")
    return check_ranks([RankItem(str(x["seriesId"]), x["title"], int(x["rank"]), num(x.get("heatText")),
                                 {k: x.get(k) for k in RAW}) for x in c["rankList"]])


def _router(page: str) -> dict:
    i = page.find("_ROUTER_DATA = ")
    need(i >= 0, "页面里没有 _ROUTER_DATA —— 网页可能改版了")
    d, _ = json.JSONDecoder().raw_decode(page, i + len("_ROUTER_DATA = "))
    return d.get("loaderData") or {}


def _tag(t) -> str:
    return t if isinstance(t, str) else (t.get("tag_name") or t.get("name") or "")


def parse_detail(page: str) -> tuple[WorkMeta, list[Review]]:
    p = _router(page).get("detail_page") or {}
    sd, ss = p.get("seriesDetail") or {}, p.get("seriesSocialInfo") or {}
    need(sd.get("series_id") and sd.get("series_name"), "详情页缺剧号 / 剧名")
    sid = str(sd["series_id"])
    rv = ss.get("reviews") or {}
    meta = WorkMeta(
        external_id=sid, title=sd["series_name"], author="", url=f"{BASE}/detail?series_id={sid}",
        tags=[x for x in map(_tag, sd.get("tags") or []) if x], intro=sd.get("series_intro", ""),
        heat=num((ss.get("hot_score_data") or {}).get("score")),
        stats={"episodes": sd.get("episode_cnt"), "likes": num(ss.get("series_like_count")),
               "favorites": num(ss.get("series_favorite_count")), "reviews_total": num(rv.get("total_count")),
               "rank_label": ss.get("rank_label", "")},
        units=[UnitRef(n, f"第 {n} 集", f"{BASE}/player/{sid}", True)
               for n in range(1, min(WEB_EPISODES, sd.get("episode_cnt") or 0) + 1)],
        roles=[c["sub_title"].removeprefix("饰").strip() for c in sd.get("celebrities") or []
               if str(c.get("sub_title", "")).startswith("饰")])
    reviews = [Review(r["content"], r.get("rating"), num(r.get("digg_count")), str(r.get("create_time", "")))
               for r in rv.get("reviews") or [] if r.get("content")]
    return meta, reviews


def parse_sitemap(xml: str) -> list[str]:
    ids = re.findall(r"detail\?series_id=(\d+)", xml)
    need(ids, "sitemap 解析出 0 条")
    return ids


def _match(url: str) -> str | None:
    m = re.search(r"hongguoduanju\.com/(?:detail\?series_id=|player/)(\d+)", url)
    return m.group(1) if m else None


ADAPTER = Adapter(
    key="hongguo", boards=BOARDS, content="reviews",
    list_ranks=lambda f, board: parse_rank(f.get(f"{BASE}/rank/{board}")),
    fetch_meta=lambda f, ext: parse_detail(f.get(f"{BASE}/detail?series_id={ext}"))[0],
    fetch_reviews=lambda f, ext: parse_detail(f.get(f"{BASE}/detail?series_id={ext}"))[1],
    match=_match,
    sitemap=lambda f: parse_sitemap(f.get(f"{BASE}/seo-important.xml")))
```

```python
# -*- coding: utf-8 -*-
"""晋江文学城(调研 2026-11-07-小说平台免费章节实测.md):robots 允许,免费章明文,页面编码 GB18030。
只抓免费章(章节行没有 [VIP]);积分作热度(只在晋江内比)。"""
from __future__ import annotations

import html as htmllib
import re

from tools.refs.sources.base import Adapter, RankItem, Unit, UnitRef, WorkMeta, check_ranks, need, num

BASE = "https://www.jjwxc.net"
ENC = "gb18030"
BOARDS = {"topten7": "总分排行榜"}
CHROME = ("[添加书签]", "[推荐给朋友]", "[收藏此章节]", "[投诉]")


def text_of(fragment: str) -> str:
    s = re.sub(r"<br\s*/?>|</p>", "\n", fragment)
    s = htmllib.unescape(re.sub(r"<[^>]+>", "", s))
    for c in CHROME:
        s = s.replace(c, "")
    return "\n".join(x.strip() for x in s.splitlines() if x.strip())


def parse_rank(page: str) -> list[RankItem]:
    items = []
    for row in re.findall(r"<tr[^>]*>(.*?)</tr>", page, re.S):
        m = re.search(r'href="onebook\.php\?novelid=(\d+)"[^>]*?class="tooltip">([^<]+)</a>', row, re.S)
        if not m:
            continue
        plain = text_of(re.sub(r'rel="[^"]*"', "", row))
        nums = [num(x) for x in re.findall(r"\d{1,3}(?:,\d{3})+", plain)]
        items.append(RankItem(m.group(1), m.group(2).strip(), len(items) + 1, max(nums) if nums else None, {}))
    return check_ranks(items)


def parse_book(page: str, ext: str) -> WorkMeta:
    t = re.search(r'<span itemprop="articleSection">([^<]*)</span>', page)
    need(t and t.group(1).strip(), "书页缺书名")
    genre = re.search(r'itemprop="genre">\s*([^<]+?)\s*</span>', page)
    score = re.search(r"文章积分[::]\s*(?:<[^>]+>\s*)*([\d,]+)", page)
    intro = re.search(r'<div[^>]*id="novelintro"[^>]*>(.*?)</div>', page, re.S)
    author = re.search(r'itemprop="author"[^>]*>\s*(?:<[^>]+>\s*)*([^<\s][^<]*)', page)
    units = []
    for row in re.findall(r'<tr[^>]*itemprop="chapter[^"]*"[^>]*>(.*?)</tr>', page, re.S):
        plain = text_of(row)
        n = re.match(r"(\d+)\s+(\S+)", plain)
        if n:
            units.append(UnitRef(int(n.group(1)), n.group(2), f"{BASE}/onebook.php?novelid={ext}&chapterid={n.group(1)}",
                                 "[VIP]" not in plain and "vip" not in row.lower()))
    need(units, "书页解析出 0 个章节")
    return WorkMeta(ext, t.group(1).strip(), author.group(1).strip() if author else "", f"{BASE}/onebook.php?novelid={ext}",
                    genre.group(1).split("-") if genre else [], text_of(intro.group(1)) if intro else "",
                    num(score.group(1)) if score else None, {}, units)


def parse_chapter(page: str, number: int) -> Unit:
    i = page.find('class="noveltext')
    need(i >= 0, "章节页没有正文块")
    body = page[i:]
    ends = [j for j in (body.find('<div id="favoriteshow'), body.find("note_danmu")) if j > 0]
    body = body[body.find(">") + 1: min(ends) if ends else len(body)]
    h2 = re.search(r"<h2[^>]*>(.*?)</h2>", page, re.S)
    text = text_of(re.sub(r"<h2[^>]*>.*?</h2>", "", body, flags=re.S))
    need(text, "章节正文为空")
    return Unit(number, text_of(h2.group(1)) if h2 else f"第{number}章", text, None)


ADAPTER = Adapter(
    key="jjwxc", boards=BOARDS, content="chapters",
    list_ranks=lambda f, board: parse_rank(f.get(f"{BASE}/topten.php?orderstr=7", encoding=ENC)),
    fetch_meta=lambda f, ext: parse_book(f.get(f"{BASE}/onebook.php?novelid={ext}", encoding=ENC), ext),
    fetch_unit=lambda f, meta, ref: parse_chapter(f.get(ref.url, encoding=ENC), ref.number),
    match=lambda url: (m.group(1) if (m := re.search(r"jjwxc\.net/onebook\.php\?novelid=(\d+)", url)) else None))
```

```python
"""17K 小说网(调研 2026-11-07-小说平台免费章节实测.md):robots 允许(禁 /azapp*、/ck),免费章明文,自报「本章共 N 字」。
https 主域实测连接超时,一律用 http。只抓目录里非 VIP 的章。"""
from __future__ import annotations

import html as htmllib
import re

from tools.refs.sources.base import Adapter, RankItem, Unit, UnitRef, WorkMeta, check_ranks, need, num

BASE = "http://www.17k.com"
BOARDS = {"top": "排行榜"}


def _text(fragment: str) -> str:
    s = htmllib.unescape(re.sub(r"<[^>]+>", "", re.sub(r"</p>|<br\s*/?>", "\n", fragment)))
    return "\n".join(x.strip() for x in s.splitlines() if x.strip())


def parse_rank(page: str) -> list[RankItem]:
    seen: dict[str, str] = {}
    for bid, title in re.findall(r'href="//www\.17k\.com/book/(\d+)\.html"[^>]*>([^<]*)<', page):
        if title.strip() and bid not in seen:
            seen[bid] = title.strip()
    return check_ranks([RankItem(b, t, i + 1, None, {}) for i, (b, t) in enumerate(seen.items())])


def parse_book(book: str, listing: str, ext: str) -> WorkMeta:
    title = re.search(r"<h1[^>]*>\s*(?:<a[^>]*>)?([^<]+)", book)
    need(title, "书页缺书名")
    reads = re.search(r'id="howmuchreadBook"[^>]*>(\d+)<', book)
    intro = re.search(r'<p class="intro"[^>]*>(.*?)</p>', book, re.S) or re.search(r'<dd[^>]*class="intro"[^>]*>(.*?)</dd>', book, re.S)
    units = []
    for i, (cid, attrs, span) in enumerate(re.findall(
            r'href="/chapter/' + ext + r'/(\d+)\.html"([^>]*)>\s*<span class="([^"]*)"', listing), 1):
        t = re.search(r'title="([^&"]*)', attrs)
        units.append(UnitRef(i, t.group(1).strip() if t else f"第{i}章", f"{BASE}/chapter/{ext}/{cid}.html", "vip" not in span))
    need(units, "目录解析出 0 个章节")
    return WorkMeta(ext, title.group(1).strip(), "", f"{BASE}/book/{ext}.html", [], _text(intro.group(1)) if intro else "",
                    num(reads.group(1)) if reads else None, {}, units)


def parse_chapter(page: str, number: int) -> Unit:
    i = page.find('class="readAreaBox')
    need(i >= 0, "章节页没有正文块")
    area = page[i:]
    h1 = re.search(r"<h1[^>]*>(.*?)</h1>", area, re.S)
    words = re.search(r"本章共\s*(\d+)\s*字", area)
    body = re.search(r'<div class="p">(.*?)</div>', area, re.S)
    need(body, "章节正文块为空")
    return Unit(number, _text(h1.group(1)) if h1 else f"第{number}章", _text(body.group(1)), int(words.group(1)) if words else None)


ADAPTER = Adapter(
    key="k17", boards=BOARDS, content="chapters",
    list_ranks=lambda f, board: parse_rank(f.get(f"{BASE}/top/")),
    fetch_meta=lambda f, ext: parse_book(f.get(f"{BASE}/book/{ext}.html"), f.get(f"{BASE}/list/{ext}.html"), ext),
    fetch_unit=lambda f, meta, ref: parse_chapter(f.get(ref.url), ref.number),
    match=lambda url: (m.group(1) if (m := re.search(r"17k\.com/(?:book|list)/(\d+)\.html", url)) else None))
```

```python
"""小说章节入库前的三条判定(spec 1.1 第 4 条;阈值出处 spike-results/2026-11-07-小说平台免费章节实测.md):
私用区 ≤ 1 %(超了疑似字体加密)、没有截断提示、正文汉字 ≥ 自报字数的 80 %(有自报时)。另拦太短的(< 200 汉字)。"""
from __future__ import annotations

import re

from tools.refs.sources.base import Unit, hanzi

PUA_MAX = 0.01
WORDS_MIN_RATIO = 0.8
MIN_HANZI = 200
TRUNCATION = ("下载App", "下载APP", "开通会员", "订阅本章", "请至", "App阅读", "APP阅读", "购买本章", "免费畅读")


def chapter(u: Unit) -> tuple[str, str]:
    chars = len(re.sub(r"\s", "", u.text))
    pua = len(re.findall(r"[-]", u.text))
    if chars and pua / chars > PUA_MAX:
        return "rejected", f"疑似字体加密:私用区字符 {pua}/{chars}({pua / chars:.1%})"
    hit = next((m for m in TRUNCATION if m in u.text), None)
    if hit:
        return "rejected", f"有截断提示「{hit}」—— 不是完整的免费章"
    n = hanzi(u.text)
    if n < MIN_HANZI:
        return "rejected", f"正文太短({n} 个汉字)"
    if u.self_words and n < WORDS_MIN_RATIO * u.self_words:
        return "rejected", f"字数对不上:正文 {n} 个汉字,页面自报 {u.self_words} 字"
    return "ok", ""
```

```python
# -*- coding: utf-8 -*-
"""原文指纹(spec 1.2、2.4):小说章节、视频转写、剧评,去掉标点空白后每个 WINDOW 字滑动窗口一个 64 位哈希,永久保留。
WINDOW = 12 是**暂定值**,1a 最后一个任务做标定实验后定(spike-results/2026-11-07-指纹重叠阈值标定.md)。"""
from __future__ import annotations

import hashlib
import re

WINDOW = 12


def normalize(text: str) -> str:
    return "".join(re.findall(r"[一-鿿-A-Za-z0-9]", text))


def _h(s: str) -> int:
    return int.from_bytes(hashlib.blake2b(s.encode("utf-8"), digest_size=8).digest(), "big", signed=True)


def windows(text: str) -> list[int]:
    t = normalize(text)
    return [_h(t[i:i + WINDOW]) for i in range(len(t) - WINDOW + 1)]


def add(store, content_id: int, text: str) -> int:
    hs = windows(text)
    with store.write() as c:
        c.executemany("INSERT OR IGNORE INTO ref_fingerprint (content_id, pos, h) VALUES (?, ?, ?)",
                      [(content_id, i, h) for i, h in enumerate(hs)])
    return len(hs)


def overlaps(store, text: str) -> list[dict]:
    """text 里与库中原文连续 ≥ WINDOW 字相同的片段:[{content_id, start, length}],按出现先后。"""
    t = normalize(text)
    hs = [_h(t[i:i + WINDOW]) for i in range(len(t) - WINDOW + 1)]
    hits: dict[int, int] = {}
    for i, h in enumerate(hs):
        row = store.conn.execute("SELECT content_id FROM ref_fingerprint WHERE h = ? LIMIT 1", (h,)).fetchone()
        if row:
            hits[i] = row[0]
    out: list[dict] = []
    for i in sorted(hits):
        if out and out[-1]["content_id"] == hits[i] and out[-1]["start"] + out[-1]["length"] - WINDOW + 1 == i:
            out[-1]["length"] += 1
        else:
            out.append({"content_id": hits[i], "start": i, "length": WINDOW})
    return out
```

```python
"""热度时间序列(spec 1.2 ref_heat、1.1 第 6 条):同一作品同一榜单一天一行,写入后不改;
曲线按天补齐,缺的天标 gap(控制台关着的日子);「涨得快」由代码按 config 阈值判。"""
from __future__ import annotations

import json
from datetime import date, timedelta

from tools.refs import clock, config


def record(store, wid: int, board: str, day: str, rank: int | None, value: float | None, raw: dict) -> bool:
    with store.write() as c:
        cur = c.execute("INSERT OR IGNORE INTO ref_heat (work_id, board, day, at, rank, heat, raw_json) VALUES (?, ?, ?, ?, ?, ?, ?)",
                        (wid, board, day, clock.now_iso(), rank, value, json.dumps(raw, ensure_ascii=False)))
        return cur.rowcount == 1


def series(store, wid: int, board: str) -> list[dict]:
    rows = {r["day"]: dict(r) for r in store.conn.execute(
        "SELECT day, rank, heat FROM ref_heat WHERE work_id = ? AND board = ? ORDER BY day", (wid, board))}
    if not rows:
        return []
    d, end, out = date.fromisoformat(min(rows)), date.fromisoformat(max(rows)), []
    while d <= end:
        k = d.isoformat()
        out.append({**rows[k], "gap": False} if k in rows else {"day": k, "rank": None, "heat": None, "gap": True})
        d += timedelta(days=1)
    return out


def boards_of(store, wid: int) -> list[str]:
    return [r[0] for r in store.conn.execute("SELECT DISTINCT board FROM ref_heat WHERE work_id = ? ORDER BY board", (wid,))]


def rising(store, wid: int, board: str, day: str) -> str | None:
    t = store.conn.execute("SELECT rank, heat FROM ref_heat WHERE work_id = ? AND board = ? AND day = ?", (wid, board, day)).fetchone()
    if t is None:
        return None
    y_day = (date.fromisoformat(day) - timedelta(days=1)).isoformat()
    y = store.conn.execute("SELECT rank, heat FROM ref_heat WHERE work_id = ? AND board = ? AND day = ?", (wid, board, y_day)).fetchone()
    if y is None:
        earlier = store.conn.execute("SELECT 1 FROM ref_heat WHERE work_id = ? AND board = ? AND day < ?", (wid, board, day)).fetchone()
        if earlier is None and t["rank"] is not None and t["rank"] <= config.NEW_TOP:
            return f"新上榜第 {t['rank']} 名"
        return None
    if y["heat"] and t["heat"] is not None and (t["heat"] - y["heat"]) / y["heat"] >= config.GROWTH:
        return f"热度涨 {(t['heat'] - y['heat']) / y['heat']:.0%}"
    if y["rank"] is not None and t["rank"] is not None and y["rank"] - t["rank"] >= config.RANK_JUMP:
        return f"名次升 {y['rank'] - t['rank']} 名"
    return None
```

```python
# -*- coding: utf-8 -*-
"""本机抓取队列(spec 1.1 第 2 条):单独的后台线程一次跑一条,不进 GPU worker。参数写下不可改(触发器)。"""
from __future__ import annotations

import json

from tools.refs import clock, settings


def enqueue(store, source: str, kind: str, *, day: str, work_id: int | None = None, params: dict | None = None,
            auto: bool = False) -> int:
    pj = json.dumps(params or {}, ensure_ascii=False, sort_keys=True)
    with store.write() as c:
        row = c.execute("SELECT id FROM ref_job WHERE source = ? AND kind = ? AND work_id IS ? AND params_json = ? AND day = ?"
                        " AND state != 'failed'", (source, kind, work_id, pj, day)).fetchone()
        if row:
            return row[0]
        return c.execute("INSERT INTO ref_job (source, kind, work_id, params_json, auto, day, created_at) VALUES (?, ?, ?, ?, ?, ?, ?)",
                         (source, kind, work_id, pj, int(auto), day, clock.now_iso())).lastrowid


def get(store, jid: int) -> dict:
    return dict(store.conn.execute("SELECT * FROM ref_job WHERE id = ?", (jid,)).fetchone())


def params(job: dict) -> dict:
    return json.loads(job["params_json"])


def next_waiting(store, today: str) -> dict | None:
    for row in store.conn.execute("SELECT * FROM ref_job WHERE state = 'waiting' ORDER BY id"):
        if settings.blocked_reason(settings.get(store, row["source"]), today) is None:
            return dict(row)
    return None


def start(store, jid: int) -> None:
    with store.write() as c:
        c.execute("UPDATE ref_job SET state = 'running', started_at = ? WHERE id = ?", (clock.now_iso(), jid))


def finish(store, jid: int, *, error: str = "") -> None:
    with store.write() as c:
        c.execute("UPDATE ref_job SET state = ?, error = ?, finished_at = ? WHERE id = ?",
                  ("failed" if error else "done", error, clock.now_iso(), jid))


def retry(store, jid: int, error: str) -> None:
    with store.write() as c:
        c.execute("UPDATE ref_job SET state = 'waiting', attempts = attempts + 1, error = ? WHERE id = ?", (error, jid))


def fail_stale(store) -> int:
    """控制台重启:上次停在 running 的标失败(不重跑,人可以手动再点)。"""
    with store.write() as c:
        return c.execute("UPDATE ref_job SET state = 'failed', error = '控制台重启时没跑完', finished_at = ? WHERE state = 'running'",
                         (clock.now_iso(),)).rowcount


def auto_content_today(store, source: str, day: str) -> int:
    return store.conn.execute("SELECT COUNT(*) FROM ref_job WHERE source = ? AND kind = 'content' AND auto = 1 AND day = ?",
                              (source, day)).fetchone()[0]


def recent(store, *, source: str | None = None, work_id: int | None = None, limit: int = 30) -> list[dict]:
    q, a = "SELECT * FROM ref_job WHERE 1 = 1", []
    if source:
        q += " AND source = ?"; a.append(source)
    if work_id is not None:
        q += " AND work_id = ?"; a.append(work_id)
    return [dict(r) for r in store.conn.execute(q + " ORDER BY id DESC LIMIT ?", (*a, limit))]
```

```python
# -*- coding: utf-8 -*-
"""跑一条参考库任务(spec 1.1、4.1)。一次只跑一条;出错不吞,按种类落成熔断 / 暂停 / 重试 / 失败,原因写进任务与 journal/refs。"""
from __future__ import annotations

import hashlib
import json
from collections.abc import Callable
from pathlib import Path

from tools.refs import clock, config, fingerprint, heat, jobs, judge, rjournal, settings, works
from tools.refs.net import FetchError, Fetcher
from tools.refs.sources import ADAPTERS
from tools.refs.sources.base import AdapterError


def _store_file(root: Path, rel: str, text: str) -> str:
    p = Path(root) / rel
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(text, encoding="utf-8")
    return hashlib.sha256(text.encode("utf-8")).hexdigest()


def _content_row(store, root, src, wid, kind, unit, title, rel, text, origin, verdict, reason) -> int:
    sha = _store_file(root, rel, text)
    at = clock.now_iso()
    days = src["retention_days"]
    with store.write() as c:
        cid = c.execute(
            "INSERT INTO ref_content (work_id, kind, unit, title, path, sha256, origin, verdict, reason, fetched_at, due_at, hard_due_at)"
            " VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)",
            (wid, kind, unit, title, rel, sha, origin, verdict, reason, at, clock.add_days(at, days), clock.add_days(at, 2 * days))).lastrowid
    if verdict == "ok":
        fingerprint.add(store, cid, text)
    return cid


def _ranks(store, root, adapter, f, job, today):
    board = jobs.params(job)["board"]
    items = adapter.list_ranks(f, board)
    for it in items:
        wid = works.upsert(store, adapter.key, it.external_id, title=it.title)
        heat.record(store, wid, board, today, it.rank, it.heat, it.raw)
    # 这个来源今天**所有榜单**都成功了才算「今天拉过榜单」(只成功一个不算,别的榜单下一轮照排)
    done = {jobs.params(j)["board"] for j in store.conn.execute(
        "SELECT * FROM ref_job WHERE source = ? AND kind = 'ranks' AND day = ? AND state = 'done'", (adapter.key, today))}
    if set(adapter.boards) <= done | {board}:
        with store.write() as c:
            c.execute("UPDATE ref_source SET last_ranks_day = ? WHERE key = ?", (today, adapter.key))
    return f"榜单 {board}:{len(items)} 条"


def _meta(store, root, adapter, f, job, today):
    w = works.get(store, job["work_id"])
    m = adapter.fetch_meta(f, w["external_id"])
    works.apply_meta(store, w["id"], m)
    heat.record(store, w["id"], "detail", today, None, m.heat, m.stats)
    return f"详情:{m.title}"


def _content(store, root, adapter, f, job, today):
    src = settings.get(store, adapter.key)
    p = settings.fetch_problems(src)
    if p:
        raise ValueError(";".join(p))
    w = works.get(store, job["work_id"])
    if w["blocked"]:
        raise ValueError("这部作品已拉黑,不抓")
    base = f"content/{adapter.key}/{w['external_id']}"
    if adapter.content == "reviews":
        reviews = adapter.fetch_reviews(f, w["external_id"])
        text = "\n".join(json.dumps({"text": r.text, "rating": r.rating, "likes": r.likes, "at": r.at}, ensure_ascii=False)
                         for r in reviews)
        _content_row(store, root, src, w["id"], "review", None, f"剧评 {len(reviews)} 条", f"{base}/reviews-{today}.jsonl",
                     text, adapter.key, "ok" if reviews else "rejected", "" if reviews else "没有剧评")
        return f"剧评 {len(reviews)} 条"
    if adapter.content != "chapters":
        raise ValueError("这个来源只有元数据,不抓正文")
    m = adapter.fetch_meta(f, w["external_id"])
    works.apply_meta(store, w["id"], m)
    free = [u for u in m.units if u.free][: src["chapters_n"]]
    ok = 0
    for ref in free:
        u = adapter.fetch_unit(f, m, ref)
        verdict, reason = judge.chapter(u)
        _content_row(store, root, src, w["id"], "chapter", u.number, u.title, f"{base}/{u.number}.txt", u.text,
                     ref.url, verdict, reason)
        ok += verdict == "ok"
    return f"免费章 {len(free)} 个,收 {ok} 个"


def _sitemap(store, root, adapter, f, job, today):
    ids = adapter.sitemap(f)
    for ext in ids:
        works.upsert(store, adapter.key, ext)
    with store.write() as c:
        c.execute("UPDATE ref_source SET last_sitemap_at = ? WHERE key = ?", (clock.now_iso(), adapter.key))
    return f"sitemap {len(ids)} 条"


KINDS: dict[str, Callable] = {"ranks": _ranks, "meta": _meta, "content": _content, "sitemap": _sitemap}


def _streak(store, key: str, value: int | None) -> int:
    with store.write() as c:
        if value is None:
            c.execute("UPDATE ref_source SET fail_streak = fail_streak + 1 WHERE key = ?", (key,))
        else:
            c.execute("UPDATE ref_source SET fail_streak = ? WHERE key = ?", (value, key))
        return c.execute("SELECT fail_streak FROM ref_source WHERE key = ?", (key,)).fetchone()[0]


def run_one(store, root: Path, *, adapters: dict | None = None, fetcher: Callable[[str], object] | None = None,
            today: str | None = None) -> dict | None:
    """跑下一条等待中的任务;没有可跑的返回 None。"""
    adapters = adapters or ADAPTERS
    fetcher = fetcher or (lambda src: Fetcher(root, src))
    today = today or clock.today()
    job = jobs.next_waiting(store, today)
    if job is None:
        return None
    key = job["source"]
    jobs.start(store, job["id"])
    try:
        msg = KINDS[job["kind"]](store, root, adapters[key], fetcher(key), job, today)
    except FetchError as e:
        if e.kind == "unreachable" and job["attempts"] < config.FETCH_RETRIES:
            jobs.retry(store, job["id"], str(e))
        else:
            jobs.finish(store, job["id"], error=str(e))
            if e.kind == "disallowed":
                settings.pause(store, root, key, str(e))
            elif e.kind == "blocked":
                settings.trip(store, root, key, str(e), today)
        rjournal.append(root, "job", source=key, work_id=job["work_id"], job=job["id"], kind=job["kind"], error=str(e))
        return jobs.get(store, job["id"])
    except AdapterError as e:
        jobs.finish(store, job["id"], error=f"解析失败:{e}")
        if _streak(store, key, None) >= config.TRIP_AFTER:
            settings.trip(store, root, key, f"连续 {config.TRIP_AFTER} 次解析失败,多半网页改版 —— 最后一次:{e}", today)
        rjournal.append(root, "job", source=key, work_id=job["work_id"], job=job["id"], kind=job["kind"], error=str(e))
        return jobs.get(store, job["id"])
    except (ValueError, KeyError) as e:
        jobs.finish(store, job["id"], error=str(e))
        rjournal.append(root, "job", source=key, work_id=job["work_id"], job=job["id"], kind=job["kind"], error=str(e))
        return jobs.get(store, job["id"])
    _streak(store, key, 0)
    jobs.finish(store, job["id"])
    rjournal.append(root, "job", source=key, work_id=job["work_id"], job=job["id"], kind=job["kind"], done=msg)
    return jobs.get(store, job["id"])
```

```python
# -*- coding: utf-8 -*-
"""调度(spec 1.1 第 6 条、3.2):按来源的模式排任务。每轮跑一次,幂等(队列同日去重)。

- C:什么都不自动做;
- A / B:每天拉一次各榜单 + 已关注作品的详情;红果 sitemap 每周一次;控制台关过的日子不补,只拉今天(曲线上标缺口);
- 今天榜单齐了以后:A 按「前 K 名或涨得快」自动排抓正文,到每日上限就停(不补跑;上限改了马上生效);
  B 只标候选与理由,人勾选。人手动点的不算上限。
"""
from __future__ import annotations

from datetime import datetime, timedelta, timezone

from tools.refs import clock, config, heat, jobs, settings, works
from tools.refs.sources import ADAPTERS


def _today_ranked(store, key: str, today: str) -> list[dict]:
    return [dict(r) for r in store.conn.execute(
        "SELECT w.id, w.blocked, MIN(h.rank) AS best, h.board FROM ref_heat h JOIN ref_work w ON w.id = h.work_id"
        " WHERE w.source = ? AND h.day = ? AND h.board != 'detail' GROUP BY w.id ORDER BY best", (key, today))]


def _reason(store, row: dict, top_k: int | None, today: str) -> str:
    if top_k is not None and row["best"] is not None and row["best"] <= top_k:
        return f"今天第 {row['best']} 名(前 {top_k} 名)"
    return heat.rising(store, row["id"], row["board"], today) or ""


def _has_content(store, wid: int) -> bool:
    return store.conn.execute("SELECT 1 FROM ref_content WHERE work_id = ? LIMIT 1", (wid,)).fetchone() is not None or \
        store.conn.execute("SELECT 1 FROM ref_job WHERE work_id = ? AND kind = 'content' AND state != 'failed' LIMIT 1",
                           (wid,)).fetchone() is not None


def plan(store, today: str | None = None, now: str | None = None) -> list[str]:
    today = today or clock.today()
    now = now or clock.now_iso()
    did: list[str] = []
    for src in settings.all_sources(store):
        key = src["key"]
        if key == "manual" or src["mode"] == "C" or settings.blocked_reason(src, today):
            continue
        a = ADAPTERS[key]
        if src["last_ranks_day"] != today:
            for b in a.boards:
                jobs.enqueue(store, key, "ranks", day=today, params={"board": b})
            did.append(f"{key}:排今天的榜单")
        for (wid,) in store.conn.execute("SELECT id FROM ref_work WHERE source = ? AND followed = 1 AND blocked = 0", (key,)).fetchall():
            if a.content != "none":
                jobs.enqueue(store, key, "meta", work_id=wid, day=today)
        if a.sitemap and (not src["last_sitemap_at"] or
                          datetime.fromisoformat(now) - datetime.fromisoformat(src["last_sitemap_at"]) >= timedelta(days=config.SITEMAP_DAYS)):
            jobs.enqueue(store, key, "sitemap", day=today)
        if src["last_ranks_day"] != today or a.content == "none":      # 今天所有榜单都成功了才选作品(见 runner._ranks)
            continue
        for row in _today_ranked(store, key, today):
            if row["blocked"]:
                continue
            why = _reason(store, row, src["top_k"], today)
            if not why:
                continue
            if src["mode"] == "B":
                works.set_candidate(store, row["id"], why, today)
                continue
            if jobs.auto_content_today(store, key, today) >= src["cap_fetch"]:
                did.append(f"{key}:今日自动抓取已满 {src['cap_fetch']}/{src['cap_fetch']}")
                break
            if not _has_content(store, row["id"]):
                jobs.enqueue(store, key, "content", work_id=row["id"], day=today, auto=True)
                did.append(f"{key}:自动抓 {row['id']}({why})")
    return did
```

```python
# -*- coding: utf-8 -*-
"""到期清理(spec 1.3):删原文文件、置 purged_at;指纹与内容行留着。
顺延:还没拆解的作品顺延到 hard_due_at(保留天数 × 2);拉黑的作品立即删。1a 没有拆解 → has_breakdown 恒假。"""
from __future__ import annotations

from pathlib import Path

from tools.refs import clock, rjournal


def has_breakdown(store, work_id: int) -> bool:
    return False          # 1c:查 ref_card 里这部作品有没有已批准 / 待审的卡


def purge(store, root: Path, *, now: str | None = None) -> tuple[list[int], list[str]]:
    now = now or clock.now_iso()
    done, errors = [], []
    rows = store.conn.execute(
        "SELECT c.id, c.path, c.work_id, c.due_at, c.hard_due_at, w.blocked FROM ref_content c JOIN ref_work w ON w.id = c.work_id"
        " WHERE c.purged_at = ''").fetchall()
    for r in rows:
        due = r["due_at"] if has_breakdown(store, r["work_id"]) else r["hard_due_at"]
        if not (r["blocked"] or due <= now):
            continue
        p = Path(root) / r["path"]
        try:
            p.unlink()
        except FileNotFoundError:
            errors.append(f"内容 {r['id']} 的文件找不到:{r['path']}(没有标已清理,请人看一下)")
            continue
        except OSError as e:
            errors.append(f"内容 {r['id']} 删除失败:{e}")
            continue
        with store.write() as c:
            c.execute("UPDATE ref_content SET purged_at = ? WHERE id = ?", (now, r["id"]))
        done.append(r["id"])
    if done or errors:
        rjournal.append(root, "purge", purged=done, errors=errors)
    return done, errors
```

```python
from __future__ import annotations

import hashlib
from pathlib import Path

from tools.refs import clock, fingerprint, judge, settings, works
from tools.refs.sources.base import Unit


def add_text(store, root: Path, *, title: str, text: str, origin: str, work_id: int | None = None) -> int:
    if not origin.strip():
        raise ValueError("来源说明没填 —— 写清楚这段文字是怎么来的(例如「自己看完第 1–5 集整理」)")
    src = settings.get(store, "manual")
    p = settings.fetch_problems(src)
    if p:
        raise ValueError(";".join(p))
    if work_id is None:
        n = store.conn.execute("SELECT COUNT(*) FROM ref_work WHERE source = 'manual'").fetchone()[0] + 1
        work_id = works.upsert(store, "manual", f"manual-{n}", title=title)
    verdict, reason = judge.chapter(Unit(0, title, text, None))
    at = clock.now_iso()
    rel = f"content/manual/{work_id}/{hashlib.sha256(text.encode('utf-8')).hexdigest()[:16]}.txt"
    (Path(root) / rel).parent.mkdir(parents=True, exist_ok=True)
    (Path(root) / rel).write_text(text, encoding="utf-8")
    days = src["retention_days"]
    with store.write() as c:
        cid = c.execute("INSERT INTO ref_content (work_id, kind, title, path, sha256, origin, verdict, reason, fetched_at, due_at, hard_due_at)"
                        " VALUES (?, 'manual', ?, ?, ?, ?, ?, ?, ?, ?, ?)",
                        (work_id, title, rel, hashlib.sha256(text.encode("utf-8")).hexdigest(), origin.strip(), verdict, reason,
                         at, clock.add_days(at, days), clock.add_days(at, 2 * days))).lastrowid
    if verdict == "ok":
        fingerprint.add(store, cid, text)
    return cid
```

```python
def _curve(points: list[dict], w: int = 320, h: int = 60) -> list[list[tuple[float, float]]]:
    """热度曲线的折线段(缺口处断开),坐标由代码算好给模板画 SVG。"""
    vals = [p["heat"] for p in points if p["heat"] is not None]
    if len(points) < 2 or not vals:
        return []
    lo, hi = min(vals), max(vals)
    segs, cur = [], []
    for i, p in enumerate(points):
        if p["heat"] is None:
            if cur:
                segs.append(cur)
            cur = []
            continue
        x = i * w / (len(points) - 1)
        y = h - (p["heat"] - lo) / ((hi - lo) or 1) * h
        cur.append((round(x, 1), round(y, 1)))
    if cur:
        segs.append(cur)
    return segs
```

```python
# -*- coding: utf-8 -*-
"""验收的来源冒烟(spec 4.3):每个来源真联网拉一个榜单,过解析校验。不写参考库。

    python -X utf8 -m tools.refs.smoke --out <目录>     # 打印并写 <目录>/smoke.json;退出码 0 = PASS,1 = FAIL,2 = SKIP
"""
from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path

from tools.refs.net import FetchError, Fetcher
from tools.refs.sources import ADAPTERS
from tools.refs.sources.base import AdapterError

BOARD = {"hongguo": "hot-real-drama"}


def check_one(list_ranks, board: str, fetcher) -> dict:
    try:
        items = list_ranks(fetcher, board)
        return {"state": "PASS", "items": len(items)}
    except AdapterError as e:
        return {"state": "FAIL", "reason": f"网站返回了页面,但解析失败(适配器坏了):{e}"}
    except FetchError as e:
        return {"state": "UNAVAILABLE", "reason": f"外部不可用({e.kind}):{e}"}


def overall(states: dict[str, str]) -> str:
    v = set(states.values())
    if "FAIL" in v:
        return "FAIL"
    if v <= {"UNAVAILABLE"}:
        return "SKIP"
    return "PASS"


def main(argv: list[str] | None = None) -> int:
    ap = argparse.ArgumentParser(description="参考库来源冒烟")
    ap.add_argument("--out", required=True)
    a = ap.parse_args(argv)
    out = Path(a.out)
    out.mkdir(parents=True, exist_ok=True)
    res = {}
    for key, ad in ADAPTERS.items():
        board = BOARD.get(key, next(iter(ad.boards)))
        res[key] = {"board": board, **check_one(ad.list_ranks, board, Fetcher(out, key))}
        print(f"  [{res[key]['state']}] {key} {board} {res[key].get('reason', res[key].get('items', ''))}", flush=True)
    verdict = overall({k: v["state"] for k, v in res.items()})
    (out / "smoke.json").write_text(json.dumps({"verdict": verdict, "sources": res}, ensure_ascii=False, indent=2), encoding="utf-8")
    return {"PASS": 0, "FAIL": 1, "SKIP": 2}[verdict]


if __name__ == "__main__":
    sys.exit(main())
```
