> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 补上设计「第一版范围」里的两项:**增量补全**(一集 / 一部剧只跑缺的、失败的,先报清单与耗时再提交)与**工程导出 / 导入**(一部剧连同全部生成历史、帧绑定、LLM 审批记录打包,能在另一处导入成新剧)。

`tools/gen/fill.py`:

```python
# -*- coding: utf-8 -*-
"""增量补全:一集(或一部剧)里**只跑缺的 / 失败的**(设计 §七「贯穿所有面板」、调研结论「禁止全量重跑」)。

每个目标归入四类之一,界面上逐条列出:
- `has`:已经有当前产物 —— 跳过(换版本走「用这一版」,不在这里);
- `in_flight`:已有在途任务 —— 跳过(不重复交,重交就是白烧一次 GPU);
- `blocked`:内容不就绪 —— 跳过,写明原因(与单条生成时同一套检查);
- `todo`:要做 —— 包括上次失败、至今没有产物的。

两态:`plan_fill` 只算不改,给出清单、每条的种子与估算总耗时;人确认后 `confirm_fill` 才建任务。
确认时**只认确认页上列出的那几条**,并逐条重新判一次 —— 这期间状态变了(比如别人刚点过单条生成)的就跳过,不重复交。
"""
from __future__ import annotations

from dataclasses import dataclass

from tools.drama import repo
from tools.drama.db import Store
from tools.gen.assets import SPEC, asset_problems, plan_asset_image
from tools.gen.capability import MINIMAX_H3, QWEN_IMAGE_EDIT_2511
from tools.gen.planner import first_frame_problems, plan_first_frame, plan_video, video_problems

IN_FLIGHT = ("pending", "queued", "running", "waiting_host")
KINDS = {"frames": ("episode", "首帧"), "videos": ("episode", "视频"), "assets": ("drama", "资产图")}
_ASSET_LABEL = {"character": "角色", "scene": "场景", "prop": "道具"}


@dataclass(frozen=True)
class Item:
    target_type: str
    target_id: int
    label: str
    status: str          # has / in_flight / blocked / todo
    reason: str = ""
    est_seconds: float = 0.0


@dataclass(frozen=True)
class FillPlan:
    kind: str
    scope_id: int
    items: tuple[Item, ...]

    @property
    def todo(self) -> list[Item]:
        return [i for i in self.items if i.status == "todo"]

    def est_seconds(self) -> float:
        return sum(i.est_seconds for i in self.todo)

    def summary(self) -> str:
        count = {s: sum(1 for i in self.items if i.status == s) for s in ("has", "in_flight", "blocked")}
        what = KINDS[self.kind][1]
        head = (f"将提交 {len(self.todo)} 条{what},约 {self.est_seconds() / 3600:.2f} 小时(按实测耗时估算)"
                if self.todo else f"没有要补的{what}")
        return (f"{head};跳过:已有 {count['has']}、在途 {count['in_flight']}、不能做 {count['blocked']}")


def _in_flight(store: Store, target_type: str, target_id: int, kind: str) -> bool:
    return store.conn.execute(
        f"SELECT 1 FROM task WHERE target_type = ? AND target_id = ? AND kind = ? AND state IN "
        f"({','.join('?' * len(IN_FLIGHT))}) LIMIT 1", (target_type, target_id, kind, *IN_FLIGHT)).fetchone() is not None


def _item(store: Store, target_type: str, row, label: str, kind: str, field: str, problems, est: float) -> Item:
    if row[field]:
        return Item(target_type, row["id"], label, "has")
    if _in_flight(store, target_type, row["id"], kind):
        return Item(target_type, row["id"], label, "in_flight")
    probs = problems()
    if probs:
        return Item(target_type, row["id"], label, "blocked", ";".join(probs))
    return Item(target_type, row["id"], label, "todo", est_seconds=est)


def plan_fill(store: Store, kind: str, scope_id: int) -> FillPlan:
    if kind not in KINDS:
        raise ValueError(f"未知的补全类型 {kind},可选:{sorted(KINDS)}")
    items = []
```

```python
# -*- coding: utf-8 -*-
"""工程导出 / 导入(设计 §九「工程导出/导入(含图片历史与帧绑定)」)。

一部剧打成一个 zip:`manifest.json`(全部行 + 每个文件的 sha256)+ `files/<相对路径>`。
- **带历史**:这部剧的全部 task(图片 / 视频的每一版)、段落与资产的绑定、LLM 记录(含审批锁定);
- **不带**:云机登记(私钥路径是本机的)、成片导出记录(那是发布产物,ProduceID 不能在两处重复);
- **导出前拒绝**:有在途任务 / 在途 LLM(导出后永远收不回来)、被引用的文件不在本机 —— 列出原因,不导出残缺包。

导入 = **新建一部剧**,不碰已有数据:
- 全部 id 重新分配,外键逐个映射;
- 文件放到 `imports/<导入编号>/` 下,库里所有路径统一加这个前缀 —— 与已有产物(如 outputs/tasks/2/…)不会撞名;
- 每个文件按 manifest 的 sha256 校验,zip 里的路径不许跳出目录;库结构版本不一致直接拒绝;
- 先落文件、再一个事务写库;写库失败就删掉刚落的文件。原来机器上的 host 不导入,task 的 host_id 置空。
"""
from __future__ import annotations

import hashlib
import json
import shutil
import uuid
import zipfile
from datetime import datetime, timezone
from pathlib import Path, PurePosixPath

from tools.drama.db import SCHEMA_VERSION, Store, now

FORMAT = 1
IN_FLIGHT = ("pending", "queued", "running", "waiting_host")
PATH_FIELDS = {"character": ("turnaround_path", "voice_ref_path"), "scene": ("image_path",), "prop": ("image_path",),
               "segment": ("first_frame_path", "last_frame_path", "video_path", "subtitle_path")}
PARAM_PATH_KEYS = ("image_refs", "audio_ref")      # task.params_json 里存路径的键


def _sha256(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def _rows(store: Store, sql: str, args: tuple) -> list[dict]:
    return [dict(r) for r in store.conn.execute(sql, args)]


def _in(ids: list[int]) -> str:
    return ",".join(str(int(i)) for i in ids) or "NULL"


def collect(store: Store, drama_id: int) -> dict:
    """这部剧要导出的全部行。"""
    t: dict[str, list[dict]] = {"drama": _rows(store, "SELECT * FROM drama WHERE id = ?", (drama_id,))}
    if not t["drama"]:
        raise KeyError(f"drama {drama_id} 不存在")
    for table in ("episode", "character", "scene", "prop"):
        t[table] = _rows(store, f"SELECT * FROM {table} WHERE drama_id = ? ORDER BY id", (drama_id,))
    eps = [r["id"] for r in t["episode"]]
    t["segment"] = _rows(store, f"SELECT * FROM segment WHERE episode_id IN ({_in(eps)}) ORDER BY id", ())
    segs = [r["id"] for r in t["segment"]]
    for link in ("segment_character", "segment_prop"):
        t[link] = _rows(store, f"SELECT * FROM {link} WHERE segment_id IN ({_in(segs)}) ORDER BY segment_id, position", ())
    owned = {"segment": segs, **{k: [r["id"] for r in t[k]] for k in ("character", "scene", "prop")}}
    t["task"] = _rows(store, " UNION ".join(
        f"SELECT * FROM task WHERE target_type = '{k}' AND target_id IN ({_in(v)})" for k, v in owned.items())
        + " ORDER BY id", ())
    llm_owned = {"drama": [drama_id], "episode": eps, "character": owned["character"]}
    t["llm_run"] = _rows(store, " UNION ".join(
        f"SELECT * FROM llm_run WHERE target_type = '{k}' AND target_id IN ({_in(v)})" for k, v in llm_owned.items())
        + " ORDER BY id", ())
    return t


def _paths(tables: dict) -> set[str]:
    out = set()
    for table, fields in PATH_FIELDS.items():
        for r in tables[table]:
            out |= {r[f] for f in fields if r[f]}
    for r in tables["task"]:
        out |= set(json.loads(r["refs_json"]))
        if r["result_path"]:
            out.add(r["result_path"])
    return out


def export_bundle(store: Store, data_root: Path, drama_id: int) -> str:
    """打包到 `<data_root>/outputs/bundles/`,返回相对 data_root 的路径。"""
    root = Path(data_root).resolve()
    tables = collect(store, drama_id)
    problems = [f"task #{r['id']} 还在「{r['state']}」—— 等它结束再导出(导出后收不回来)"
                for r in tables["task"] if r["state"] in IN_FLIGHT]
    problems += [f"LLM 任务 #{r['id']} 还在生成 —— 等它结束再导出" for r in tables["llm_run"] if r["state"] == "running"]
    files = {p: root / p for p in sorted(_paths(tables))}
    journals = {}
    for r in tables["llm_run"]:
        if r["journal_path"]:
            j = Path(r["journal_path"]).resolve()
            if root not in j.parents or not j.is_file():
                problems.append(f"LLM 任务 #{r['id']} 的记录文件不在素材根下:{r['journal_path']}")
            else:
                rel = j.relative_to(root).as_posix()
                files[rel] = j
                journals[r["id"]] = rel
    problems += [f"文件不在本机:{p}" for p, f in files.items() if not f.is_file()]
    if problems:
        raise ValueError(";".join(problems))
    manifest = {"format": FORMAT, "schema_version": SCHEMA_VERSION, "exported_at": now(), "tables": tables,
                "llm_journals": journals, "files": {p: _sha256(f) for p, f in files.items()}}
    stamp = datetime.now(timezone.utc).strftime("%Y%m%d-%H%M%S")
    rel = f"outputs/bundles/drama-{drama_id}-{stamp}.zip"
    (root / rel).parent.mkdir(parents=True, exist_ok=True)
    with zipfile.ZipFile(root / rel, "w", zipfile.ZIP_STORED) as z:     # 图片视频本就压缩过
        z.writestr("manifest.json", json.dumps(manifest, ensure_ascii=False, indent=1))
        for p, f in files.items():
            z.write(f, f"files/{p}")
    return rel


def _safe(rel: str) -> bool:
    p = PurePosixPath(rel)
    return bool(rel) and not p.is_absolute() and ".." not in p.parts and "\\" not in rel and ":" not in rel


def import_bundle(store: Store, data_root: Path, bundle: Path) -> int:
    """导入成一部新剧,返回新 drama id。"""
    root = Path(data_root).resolve()
    with zipfile.ZipFile(bundle) as z:
        try:
            m = json.loads(z.read("manifest.json"))
        except KeyError as e:
            raise ValueError("不是工程包:缺少 manifest.json") from e
        if m.get("format") != FORMAT:
            raise ValueError(f"工程包格式 {m.get('format')} 不认识(本版只认 {FORMAT})")
        if m.get("schema_version") != SCHEMA_VERSION:
            raise ValueError(f"工程包来自库结构 v{m.get('schema_version')},本机是 v{SCHEMA_VERSION} —— 不许跨版本导入")
        bad = [p for p in m["files"] if not _safe(p)]
        if bad:
            raise ValueError(f"工程包里有不安全的路径:{bad[:5]}")
        missing = [p for p in _paths(m["tables"]) | set(m["llm_journals"].values()) if p not in m["files"]]
        if missing:
            raise ValueError(f"工程包不完整,缺文件:{sorted(missing)[:5]}")
        prefix = f"imports/{datetime.now(timezone.utc).strftime('%Y%m%d%H%M%S')}-{uuid.uuid4().hex[:8]}"
        dest = root / prefix
        try:
            for p, sha in m["files"].items():
                out = dest / p
                out.parent.mkdir(parents=True, exist_ok=True)
                with z.open(f"files/{p}") as src, out.open("wb") as dst:
                    shutil.copyfileobj(src, dst)
                if _sha256(out) != sha:
                    raise ValueError(f"文件 {p} 的 sha256 与清单不符 —— 工程包损坏或被改过")
            return _insert(store, m, prefix, root)
        except BaseException:
            shutil.rmtree(dest, ignore_errors=True)
            raise


def _insert(store: Store, m: dict, prefix: str, root: Path) -> int:
    t = m["tables"]

    def mv(p):
        return f"{prefix}/{p}" if p else p

    def insert(c, table: str, row: dict, **over) -> int:
        row = {k: v for k, v in row.items() if k != "id"} | over
        return c.execute(f"INSERT INTO {table} ({', '.join(row)}) VALUES ({', '.join('?' * len(row))})",
                         tuple(row.values())).lastrowid

    ids: dict[str, dict[int, int]] = {k: {} for k in ("drama", "episode", "character", "scene", "prop", "segment")}
    with store.write() as c:
        [d] = t["drama"]
        ids["drama"][d["id"]] = new_drama = insert(c, "drama", d)
        for r in t["episode"]:
            ids["episode"][r["id"]] = insert(c, "episode", r, drama_id=new_drama)
        for table in ("character", "scene", "prop"):
            for r in t[table]:
                over = {"drama_id": new_drama, **{f: mv(r[f]) for f in PATH_FIELDS[table]}}
                if table == "scene" and r["episode_id"] is not None:
                    over["episode_id"] = ids["episode"][r["episode_id"]]
                ids[table][r["id"]] = insert(c, table, r, **over)
        for r in t["segment"]:
            over = {"episode_id": ids["episode"][r["episode_id"]], **{f: mv(r[f]) for f in PATH_FIELDS["segment"]}}
            if r["scene_id"] is not None:
                over["scene_id"] = ids["scene"][r["scene_id"]]
            ids["segment"][r["id"]] = insert(c, "segment", r, **over)
        for link, col, table in (("segment_character", "character_id", "character"), ("segment_prop", "prop_id", "prop")):
            for r in t[link]:
                c.execute(f"INSERT INTO {link} (segment_id, {col}, position) VALUES (?, ?, ?)",
                          (ids["segment"][r["segment_id"]], ids[table][r[col]], r["position"]))
        for r in t["task"]:
            params = json.loads(r["params_json"])
            for k in PARAM_PATH_KEYS:
                if isinstance(params.get(k), list):
                    params[k] = [mv(p) for p in params[k]]
                elif params.get(k):
                    params[k] = mv(params[k])
            insert(c, "task", r, target_id=ids[r["target_type"]][r["target_id"]], host_id=None,
                   result_path=mv(r["result_path"]), refs_json=json.dumps([mv(p) for p in json.loads(r["refs_json"])],
                                                                          ensure_ascii=False),
                   params_json=json.dumps(params, ensure_ascii=False, sort_keys=True))
        for r in t["llm_run"]:
            j = m["llm_journals"].get(str(r["id"]))
            insert(c, "llm_run", r, target_id=ids[r["target_type"]][r["target_id"]],
                   journal_path=str(root / prefix / j) if j else None)
    return new_drama
```

```python
# -*- coding: utf-8 -*-
"""增量补全:参考帧 / 视频 / 资产面板上的「补全缺的」按钮都 POST 到这里。

先出确认页(每条的状态、原因、种子、估算总耗时),人点确认才建任务 —— 单条视频约 15 分钟,不许一键就交。
"""
from __future__ import annotations

from fastapi import APIRouter, Request
from fastapi.responses import RedirectResponse

from tools.drama import repo
from tools.gen.assets import ensure_canvas
from tools.gen.fill import KINDS, confirm_fill, plan_fill
from tools.serve.panels.common import render, rt

router = APIRouter()
STATUS = {"has": "已有", "in_flight": "在途", "blocked": "不能做", "todo": "要做"}


def _back(store, kind: str, scope_id: int) -> tuple[str, dict]:
    if kind == "assets":
        return f"/drama/{scope_id}/assets", {"panel_key": "assets", "drama_id": scope_id}
    repo.get(store, "episode", scope_id)
    return f"/episode/{scope_id}/{kind}", {"panel_key": kind, "episode_id": scope_id}


@router.post("/fill/{kind}/{scope_id}")
async def fill(request: Request, kind: str, scope_id: int):
    if kind not in KINDS:
        raise ValueError(f"未知的补全类型 {kind}")
    runtime = rt(request)
    form = await request.form()
    raw = str(form.get("base_seed", "")).strip()
    if not raw.lstrip("-").isdigit():
        raise ValueError(f"起始种子要填整数,填的是 {raw!r}")
    base_seed = int(raw)
    back, ctx = _back(runtime.store, kind, scope_id)
    if form.get("confirm") != "1":
        plan = plan_fill(runtime.store, kind, scope_id)
        seeds = {(i.target_type, i.target_id): base_seed + n for n, i in enumerate(plan.todo)}
        return render(request, "fill_confirm.html", ctx.pop("panel_key"), plan=plan, seeds=seeds, status=STATUS,
                      base_seed=base_seed, action=f"/fill/{kind}/{scope_id}", back=back, **ctx)
    targets = []
    for t in form.getlist("t"):
        ttype, _, tid = str(t).partition(":")
        targets.append((ttype, int(tid)))
    if kind == "assets":
        ensure_canvas(runtime.data_root)
    confirm_fill(runtime.store, kind, scope_id, targets, base_seed)
    return RedirectResponse(back, status_code=303)
```

```python
# -*- coding: utf-8 -*-
"""工程导出 / 导入(立项面板上的入口)。打包与校验逻辑在 `tools.drama.bundle`。"""
from __future__ import annotations

import shutil
import tempfile
from pathlib import Path

from fastapi import APIRouter, File, Request, UploadFile
from fastapi.responses import RedirectResponse

from tools.drama import bundle
from tools.serve import journal
from tools.serve.panels.common import rt

router = APIRouter()


def listing(data_root: Path) -> list[str]:
    """已导出的工程包(相对 data_root),新的在前。"""
    d = Path(data_root) / "outputs" / "bundles"
    return [p.relative_to(data_root).as_posix() for p in sorted(d.glob("*.zip"), reverse=True)] if d.is_dir() else []


@router.post("/drama/{drama_id}/bundle")
def export(request: Request, drama_id: int):
    runtime = rt(request)
    rel = bundle.export_bundle(runtime.store, runtime.data_root, drama_id)
    journal.append(runtime.data_root, [f"工程导出 drama {drama_id} → {rel}"], prefix="console ")
    return RedirectResponse("/#bundles", status_code=303)


@router.post("/bundles/import")
def import_(request: Request, file: UploadFile = File(...)):
    runtime = rt(request)
    with tempfile.TemporaryDirectory() as d:
        tmp = Path(d) / "bundle.zip"
        with tmp.open("wb") as out:
            shutil.copyfileobj(file.file, out)
        new = bundle.import_bundle(runtime.store, runtime.data_root, tmp)
    journal.append(runtime.data_root, [f"工程导入 {file.filename} → drama {new}"], prefix="console ")
    return RedirectResponse(f"/drama/{new}/assets", status_code=303)
```
