# AI 短剧控制台 · P13 台词外多说提示 · 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 视频里人物在台词之外多说了话(如把舞台指示「林用普通话说」念出来)时,在视频面板提示换种子重出。

**Architecture:** 不新跑识别 —— 字幕对齐本来就存下了 whisper 逐字结果与能量曲线(`align.json`)。纯函数复用对齐的匹配找出台词覆盖之外的识别字,
对齐不可信不判、不足 3 字不报、能量不到开口门槛不报(防 whisper 在静音上编字);面板只认当前视频的对齐,只提示不拦。

**Tech Stack:** Python 3.14 标准库 · 现有 FastAPI / Jinja2 / Playwright。**不加依赖、不改库结构、不开卡。**

```python
LOUD = [100.0] * 80                  # 8 秒全程有声(0.1 秒一窗)


def _chars(text, start=0.0, step=0.2):
    return [[c, round(start + i * step, 2), round(start + (i + 1) * step, 2)] for i, c in enumerate(text)]


def _run(r):
    return subtitles.extra_speech(r["lines"], [tuple(c) for c in r["chars"]], r["envelope"])


def test_real_clip_that_read_the_stage_direction():
    """2026-09-29 那条:whisper 听到「林用普通话说这块玉佩……」,能量在 3.7–4.7 秒有一段真的说话。"""
    runs, problem = _run(REAL["2026-09-29"])
    assert problem == "" and [(r["text"], r["start"], r["end"]) for r in runs] == [("林用普通话说", 0.0, 4.76)]


@pytest.mark.parametrize("day", ["2026-10-09", "2026-10-10"])
def test_real_clips_that_only_said_the_line(day):
    assert _run(REAL[day]) == ([], "")


def test_after_and_between_lines():
    lines = ["这块玉佩", "我会带着它"]
    runs, _ = subtitles.extra_speech(lines, _chars("这块玉佩沈墨说话了我会带着它谢谢大家收看"), LOUD)
    assert [r["text"] for r in runs] == ["沈墨说话了", "谢谢大家收看"]


def test_short_runs_and_punctuation_are_ignored():
    runs, _ = subtitles.extra_speech(["这块玉佩"], _chars("嗯，这块玉佩，啊。"), LOUD)
    assert runs == []                                   # 「嗯，」「，啊。」去掉标点都不足 3 字


def test_words_made_up_over_silence_are_not_speech():
    """whisper 会在静音尾巴上编出「字幕由……提供」:能量到不了开口门槛,不算多说。"""
    env = [100.0] * 20 + [1.0] * 60                     # 前 2 秒有声,之后静音
    chars = _chars("这块玉佩", 0.0, 0.4) + _chars("字幕由某某提供", 3.0, 0.3)
    assert subtitles.extra_speech(["这块玉佩"], chars, env) == ([], "")


def test_untrustworthy_alignment_is_not_judged():
```

```python
EXTRA_MIN_CHARS = 3     # 台词之外多说至少这么多字(去标点)才报:单个「嗯」「啊」不算


def extra_speech(lines: list[str], chars: list[tuple[str, float, float]],
                 envelope: list[float]) -> tuple[list[dict], str]:
    """台词之外有没有多说(路线图未决项 #7:人物把舞台指示念了出来)。返回 ([{text, start, end}], 不能判断的原因)。
    - 复用 align 的匹配:每句覆盖识别字 [first − lead, last];覆盖之外的连续识别字是候选;
    - 任一句台词命中率低于 MIN_HIT → 不判(否则一句没识别出来,所有识别字都成了「多说」);
    - **用数据审 whisper**:候选所在时间段的能量要达到开口门槛 —— whisper 会在静音尾巴上编出「字幕由……提供」。"""
    aligned = align(lines, chars)
    if any(a["total"] and a["hit"] / a["total"] < MIN_HIT for a in aligned):
        return [], "识别与台词对不上,无法判断是否多说"
    covered: set[int] = set()
    for a in aligned:
        if a["first"] is not None:
            covered |= set(range(max(0, a["first"] - a["lead"]), a["last"] + 1))
    groups, cur = [], []
    for i in range(len(chars)):
        if i in covered:
            if cur:
                groups.append(cur)
            cur = []
        else:
            cur.append(i)
    if cur:
        groups.append(cur)
    th = (max(envelope) if envelope else 0.0) * ONSET_RATIO
    runs = []
    for g in groups:
        text = _clean("".join(chars[i][0] for i in g))
        start, end = chars[g[0]][1], chars[g[-1]][2]
        voiced = th > 0 and any(envelope[k] >= th for k in range(round(start / STEP), min(len(envelope), round(end / STEP) + 1)))
        if len(text) >= EXTRA_MIN_CHARS and voiced:
            runs.append({"text": text, "start": start, "end": end})
    return runs, ""
```

```python
"""视频面板的「台词外多说」提醒:只认当前视频的对齐(设计 specs/2026-10-10-台词外多说提示-design.md §二、§三)。"""
import json

from tools.drama import repo
from tools.drama.tasks import create_task, set_task_state
from tools.serve import aligner

REAL = json.loads((Path(__file__).parents[1] / "cut/fixtures/align_real.json").read_text(encoding="utf-8"))["runs"]


def _aligned(store, root, seg, video, run, state="done"):
    t = create_task(store, kind="align", target_type="segment", target_id=seg, model="whisper", seed=0, prompt="p",
                    params={"workflow": "whisper_align", "video": video, "lines": run["lines"],
                            "writes": "subtitle_path"}, refs=[video])
    out = Path(root) / "outputs/tasks" / str(t)
    out.mkdir(parents=True, exist_ok=True)
    (out / "align.json").write_text(json.dumps({"lines": run["lines"], "chars": run["chars"],
                                                "envelope": run["envelope"]}, ensure_ascii=False), encoding="utf-8")
    if state == "done":
        set_task_state(store, t, "queued", external_id=f"align-{t}")
        set_task_state(store, t, "done", result_path=f"outputs/tasks/{t}/subtitles.srt", result_sha256="x")
    else:
        set_task_state(store, t, "failed", error="判卷不过:…")
    return t


def test_extra_speech_for_the_current_video(store, sample, tmp_path):
    _aligned(store, tmp_path, sample["seg"], "videos/s1.mp4", REAL["2026-09-29"])
```

```python
def extra_speech_for(store: Store, data_root: Path, segment_id: int) -> dict | None:
    """当前视频的最新一次对齐(成功、失败都算 —— 失败的也留着 align.json)里,台词之外有没有多说。
    只认 params.video 等于当前视频的:换过视频,旧对齐属于旧视频,不拿来判新视频。没有对齐记录返回 None。"""
    seg = repo.get(store, "segment", segment_id)
    if not seg["video_path"]:
        return None
    task = store.conn.execute(
        "SELECT id FROM task WHERE kind = 'align' AND target_type = 'segment' AND target_id = ?"
        " AND state IN ('done', 'failed') AND json_extract(params_json, '$.video') = ? ORDER BY id DESC LIMIT 1",
        (segment_id, seg["video_path"])).fetchone()
    f = Path(data_root) / "outputs" / "tasks" / str(task["id"]) / "align.json" if task else None
    if f is None or not f.is_file():
        return None
    d = json.loads(f.read_text(encoding="utf-8"))
    runs, problem = subtitles.extra_speech(d["lines"], [tuple(c) for c in d["chars"]], d["envelope"])
    return {"runs": runs, "problem": problem}
```
