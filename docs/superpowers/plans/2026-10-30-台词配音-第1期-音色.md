```python
# -*- coding: utf-8 -*-
"""台词「念得对不对」的比对:按无声调拼音比,不按汉字比。

Whisper 对 TTS / 成片的转写常出同音字与繁体(「伙夫」→「霍夫」、「新来的」→「新來的」),按字比会把念对的判错
(spike-results/2026-10-29-台词与口型匹配方案调研.md 第 4.2 节:10 条里不满分的 4 条全是这种)。
"""
from __future__ import annotations

import difflib

from pypinyin import lazy_pinyin

from tools.cut.subtitles import PUNCT


def syllables(text: str) -> list[str]:
    clean = "".join(c for c in text if c not in PUNCT and not c.isspace())
    return [s.lower() for s in lazy_pinyin(clean)]


def hit(want: str, heard: str) -> float:
    """要求的台词有多少按拼音被听到(0..1,difflib 的匹配比)。要求为空时算 1。"""
    a, b = syllables(want), syllables(heard)
    if not a:
        return 1.0
    return difflib.SequenceMatcher(None, a, b, autojunk=False).ratio()
```

# -*- coding: utf-8 -*-
"""角色试听音:按「声音描述」让 Qwen3-TTS VoiceDesign 念一句(spec 2026-10-30-台词配音与口型闸门-design.md 第 3 节)。

试听句取这个角色剧本(段落描述)里最长的一句台词 —— 定稿后它就是克隆用的参考文本,句子越长音色越稳;
角色还没台词就用固定句。种子由调用方给(面板一次排 3 个)。参数规划时定死。
"""
from __future__ import annotations

from tools.drama import repo, rules
from tools.drama.db import Store
from tools.drama.tasks import create_task
from tools.gen.capability import QWEN3_TTS

FALLBACK_TEXT = "我听见了,你放心,这件事交给我。"
MAX_CHARS = 40          # 参考音频 3–10 秒为宜(Qwen3-TTS 卡片);太长的句子不当试听句


def sample_text(store: Store, character_id: int) -> str:
    ch = repo.get(store, "character", character_id)
    lines = []
    for seg in store.conn.execute("SELECT s.description FROM segment s JOIN episode e ON e.id = s.episode_id"
                                  " WHERE e.drama_id = ? ORDER BY e.number, s.number", (ch["drama_id"],)):
        lines += [line for who, line in rules.dialogue_lines(seg["description"], [ch["name"]])
                  if who.startswith(ch["name"]) and len(line) <= MAX_CHARS]
    return max(lines, key=len) if lines else FALLBACK_TEXT


def plan_voice(store: Store, character_id: int, *, seed: int) -> int:
    ch = repo.get(store, "character", character_id)
    desc = (ch["voice_desc"] or "").strip()
    if not desc:
        raise ValueError(f"角色「{ch['name']}」还没有声音描述,先到资产面板填")
    params = {"workflow": "qwen3_tts_design", "instruct": desc, "text": sample_text(store, character_id),
              "language": QWEN3_TTS.language, "writes": "voice_ref_path", "expect_ext": ".wav"}
    return create_task(store, kind="voice", target_type="character", target_id=character_id, model=QWEN3_TTS.name,
                       params=params, seed=seed, prompt=desc, refs=[])
```

(`episode` 表的列名以 `db.py` 为准:先 `grep -n "CREATE TABLE episode" -A8 tools/drama/db.py` 核对 `drama_id`、`number`。)

# -*- coding: utf-8 -*-
"""**在云机上跑**:Qwen3-TTS VoiceDesign 按声音描述念一句。源码经 SSH 的 stdin 送过去,用 qwen-tts 独立 venv 执行:

    <venv python> - <模型目录> <声音描述> <台词> <种子> <语言>

stdout 最后一行是 JSON:`{"wav_b64": ..., "sr": 24000, "sec": 2.5, "gen_sec": 4.1}`。失败时退出码非 0、stderr 写原因:
2 = 模型目录不在;3 = 显存不够。

本模块**顶层只用标准库**:torch / qwen_tts 在 `main()` 里才导入。
"""
from __future__ import annotations

import base64
import io
import json
import sys
import time
from pathlib import Path


def main(argv: list[str]) -> int:
    model_dir, instruct, text, seed, language = argv
    if not Path(model_dir).is_dir():
        print(f"不存在:{model_dir}", file=sys.stderr)
        return 2
    import soundfile as sf
    import torch
    from qwen_tts import Qwen3TTSModel

    t0 = time.time()
    try:
        model = Qwen3TTSModel.from_pretrained(model_dir, device_map="cuda:0", dtype=torch.bfloat16)
        torch.manual_seed(int(seed))
        wavs, sr = model.generate_voice_design(text=text, instruct=instruct, language=language)
    except torch.cuda.OutOfMemoryError as e:
        print(f"显存不够(ComfyUI 可能正占着卡):{e}", file=sys.stderr)
        return 3
    buf = io.BytesIO()
    sf.write(buf, wavs[0], sr, format="WAV")
    print(json.dumps({"wav_b64": base64.b64encode(buf.getvalue()).decode(), "sr": sr,
                      "sec": round(len(wavs[0]) / sr, 2), "gen_sec": round(time.time() - t0, 1)}))
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
```

def gpu_ready(client, need_gb: float, what: str) -> "Outcome | None":
    """同卡跑 ComfyUI 之外的模型前:队列里有任何任务(共用机)→ pending 不打断;空闲而显存不够 → 让 ComfyUI 卸载模型,
    等显存腾出来;等不到 → failed 的原因串。返回 None = 可以跑;返回 Outcome = 照它返回(failed 时调用方自己落库)。
    连不上云机由调用方接 ComfyUnreachable。"""
    running, pending = client.queue_ids()
    if running or pending:
        return Outcome("pending", f"ComfyUI 队列里还有 {len(running) + len(pending)} 个任务(共用机),等队列空了再{what}")
    vram = client.vram_free_gb()
    if vram < need_gb:
        client.free_models()
        deadline = time.monotonic() + FREE_WAIT_SEC
        while (vram := client.vram_free_gb()) < need_gb and time.monotonic() < deadline:
            time.sleep(2)
        if vram < need_gb:
            return Outcome("failed", f"让 ComfyUI 卸载模型后,释放后显存仍只有 {vram:.1f} GiB({what}要 {need_gb:g} GiB)"
                                     " —— 显存被 ComfyUI 之外的进程占着,请上云机查 nvidia-smi")
    return None

class FakeVoice:
    """假的云机配音脚本(行为像 ssh runner):回一段 0.5 秒静音 wav,与真脚本同格式的 JSON。"""
    def __init__(self):
        self.calls: list[list[str]] = []

    def __call__(self, args, stdin):
        import base64, io, wave
        self.calls.append(list(args))
        buf = io.BytesIO()
        with wave.open(buf, "wb") as w:
            w.setnchannels(1); w.setsampwidth(2); w.setframerate(24000); w.writeframes(b"\0\0" * 12000)
        return 0, json.dumps({"wav_b64": base64.b64encode(buf.getvalue()).decode(), "sr": 24000, "sec": 0.5,
                              "gen_sec": 0.1})

"""执行一条试听音任务(kind = voice):云机上 Qwen3-TTS 念一句 → wav 回本机 → 传回云机让 whisper 转写 → **本机**按拼音判。

与字幕对齐同一套结局(`aligner`):连不上 → pending;ComfyUI 队列有任务 → pending 不打断;脚本报错 → failed。
**没念对不算失败**:试听音照样落盘、照样能挑,`voice.json` 记下听到了什么,定稿时算不通过项、要写理由
(spec 2026-10-30-台词配音与口型闸门-design.md 第 3 节)。写回只填空位。
"""
from __future__ import annotations

import base64
import json
import sqlite3
from pathlib import Path

from tools.cut import speech
from tools.drama import repo
from tools.drama.db import Store
from tools.drama.tasks import set_task_state

from tools.serve.dispatch import Outcome, sha256_file
from tools.serve.hosts import Host

SCRIPT = Path(__file__).with_name("voice_remote.py")
HIT_OK = 0.9


def run(store: Store, client, host: Host, task: sqlite3.Row, *, data_root: Path,
        run_tts: preflight.Runner, run_align: preflight.Runner) -> Outcome:
    tid, root, p = task["id"], Path(data_root), json.loads(task["params_json"])

    def fail(msg: str) -> Outcome:
        set_task_state(store, tid, "failed", error=msg)
        return Outcome("failed", msg)

    try:
        wait = aligner.gpu_ready(client, QWEN3_TTS.vram_gb, "配音")
    except ComfyUnreachable as e:
        return Outcome("pending", f"连不上云机,下轮再试:{e}")
    if wait is not None:
        return fail(wait.message) if wait.state == "failed" else wait

    code, out = run_tts([QWEN3_TTS.design_dir, p["instruct"], p["text"], str(task["seed"]), p["language"]],
                        SCRIPT.read_bytes())
    if code == aligner.SSH_UNREACHABLE:
        return Outcome("pending", f"ssh 连不上云机,下轮再试:{out.strip()[-300:]}")
    if code != 0:
        return fail(f"云机配音脚本失败(退出码 {code}):{out.strip()[-500:]}")
    rows = [line for line in out.splitlines() if line.startswith('{"wav_b64"')]
    if not rows:
        return fail(f"云机配音脚本没有输出结果:{out.strip()[-500:]}")
    got = json.loads(rows[-1])
    out_dir = root / "outputs" / "tasks" / str(tid)
    out_dir.mkdir(parents=True, exist_ok=True)
    wav = out_dir / "voice.wav"
    wav.write_bytes(base64.b64decode(got["wav_b64"]))

    try:
        name = client.upload_image(wav, subfolder=f"ai-short-drama/{sha256_file(wav)[:16]}")
    except ComfyUnreachable as e:
        return Outcome("pending", f"连不上云机,下轮再试:{e}")
    except ComfyError as e:
        return fail(f"试听音传不上云机:{e}")
    code, out = run_align([WHISPER_ALIGN.model_dir, f"{WHISPER_ALIGN.input_dir}/{name}"], aligner.SCRIPT.read_bytes())
    if code == aligner.SSH_UNREACHABLE:
        return Outcome("pending", f"ssh 连不上云机,下轮再试:{out.strip()[-300:]}")
    if code != 0:
        return fail(f"云机转写脚本失败(退出码 {code}):{out.strip()[-500:]}")
    try:
        heard = "".join(c for c, _, _ in aligner.parse_output(out))
    except ValueError as e:
        return fail(str(e))
    hit = round(speech.hit(p["text"], heard), 3)
    (out_dir / "voice.json").write_text(json.dumps(
        {"text": p["text"], "heard": heard, "hit": hit, "ok": hit >= HIT_OK, "sec": got["sec"],
         "gen_sec": got["gen_sec"]}, ensure_ascii=False, indent=1), encoding="utf-8")

    rel = wav.relative_to(root).as_posix()
    set_task_state(store, tid, "queued", external_id=f"voice-{tid}", host_id=host.id)
    note = ""
    if repo.get(store, "character", task["target_id"])["voice_ref_path"]:
        note = "voice_ref_path 已有试听音,新版本留在任务记录里待人挑选"
    else:
        repo.update(store, "character", task["target_id"], voice_ref_path=rel, voice_ref_text=p["text"])
    set_task_state(store, tid, "done", result_path=rel, result_sha256=sha256_file(wav))
    return Outcome("done", note or rel)


def check_of(data_root: Path, path: str | None) -> dict | None:
    """这条试听音的回听结果(与 wav 同目录的 voice.json);没有(上传的、旧的)返回 None。"""
    if not path:
        return None
    f = (Path(data_root) / path).with_name("voice.json")
    return json.loads(f.read_text(encoding="utf-8")) if f.is_file() else None
