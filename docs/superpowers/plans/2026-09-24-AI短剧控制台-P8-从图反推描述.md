# AI 短剧控制台 · P8 从图反推描述 · 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 资产面板能**上传参考图**,并能**从图反推描述**:视觉 LLM 看角色 / 场景 / 道具的参考图,填出外貌、造型、地点、光线、生成描述等字段,人在审阅页批准才写回。

**Architecture:** `tools/drama/uploads.py` 把上传的 PNG / JPEG 按内容寻址落盘、记成创建即完成的 task,替换已用的图走 plan / confirm。`tools/llm/describe.py` 是一个新的 LLM 任务:参考图缩到最长边 768 后在底部**印一个校验数字**,schema 要模型读出它 —— 读错就是没看到图,闸门打回、轮次用完 fail-loud。闸门与调度层补上图片输入、任务专用模型、看图任务的素材根;库升到 v5 让 LLM 记录能挂在场景 / 道具上。客户端顺带修掉 SynScale 的 `tool_choice` 拼法问题。

**Tech Stack:** Python 3.14 标准库 · 本机 ffmpeg(缩图、印数字,Noto Sans SC)· SynScale 视觉模型 `gpt-5.6-luna` · 现有 FastAPI / Jinja2 / Playwright。**不加依赖**。

def request_body(*, model: str, messages: list[dict], tool_name: str, description: str, schema: dict,
                 max_tokens: int, spelling: str = "both") -> dict:
    """强制调用一个 strict 工具的请求体。纯函数,可单测。"""
    choice = {"both": {"type": "function", "name": tool_name, "function": {"name": tool_name}},
              "flat": {"type": "function", "name": tool_name},
              "chat": {"type": "function", "function": {"name": tool_name}}}[spelling]
    return {
        "model": model, "messages": messages, "max_tokens": max_tokens,
        "tools": [{"type": "function", "function": {"name": tool_name, "description": description,
                                                    "strict": True, "parameters": schema}}],
        "tool_choice": choice,
    }

def check_number(original: bytes) -> int:
    """校验数字由原图内容决定(100–999):同一张图每次相同,锁定版可复用。"""
    return 100 + int(hashlib.sha256(original).hexdigest()[:8], 16) % 900


def marked(path: Path) -> tuple[bytes, int]:
    """(印了校验数字的图, 数字)。"""
    n = check_number(Path(path).read_bytes())
    return media.marked_png(Path(path), n), n


def context(store: Store, table: str, row_id: int, data_root: Path) -> dict:
    if table not in FIELDS:
        raise ValueError(f"不是资产表:{table}")
    row = repo.get(store, table, row_id)
    rel = row[SPEC[table][0]]
    if not rel:
        raise ValueError(f"{KIND[table]}「{row['name']}」还没有参考图 —— 先上传或生成一张,再从图反推描述")
    path = Path(data_root) / rel
    if not path.is_file():
        raise ValueError(f"参考图不在本机:{rel}")
    image, n = marked(path)
    return {"table": table, "name": row["name"], "image_path": rel, "image": image, "check_number": n}


def images(ctx: dict) -> list[bytes]:
    return [ctx["image"]]


def schema(ctx: dict) -> dict:
    props = {k: ({"type": "string", "enum": TIMES} if k == "time" else {"type": "string"}) for k in FIELDS[ctx["table"]]}
    props["check_number"] = {"type": "string"}
    return {"type": "object", "additionalProperties": False, "required": list(props), "properties": props}


def user_message(ctx: dict) -> str:
    """注意:校验数字**不许**出现在文字里 —— 它只在图上。"""
    fields = "\n".join(f"- {k}:{v}" for k, v in FIELDS[ctx["table"]].items())
    return (f"<asset>\n类型:{KIND[ctx['table']]}\n名字:{ctx['name']}\n</asset>\n"
            f"图是这个{KIND[ctx['table']]}的参考图。要填的字段:\n{fields}\n- check_number:图底部蓝色色带上的三位数")


def judge_factory(ctx: dict):
    def judge(args: dict) -> list[str]:
        problems = []
        if args["check_number"].strip() != str(ctx["check_number"]):
            problems.append("check_number 读错了 —— 请真正看图,读出图最下方蓝色色带上的三位数;"
                            "读不出来说明本次请求没有收到图片")
        for k in FIELDS[ctx["table"]]:
            v = args[k].strip()
            if k == "time":
                continue
            if not v:
                problems.append(f"{k} 不能为空")
            if "像" in v and k != "final_prompt":
                problems.append(f"{k} 里不许用「像某某」比喻真实的人或作品")
        fp = args["final_prompt"]
        if fp and sum(c.isascii() for c in fp) / len(fp) < 0.95:
            problems.append("final_prompt 要写英文")
        if len(fp) > 600:
            problems.append(f"final_prompt 太长({len(fp)} 字符,上限 600)")
        return problems
    return judge


def apply(store: Store, row_id: int, args: dict, ctx: dict) -> None:
    repo.update(store, ctx["table"], row_id, **{k: args[k].strip() for k in FIELDS[ctx["table"]]})

# -*- coding: utf-8 -*-
"""资产上传参考图:人手里已有的图直接当角色 / 场景 / 道具的参考图(设计 §七「资产」面板)。

- 只收 PNG / JPEG(看文件头,不看扩展名),上限 20 MB;按内容寻址存到 `assets/uploads/<sha256 前 16 位>.<png|jpg>`;
- 上传记成一条**创建即完成**的 task(`model = manual:upload`):进历史、能用「用这一版」换回,溯源里照实写「人工上传」;
- 替换已有参考图会让绑定了这个资产的段落的参考帧与视频作废 → plan / confirm,先报代价(与「用这一版」同一套失效传播)。
"""
from __future__ import annotations

import hashlib
import re
from pathlib import Path

from tools.drama import invalidation, repo
from tools.drama.db import Store
from tools.drama.invalidation import Impact
from tools.drama.tasks import record_local
from tools.gen.assets import SPEC
from tools.gen.capability import MINIMAX_H3, QWEN_IMAGE_EDIT_2511

MODEL = "manual:upload"
MAX_BYTES = 20 * 1024 * 1024
_MAGIC = {b"\x89PNG\r\n\x1a\n": ".png", b"\xff\xd8\xff": ".jpg"}
_STAGED = re.compile(r"^assets/uploads/[0-9a-f]{16}\.(png|jpg)$")


def stage(data_root: Path, data: bytes) -> str:
    """校验并落盘,返回相对 data_root 的路径。同一张图传几次都是同一个文件。"""
    ext = next((e for magic, e in _MAGIC.items() if data.startswith(magic)), None)
    if ext is None:
        raise ValueError("只收 PNG 或 JPEG 图片(按文件内容判断,不看扩展名)")
    if len(data) > MAX_BYTES:
        raise ValueError(f"图片太大({len(data) / 1024 / 1024:.1f} MB,上限 {MAX_BYTES // 1024 // 1024} MB)")
    rel = f"assets/uploads/{hashlib.sha256(data).hexdigest()[:16]}{ext}"
    path = Path(data_root) / rel
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(data)
    return rel


def plan_use(store: Store, table: str, row_id: int) -> Impact:
    """换成上传的图会让哪些产物作废(只算不改)。还没有参考图时不影响任何产物。"""
    if table not in SPEC:
        raise ValueError(f"不是资产表:{table}")
    if not repo.get(store, table, row_id)[SPEC[table][0]]:
        return Impact((), 0, 0, 0.0)
    return invalidation.impact(store, table, row_id, video_cap=MINIMAX_H3, image_cap=QWEN_IMAGE_EDIT_2511)


def confirm_use(store: Store, data_root: Path, table: str, row_id: int, rel: str, filename: str, imp: Impact) -> int:
    if not _STAGED.match(rel) or not (Path(data_root) / rel).is_file():
        raise ValueError(f"不是本机暂存的上传图:{rel}")
    field = SPEC[table][0]
    sha = hashlib.sha256((Path(data_root) / rel).read_bytes()).hexdigest()
    tid = record_local(store, kind="image", target_type=table, target_id=row_id, model=MODEL,
                       params={"workflow": "upload", "writes": field, "filename": filename},
                       prompt=f"人工上传:{filename}", refs=[], result_path=rel, result_sha256=sha)
    invalidation.apply(store, imp)
    repo.update(store, table, row_id, **{field: rel})
    return tid
