# AI 短剧控制台 · P2 执行层 · 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把闸门 0 的一次性执行器变成常驻、可恢复、可审计的生成管线:一个段落从「规划首帧」走到「视频落盘」,任务状态、产物、溯源字段全部进库。

**Tech Stack:** Python 3.14 标准库(`sqlite3`、`subprocess`、`socket`、`urllib`)· 现有 `tools/gen/comfy_client.py` · pytest 9。无新增依赖。

## 关键设计决定(执行时不要改)

1. **规划 ≠ 提交**。planner 在控制面立刻写出参数定死的 task;云机通不通不影响规划。
2. **每台机器同时只交一条**。ComfyUI 本来串行;排在我们库里才能按模型挑、才能在交出去之前取消。这台 ComfyUI 是多项目共用的。
3. **先看队列、再看 history**。任务跑完是先进 history 再出队列,反过来查会在两次请求之间漏掉刚跑完的任务。
4. **消失 ≠ 立刻失败**。闸门 0 实测 history 偶尔滞后,连续 120s 找不到才判失败。
5. **连不上 ≠ 失败**。派发时连不上 → 留在 `pending`;在途时连不上 → `waiting_host`,24 小时才放弃。
6. **写回只填空位**。目标字段已有产物时不覆盖,新版本留在 task 记录里等人挑 —— 否则会悄悄换掉首帧,而由旧首帧生成的视频还挂着。
7. **参考图按内容哈希分子目录上传**(`ai-short-drama/<sha256 前 16 位>/`),防同名产物互相覆盖。
8. **`started_at` 可能为空**:4 步出图可能在两次轮询之间跑完,没被看到 `running`。如实留空,不编时间。

# -*- coding: utf-8 -*-
"""执行面机器(云机)的登记。

- **私钥不入库**:只存 `key_ref`(密钥文件路径)。拿着私钥能在云机上做任何事,
  它不能出现在数据库、日志和 git 里。
- **SSH 端口可改**:AutoDL 每次重启都会换端口,改端口是日常操作。
- `models_dir` 是云机上 ComfyUI 模型根目录,LoRA 预检要按它定位文件。
"""
from __future__ import annotations

from dataclasses import dataclass

from tools.drama.db import Store


@dataclass(frozen=True)
class Host:
    id: int
    label: str
    ssh_host: str
    ssh_port: int
    ssh_user: str
    key_ref: str
    local_port: int
    remote_comfy_port: int
    vram_gb: float
    models_dir: str

    @property
    def local_endpoint(self) -> str:
        """ComfyClient 该连的地址。客户端只认本地端口,不知道隧道存在。"""
        return f"127.0.0.1:{self.local_port}"


def _check_key_ref(key_ref: str) -> None:
    if "PRIVATE KEY" in key_ref or "\n" in key_ref.strip():
        raise ValueError("key_ref 看起来是密钥内容而不是文件路径 —— 私钥不许入库")


def add_host(store: Store, *, label: str, ssh_host: str, ssh_port: int, ssh_user: str, key_ref: str,
             local_port: int, vram_gb: float, models_dir: str, remote_comfy_port: int = 8188) -> int:
    _check_key_ref(key_ref)
    if not models_dir.startswith("/"):
        raise ValueError(f"models_dir 必须是云机上的绝对路径,给的是 {models_dir!r}")
    with store.write() as c:
        return c.execute(
            "INSERT INTO host (label, ssh_host, ssh_port, ssh_user, key_ref, local_port,"
            " remote_comfy_port, vram_gb, models_dir) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)",
            (label, ssh_host, ssh_port, ssh_user, key_ref, local_port, remote_comfy_port,
             vram_gb, models_dir)).lastrowid


def get_host(store: Store, host_id: int) -> Host:
    row = store.conn.execute("SELECT * FROM host WHERE id = ?", (host_id,)).fetchone()
    if row is None:
        raise KeyError(f"host {host_id} 不存在")
    return Host(**{k: row[k] for k in row.keys()})


def set_ssh_port(store: Store, host_id: int, ssh_port: int) -> None:
    with store.write() as c:
        if c.execute("UPDATE host SET ssh_port = ? WHERE id = ?", (ssh_port, host_id)).rowcount == 0:
            raise KeyError(f"host {host_id} 不存在")
```

# -*- coding: utf-8 -*-
"""把任务参数拼成 ComfyUI API 格式工作流 —— 纯函数,可单测。

两个模板都来自闸门 0 已在云机跑通的工作流(`tools/gen/workflows/*.json`),
节点 id 与模板绑定:改模板要同步改这里。

规矩:
- **种子由调用方给**,不许随机;
- **提示词先做标签双向校验再拼**,不通过直接抛错 —— 不把错位的标签交给 ComfyUI;
- 未使用的可选输入**整个删掉**,不传 null(ComfyUI 会把 null 当类型错误);
- H3 只出 **20 步档**,并把 LoRA 与开关节点从图里**剪掉**:PDD 加速 LoRA 与云机的剪枝版底模
  不兼容(闸门 0-6),ComfyUI 遇到不兼容只打日志、照样出片,所以图里根本不该出现它。
"""
from __future__ import annotations

import copy
import json
from pathlib import Path

from tools.drama.tags import check_tags

WORKFLOW_DIR = Path(__file__).parent / "workflows"
QWEN_TEMPLATE = WORKFLOW_DIR / "qwen_image_edit_2511_gguf_api.json"
H3_TEMPLATE = WORKFLOW_DIR / "minimax_h3_ref2va_api.json"

QWEN_MODEL = "qwen-image-edit-2511-Q4_K_M.gguf"
H3_MODEL = "minimax_h3_ref2va_pruned_fp8_scaled.safetensors"

QWEN_PRESETS = {"fast": (4, 1.0), "baseline": (20, 4.0)}   # (步数, CFG);fast 是闸门 0 定的默认档
H3_STEPS = 20

FRAME_SIZE = {"9:16": (928, 1664), "16:9": (1664, 928)}    # Qwen 原生档
VIDEO_SIZE = {"9:16": (768, 1344), "16:9": (1344, 768)}    # H3 竖屏已在云机跑通


def _load(path: Path) -> dict:
    return copy.deepcopy(json.loads(path.read_text(encoding="utf-8")))


def _raise_if(problems: list[str]) -> None:
    if problems:
        raise ValueError(";".join(problems))


def build_qwen(prompt: str, image_names: list[str], seed: int, *, preset: str = "fast",
               size: tuple[int, int] | None = None, angles_lora: bool = False) -> dict:
    """Qwen-Image-Edit-2511(GGUF)。size 为 None 时输出画幅跟随参考图 1(改背景、出四视图);
    合成分镜首帧**必须给 size** —— 闸门 0-3 实测提示词压不过 latent 尺寸。"""
    if not 1 <= len(image_names) <= 3:
        raise ValueError(f"Qwen-Image-Edit 需要 1–3 张参考图,给了 {len(image_names)} 张")
    _raise_if(check_tags(prompt, len(image_names), dialect="qwen"))
    steps, cfg = QWEN_PRESETS[preset]
    wf = _load(QWEN_TEMPLATE)

    for idx, node in enumerate(("18", "19", "20")):
        if idx < len(image_names):
            wf[node]["inputs"]["image"] = image_names[idx]
        else:
            wf.pop(node)
            wf["21"]["inputs"].pop(f"image{idx + 1}")

    wf["4"]["inputs"]["value"] = angles_lora
    wf["7"]["inputs"]["value"] = preset == "fast"
    wf["10"]["inputs"]["value"] = wf["11"]["inputs"]["value"] = steps
    wf["13"]["inputs"]["value"] = wf["14"]["inputs"]["value"] = cfg
    wf["21"]["inputs"]["prompt"] = prompt
    wf["27"]["inputs"]["seed"] = seed

    if size is None:
        wf.pop("30")
    else:
        w, h = size
        if w % 16 or h % 16:
            raise ValueError(f"尺寸 {w}x{h} 不是 16 的倍数")
        wf["30"]["inputs"]["width"], wf["30"]["inputs"]["height"] = w, h
        wf["27"]["inputs"]["latent_image"] = ["30", 0]
        wf.pop("25")
        wf.pop("26")
    return wf


def build_h3(prompt: str, image_names: list[str], audio_name: str | None, frames: int, seed: int,
             *, size: tuple[int, int]) -> dict:
    """MiniMax H3 ref2va,20 步。frames 须已由 Capability 换算并校验过。"""
    if not 1 <= len(image_names) <= 3:
        raise ValueError(f"工作流预留 1–3 个图槽,给了 {len(image_names)} 张")
    _raise_if(check_tags(prompt, len(image_names), 1 if audio_name else 0, dialect="h3"))
    wf = _load(H3_TEMPLATE)

    # 剪掉 LoRA 与步数开关:2 LoRA · 3/4 开关 · 15/16/17 步数分支
    for node in ("2", "3", "4", "15", "16", "17"):
        wf.pop(node)
    wf["13"]["inputs"]["model"] = ["1", 0]
    wf["18"]["inputs"]["model"] = ["1", 0]
    wf["18"]["inputs"]["steps"] = H3_STEPS

    r2v = wf["12"]["inputs"]
    for idx, node in enumerate(("8", "9", "10")):
        if idx < len(image_names):
            wf[node]["inputs"]["image"] = image_names[idx]
        else:
            wf.pop(node)
            r2v.pop(f"ref_images.ref_image_{idx}")
    if audio_name:
        wf["11"]["inputs"]["audio"] = audio_name
    else:
        wf.pop("11")
        r2v.pop("ref_audios.ref_audio_0")

    r2v["prompt"] = prompt
    r2v["length"] = frames
    r2v["width"], r2v["height"] = size
    wf["19"]["inputs"]["noise_seed"] = seed
    return wf
```

# -*- coding: utf-8 -*-
"""把「这个段落要出首帧 / 要出视频」变成一条不可变的生成任务。

**规划 ≠ 提交**:规划只看**内容**是否就绪(素材、提示词、引用、时长),在控制面上立刻完成;
云机通不通、模型在不在,是提交时(worker)的事。

任务一旦创建,提示词、参考素材、种子、参数都定死(P1 的数据库触发器保证),
worker 只负责把它原样交给云机 —— 这样「当时到底用了什么生成的」永远可查。

闸门 0 带来的两条规矩:
- 首帧是**合成**:必须用目标画幅的空 latent,尺寸来自剧的画幅,不来自参考图(0-3);
- 视频**不喂参考音频**:不喂也一致、喂了会复读(0-4)。音色靠角色的声音描述维持。
"""
from __future__ import annotations

from tools.drama import refs, repo, rules, tags
from tools.drama.db import Store
from tools.drama.readiness import segment_readiness
from tools.drama.tasks import create_task
from tools.gen.capability import MINIMAX_H3, QWEN_IMAGE_EDIT_2511
from tools.gen.workflows import FRAME_SIZE, H3_MODEL, QWEN_MODEL, VIDEO_SIZE

# readiness 里属于「提交时」的检查项;规划阶段不看
RUNTIME_KEYS = {"host_ready", "video_model_ready"}


def _aspect(store: Store, segment_id: int) -> str:
    row = store.conn.execute("SELECT d.aspect_ratio FROM segment s JOIN episode e ON e.id = s.episode_id "
                             "JOIN drama d ON d.id = e.drama_id WHERE s.id = ?", (segment_id,)).fetchone()
    return row[0]


def _slot_refs(seg, assets) -> tuple[list[refs.Ref], list[refs.Ref]]:
    chars, scene, props = assets["characters"], assets["scene"], assets["props"]
    spoken = rules.dialogue_by_speaker(seg["description"])
    weighted = [(refs.Ref("character", c["name"], c["turnaround_path"]),
                 sum(n for who, n in spoken.items() if who.startswith(c["name"]))) for c in chars]
    scene_ref = refs.Ref("scene", scene["name"], scene["image_path"]) if scene is not None else None
    prop_refs = [refs.Ref("prop", p["name"], p["image_path"]) for p in props]
    return refs.allocate_slots(weighted, scene_ref, prop_refs, QWEN_IMAGE_EDIT_2511.max_ref_images)


def plan_first_frame(store: Store, segment_id: int, *, seed: int) -> int:
    seg = repo.get(store, "segment", segment_id)
    if seg["reference_mode"] != "first":
        raise ValueError(f"参考模式 {seg['reference_mode']} 的首帧规划尚未实现(当前只支持 first)")
    assets = repo.segment_assets(store, segment_id)
    chars, scene, props = assets["characters"], assets["scene"], assets["props"]

    problems = []
    if not seg["image_prompt"].strip():
        problems.append("缺少首帧图提示词")
    if scene is None:
        problems.append("未绑定场景")
    problems += [f"角色「{c['name']}」缺四视图" for c in chars if not c["turnaround_path"]]
    if scene is not None and not scene["image_path"]:
        problems.append(f"场景「{scene['name']}」缺空镜图")
    problems += [f"道具「{p['name']}」缺白底图" for p in props if not p["image_path"]]
    bound = [c["name"] for c in chars] + ([scene["name"]] if scene is not None else []) + [p["name"] for p in props]
    found, unknown = refs.find_mentions(seg["image_prompt"], bound)
    problems += [f"@{u} 不是本镜绑定的素材" for u in unknown]
    if len(bound) >= 2:
        problems += [f"绑定了「{n}」但首帧图提示词从未 @ 它" for n in bound if n not in found]
    if problems:
        raise ValueError(";".join(problems))

    kept, dropped = _slot_refs(seg, assets)
    prompt = tags.expand(seg["image_prompt"], kept, dropped, "qwen")
    params = {
        "workflow": "qwen_image", "preset": "fast", "size": list(FRAME_SIZE[_aspect(store, segment_id)]),
        "image_refs": [r.path for r in kept], "writes": "first_frame_path", "expect_ext": ".png",
        "dropped": [r.name for r in dropped],   # 出槽退化成文字的素材,溯源里要看得到
    }

# -*- coding: utf-8 -*-
"""提交前的 LoRA/底模形状预检 —— 在模型所在的云机上跑。

闸门 0-6:PDD LoRA 与剪枝版底模不兼容,ComfyUI 只打日志、照样出片。
所以每条要提交的工作流,凡用到 LoRA,都先核对形状;不兼容就判任务失败,不交给 ComfyUI。

实现:把 `tools/gen/lora_compat.py`(纯标准库)的源码经 ssh 的 stdin 喂给云机上的 `python3 -`,
不往云机传文件。

两种「过不了」要分清:
- 形状不兼容 → 失败,并给出哪一层不符;
- **预检本身没跑成**(ssh 断了、云机没有 python3)→ 也按失败处理。**不能核验就不放行。**
GGUF 底模不是 safetensors,读不了文件头 —— 如实给出「无法核验」的告警,不假装核验通过。
"""
from __future__ import annotations

import shlex
import subprocess
from collections.abc import Callable
from dataclasses import dataclass
from pathlib import Path

from tools.gen.lora_compat import EXIT_INCOMPATIBLE

# 远端执行器:(参数列表, 喂给 stdin 的程序源码) -> (退出码, 输出)
Runner = Callable[[list[str], bytes], tuple[int, str]]

COMPAT_SOURCE = (Path(__file__).parent / "lora_compat.py").read_bytes()


@dataclass(frozen=True)
class Preflight:
    ok: bool
    problems: tuple[str, ...]
    warnings: tuple[str, ...]


def lora_pairs(workflow: dict) -> tuple[str | None, list[str]]:
    """从工作流里找出底模文件名和用到的全部 LoRA 文件名。"""
    base, loras = None, []
    for node in workflow.values():
        kind, ins = node["class_type"], node["inputs"]
        if kind in ("UNETLoader", "UnetLoaderGGUF"):
            base = ins["unet_name"]
        elif kind == "LoraLoaderModelOnly":
            loras.append(ins["lora_name"])
    return base, loras


def check(workflow: dict, models_dir: str, run: Runner) -> Preflight:
    base, loras = lora_pairs(workflow)
    if not loras:
        return Preflight(True, (), ())
    if base is None:
        return Preflight(False, ("工作流用了 LoRA,却找不到底模节点",), ())
    if base.endswith(".gguf"):
        return Preflight(True, (), tuple(
            f"LoRA {name} 叠在 GGUF 底模 {base} 上,GGUF 读不了 safetensors 文件头,形状无法核验"
            for name in loras))
    problems = []
    for name in loras:
        rc, out = run([f"{models_dir}/diffusion_models/{base}", f"{models_dir}/loras/{name}"], COMPAT_SOURCE)
        if rc == EXIT_INCOMPATIBLE:
            problems.append(f"LoRA {name} 与底模 {base} 形状不兼容:{out.strip()}")
        elif rc != 0:
            problems.append(f"LoRA 预检没跑成(退出码 {rc}),不能核验就不放行:{out.strip()[:300]}")
    return Preflight(not problems, tuple(problems), ())


def ssh_runner(ssh_host: str, ssh_port: int, ssh_user: str, key_ref: str, timeout: int = 120) -> Runner:
    """在云机上执行 `python3 - <args>`,程序源码从 stdin 送过去。"""
    def run(args: list[str], stdin: bytes) -> tuple[int, str]:
        cmd = ["ssh", "-o", "BatchMode=yes", "-o", "ConnectTimeout=20", "-i", key_ref, "-p", str(ssh_port),
               f"{ssh_user}@{ssh_host}", "PYTHONIOENCODING=utf-8", "python3", "-",
               *(shlex.quote(a) for a in args)]
        try:
            p = subprocess.run(cmd, input=stdin, capture_output=True, timeout=timeout)
        except subprocess.TimeoutExpired:
            return 124, f"ssh 执行超时 {timeout}s"
        return p.returncode, (p.stdout + p.stderr).decode("utf-8", "replace")
    return run
```

# -*- coding: utf-8 -*-
"""SSH 隧道管理:把云机上只监听 127.0.0.1 的 ComfyUI 映射到本机端口。

**隧道由控制面自己起**:AutoDL 每次重启都换 SSH 端口,隧道若在应用之外,
界面上「改端口」就没有意义 —— 改完还得有人手工重建。

判活分两层,本模块只管第一层(**本机端口有没有在转发**);
「转发通不通、ComfyUI 活不活」是第二层,由调用方用 ComfyClient 判断。
**ssh 进程活着 ≠ 转发通**:云机关机时 ssh 可能僵着。

几条踩出来的规矩:
- 本机端口已经在转发(比如控制面重启前起的那条 ssh 还活着)就**接管**,不再起新的;
  否则新 ssh 会因 ExitOnForwardFailure 秒死,每轮生一条必死进程。
- 云机不可达时**退避**(5s 起翻倍、封顶 300s),不每轮都拉一条 ssh。
- 闸门 0 实测:**Windows 控制面上没有 `setsid`**。用 `subprocess.Popen` + DETACHED_PROCESS 脱离。
"""
from __future__ import annotations

import socket
import subprocess
import sys
import time
from collections.abc import Callable

from tools.serve.hosts import Host

BACKOFF_START = 5
BACKOFF_CAP = 300

# Windows:DETACHED_PROCESS | CREATE_NEW_PROCESS_GROUP —— 控制面退出时隧道不跟着被杀
_DETACH = 0x00000008 | 0x00000200 if sys.platform == "win32" else 0


def command_for(host: Host) -> list[str]:
    return [
        "ssh", "-N",
        "-o", "StrictHostKeyChecking=accept-new",
        "-o", "ExitOnForwardFailure=yes",      # 端口占用时立刻失败,别假装成功
        "-o", "ServerAliveInterval=30",
        "-o", "BatchMode=yes",                 # 拿不到 tty 也不卡在交互提示上
        "-i", host.key_ref,
        "-p", str(host.ssh_port),
        "-L", f"{host.local_port}:127.0.0.1:{host.remote_comfy_port}",
        f"{host.ssh_user}@{host.ssh_host}",
    ]


def _port_forwarding(host: Host) -> bool:
    with socket.socket() as s:
        s.settimeout(1)
        return s.connect_ex(("127.0.0.1", host.local_port)) == 0


def _spawn(cmd: list[str]):
    return subprocess.Popen(cmd, stdin=subprocess.DEVNULL, stdout=subprocess.DEVNULL,
                            stderr=subprocess.DEVNULL, creationflags=_DETACH)


class TunnelManager:
    def __init__(self, spawn: Callable[[list[str]], object] = _spawn,
                 forwarding: Callable[[Host], bool] = _port_forwarding,
                 clock: Callable[[], float] = time.monotonic):
        self._spawn = spawn
        self._forwarding = forwarding
        self._clock = clock
        self._procs: dict[int, object] = {}
        self._next_try: dict[int, float] = {}
        self._backoff: dict[int, int] = {}

    def ensure(self, host: Host) -> bool:
        """确保本机端口在转发。返回**此刻**是否在转发(刚拉起的 ssh 要过一会儿才通)。"""
        if self._forwarding(host):
            self._backoff.pop(host.id, None)
            self._next_try.pop(host.id, None)
            return True
        proc = self._procs.get(host.id)
        if proc is not None and proc.poll() is None:
            return False  # 我们起的 ssh 还活着,只是还没通,等它
        now = self._clock()
        if now < self._next_try.get(host.id, 0.0):
            return False  # 退避中
        self._procs[host.id] = self._spawn(command_for(host))
        wait = BACKOFF_START if host.id not in self._backoff else min(self._backoff[host.id] * 2, BACKOFF_CAP)
        self._backoff[host.id] = wait
        self._next_try[host.id] = now + wait
        return False

    def stop(self, host_id: int) -> None:
        """改端口前必须先停旧的:不停就换地址,旧隧道占着本机端口,新的起不来。"""
        proc = self._procs.pop(host_id, None)
        self._backoff.pop(host_id, None)
        self._next_try.pop(host_id, None)
        if proc is None:
            return
        proc.terminate()
        try:
            proc.wait(timeout=5)
        except subprocess.TimeoutExpired:
            proc.kill()

    def owns(self, host_id: int) -> bool:
        """这条隧道是不是我们起的。接管来的(控制面重启前起的)我们没有句柄,**杀不掉**,
        改端口时要如实告诉人去系统里结束那条 ssh,而不是静默地什么也没做。"""
        proc = self._procs.get(host_id)
        return proc is not None and proc.poll() is None
```

# -*- coding: utf-8 -*-
"""行为像 ComfyClient 的假云机:队列、history、产物、断线都可由测试操控。"""
from __future__ import annotations

from pathlib import Path

from tools.gen.comfy_client import ComfyClient, ComfyError, ComfyUnreachable


class FakeComfy:
    def __init__(self):
        self.reachable = True
        self.reject_next_submit: str | None = None   # 模拟 HTTP 400(工作流本身写错)
        self.uploaded: list[str] = []
        self.submitted: dict[str, dict] = {}
        self.running: list[str] = []
        self.pending: list[str] = []
        self.records: dict[str, dict] = {}
        self.files: dict[str, bytes] = {}
        self._n = 0

    # ── ComfyClient 的接口 ──────────────────────────
    def _up(self):
        if not self.reachable:
            raise ComfyUnreachable("fake: 连不上 ComfyUI")

    def upload_image(self, path, subfolder=""):
        self._up()
        self.uploaded.append(Path(path).as_posix())
        return f"{subfolder}/{Path(path).name}" if subfolder else Path(path).name

    def submit(self, workflow):
        self._up()
        if self.reject_next_submit:
            msg, self.reject_next_submit = self.reject_next_submit, None
            raise ComfyError(f"/prompt 返回 HTTP 400:{msg}")
        self._n += 1
        pid = f"p{self._n}"
        self.submitted[pid] = workflow
        self.pending.append(pid)
        return pid

    def queue_ids(self):
        self._up()
        return list(self.running), list(self.pending)

    def history(self, prompt_id):
        self._up()
        return self.records.get(prompt_id)

    def outputs(self, record):
        return ComfyClient.outputs(self, record)

    def fetch(self, item, dest_dir):
        self._up()
        dest = Path(dest_dir)
        dest.mkdir(parents=True, exist_ok=True)
        out = dest / item["filename"]
        out.write_bytes(self.files[item["filename"]])
        return out

    # ── 测试操控 ────────────────────────────────────
    def start(self, pid):
        self.pending.remove(pid)
        self.running.append(pid)

    def finish(self, pid, filename, data=b"generated"):
        for q in (self.pending, self.running):
            if pid in q:
                q.remove(pid)
        self.files[filename] = data
        self.records[pid] = {"status": {"status_str": "success", "completed": True},
                             "outputs": {"9": {"images": [{"filename": filename, "subfolder": "", "type": "output"}]}}}

    def fail(self, pid, message):
        for q in (self.pending, self.running):
            if pid in q:
                q.remove(pid)
        self.records[pid] = {"status": {"status_str": "error", "completed": False,
                                        "messages": [["execution_error", {"exception_message": message}]]},
                             "outputs": {}}

    def vanish(self, pid):
        """任务从云机消失:不在队列、也不在 history(比如 ComfyUI 被重启过)。"""
        for q in (self.pending, self.running):
            if pid in q:
                q.remove(pid)

# -*- coding: utf-8 -*-
"""把一条 pending 任务交给云机:上传参考图 → 拼工作流 → LoRA 预检 → 提交。

任务的提示词、参数、种子在规划时已定死;这里**只做搬运**,不改任何生成参数。

结局只有三种,每种都落到任务状态上,不存在「提交了但库里没记」:
- 提交成功 → `queued`,记下 `external_id`(prompt_id)与 `host_id` —— 控制面重启后靠它接管;
- 连不上云机 → **留在 `pending`**,下一轮再试。还没交出去,不算失败;
- 其余一切(参考图丢了、标签错位、预检不过、ComfyUI 拒收)→ `failed` 并写明原因。
  这些重试也不会好,要人看。

参考图按**内容哈希分子目录**上传(`ai-short-drama/<sha256 前 16 位>/文件名`):
不同任务的产物常常同名(都叫 ComfyUI_00001_.png),按文件名传,后一张会在云机上覆盖前一张;
这台 ComfyUI 还是多个项目共用的 input 目录。
"""
from __future__ import annotations

import hashlib
import json
import sqlite3
from dataclasses import dataclass
from pathlib import Path

from tools.drama.db import Store
from tools.drama.tasks import set_task_state
from tools.gen import preflight
from tools.gen.comfy_client import ComfyError, ComfyUnreachable
from tools.gen.workflows import build_h3, build_qwen
from tools.serve.hosts import Host


@dataclass(frozen=True)
class Outcome:
    state: str                    # 任务派发后的状态:queued / pending / failed
    message: str = ""
    warnings: tuple[str, ...] = ()


def sha256_file(path: Path) -> str:
    h = hashlib.sha256()
    with Path(path).open("rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def pick_next(store: Store, host_id: int) -> sqlite3.Row | None:
    """挑下一条要派的 pending 任务。

    **按模型分组**:优先挑和这台机器上一条派发同模型的 —— 换模型要重新加载几十 GB 权重。
    同模型内、以及没有同模型任务时,都按创建先后。
    """
    pending = store.conn.execute("SELECT * FROM task WHERE state = 'pending' ORDER BY id").fetchall()
    if not pending:
        return None
    last = store.conn.execute("SELECT model FROM task WHERE host_id = ? AND external_id IS NOT NULL"
                              " ORDER BY id DESC LIMIT 1", (host_id,)).fetchone()
    if last is not None:
        for row in pending:
            if row["model"] == last["model"]:
                return row
    return pending[0]


def build_workflow(task: sqlite3.Row, image_names: list[str]) -> dict:
    p = json.loads(task["params_json"])
    if p["workflow"] == "qwen_image":
        return build_qwen(task["prompt"], image_names, task["seed"], preset=p["preset"], size=tuple(p["size"]))
    if p["workflow"] == "minimax_h3":
        if p["audio_ref"] is not None:
            raise ValueError("参考音频尚未接入(闸门 0-4:喂参考音频会复读内容)")
        return build_h3(task["prompt"], image_names, None, p["frames"], task["seed"], size=tuple(p["size"]))
    raise ValueError(f"未知工作流 {p['workflow']!r}")


def dispatch(store: Store, client, host: Host, task: sqlite3.Row, *, data_root: Path,
             run: preflight.Runner) -> Outcome:
    task_id = task["id"]

    def fail(msg: str) -> Outcome:
        set_task_state(store, task_id, "failed", error=msg)
        return Outcome("failed", msg)

    paths = [Path(data_root) / p for p in json.loads(task["params_json"])["image_refs"]]
    missing = [p.as_posix() for p in paths if not p.is_file()]
    if missing:
        return fail(f"参考图不存在:{', '.join(missing)}")
    try:
        names = [client.upload_image(p, subfolder=f"ai-short-drama/{sha256_file(p)[:16]}") for p in paths]
        wf = build_workflow(task, names)
    except ComfyUnreachable as e:
        return Outcome("pending", f"连不上云机,下轮再试:{e}")
    except ValueError as e:
        return fail(f"拼工作流失败:{e}")

    pf = preflight.check(wf, host.models_dir, run)
    if not pf.ok:
        return fail(";".join(pf.problems))

    try:
        pid = client.submit(wf)
    except ComfyUnreachable as e:
        return Outcome("pending", f"连不上云机,下轮再试:{e}", pf.warnings)
    except ComfyError as e:
        return fail(f"ComfyUI 拒收:{e}")
    set_task_state(store, task_id, "queued", external_id=pid, host_id=host.id)
    return Outcome("queued", pid, pf.warnings)
```

# -*- coding: utf-8 -*-
"""推进已交给云机的任务(queued / running / waiting_host):查进度、取产物、写回。

**只认 external_id**:这台 ComfyUI 多个项目共用,按 prompt_id 找自己的任务,不看队列长度。
这也是「接管」的全部机制 —— 控制面重启后,库里在途任务带着 external_id,下一轮照常推进,
不需要单独的恢复流程,**更不会重新提交**(重交就是白烧一次 GPU)。

判定顺序是**先看队列、再看 history**:任务跑完是先进 history、再出队列,
反过来查会在两次请求之间漏掉刚跑完的任务,把它误判成「消失」。
即便如此,闸门 0 实测 history 偶尔会滞后,所以「消失」要连续 GRACE_SEC 才判失败。

连不上云机 → `waiting_host`,不是失败;连续 24 小时连不上才放弃。

产物落在 `<data_root>/outputs/tasks/<task_id>/`,库里存相对 data_root 的路径和 sha256。
**写回只填空位**:目标字段已有产物时不覆盖 —— 新版本留在任务记录里等人挑,
否则会悄悄换掉一张首帧,而由旧首帧生成的视频还挂在那里。
"""
from __future__ import annotations

import json
import sqlite3
from collections.abc import Callable
from datetime import datetime, timedelta, timezone
from pathlib import Path

from tools.drama import repo
from tools.drama.db import Store
from tools.drama.tasks import set_task_state
from tools.gen.comfy_client import ComfyUnreachable
from tools.serve.dispatch import Outcome, sha256_file

GRACE_SEC = 120
HOST_GIVEUP = timedelta(hours=24)
IN_FLIGHT = ("queued", "running", "waiting_host")


def _utcnow() -> datetime:
    return datetime.now(timezone.utc)


class Collector:
    def __init__(self, data_root: Path, clock: Callable[[], datetime] = _utcnow):
        self.data_root = Path(data_root)
        self._clock = clock
        self._missing_since: dict[int, datetime] = {}

    def advance(self, store: Store, client, task: sqlite3.Row) -> Outcome:
        tid, pid, state = task["id"], task["external_id"], task["state"]
        try:
            running, pending = client.queue_ids()
            rec = client.history(pid)
        except ComfyUnreachable as e:
            return self._unreachable(store, task, e)

        if state == "waiting_host":           # 云机回来了:先回到在途,再按实际情况推进
            set_task_state(store, tid, "queued")
            state = "queued"

        if rec is not None:
            self._missing_since.pop(tid, None)
            return self._finish(store, client, task, rec)
        if pid in running or pid in pending:
            self._missing_since.pop(tid, None)
            if pid in running and state == "queued":
                set_task_state(store, tid, "running")
                return Outcome("running")
            return Outcome(state)

        since = self._missing_since.setdefault(tid, self._clock())
        if (self._clock() - since).total_seconds() < GRACE_SEC:
            return Outcome(state, "队列和 history 里都暂时找不到,观察中")
        self._missing_since.pop(tid, None)
        msg = f"云机上找不到任务 {pid}(不在队列也不在 history,已持续 {GRACE_SEC}s)—— ComfyUI 可能被重启过"
        set_task_state(store, tid, "failed", error=msg)
        return Outcome("failed", msg)

    def _unreachable(self, store: Store, task: sqlite3.Row, e: Exception) -> Outcome:
        task = repo.get(store, "task", task["id"])   # 本轮可能已迁移过状态,手里那行是旧的
        tid, state = task["id"], task["state"]
        if state != "waiting_host":
            set_task_state(store, tid, "waiting_host")
            return Outcome("waiting_host", str(e))
        waited = self._clock() - datetime.fromisoformat(task["state_since"])
        if waited > HOST_GIVEUP:
            msg = f"云机连续 {waited} 不可达,放弃。任务 {task['external_id']} 若仍在云机上,产物需人工核对"
            set_task_state(store, tid, "failed", error=msg)
            return Outcome("failed", msg)
        return Outcome("waiting_host", str(e))

    def _finish(self, store: Store, client, task: sqlite3.Row, rec: dict) -> Outcome:
        tid = task["id"]
        status = rec.get("status") or {}
        if status.get("status_str") == "error" or status.get("completed") is False:
            msg = f"ComfyUI 执行失败:{json.dumps(status.get('messages', []), ensure_ascii=False)[:800]}"
            set_task_state(store, tid, "failed", error=msg)
            return Outcome("failed", msg)

        p = json.loads(task["params_json"])
        items = [i for i in client.outputs(rec) if i["filename"].lower().endswith(p["expect_ext"])]
        if len(items) != 1:
            msg = f"期望恰好 1 个 {p['expect_ext']} 产物,实际 {len(items)} 个:{[i['filename'] for i in items]}"
            set_task_state(store, tid, "failed", error=msg)
            return Outcome("failed", msg)
        try:
            local = client.fetch(items[0], self.data_root / "outputs" / "tasks" / str(tid))
        except ComfyUnreachable as e:
            return self._unreachable(store, task, e)
        rel = local.relative_to(self.data_root).as_posix()

        target = repo.get(store, task["target_type"], task["target_id"])
        note = ""
        if target[p["writes"]]:
            note = f"{p['writes']} 已有产物,新版本留在任务记录里待人挑选"
        else:
            repo.update(store, task["target_type"], task["target_id"], **{p["writes"]: rel})
        set_task_state(store, tid, "done", result_path=rel, result_sha256=sha256_file(local))
        return Outcome("done", note or rel)
```

# -*- coding: utf-8 -*-
"""常驻 worker:一台云机一个。每轮做三件事 —— 保隧道、推进在途任务、空闲时派下一条。

- **每台机器同时只交一条**:ComfyUI 本来就串行跑,多交只是在它的队列里排队;
  留在我们的库里排,才能按模型分组挑、才能在交出去之前取消。这台 ComfyUI 还是多个项目共用的。
- **每轮重读 host**:SSH 端口在界面上改了,下一轮就按新端口重建隧道。
  接管来的隧道(不是本进程起的)杀不掉,如实提示人去结束它,不假装已处理。
- 没有单独的「恢复」流程:重启后库里的在途任务带着 external_id,照常推进就是接管。

用法:
    python -m tools.serve.worker --db data/drama.db --host-id 1 --data-root data --once
    python -m tools.serve.worker --db data/drama.db --host-id 1 --data-root data --loop
"""
from __future__ import annotations

import argparse
import sys
import time
from collections.abc import Callable
from datetime import datetime, timezone
from pathlib import Path

from tools.drama.db import Store
from tools.gen import preflight
from tools.gen.comfy_client import ComfyClient
from tools.serve.collect import IN_FLIGHT, Collector
from tools.serve.dispatch import dispatch, pick_next
from tools.serve.hosts import Host, get_host
from tools.serve.tunnel import TunnelManager

POLL_SEC = 10


def _ssh_run(host: Host) -> preflight.Runner:
    return preflight.ssh_runner(host.ssh_host, host.ssh_port, host.ssh_user, host.key_ref)


class Worker:
    def __init__(self, store: Store, host_id: int, *, data_root: Path,
                 client_for: Callable[[Host], object] = lambda h: ComfyClient(h.local_endpoint),
                 tunnel: TunnelManager | None = None,
                 run_for: Callable[[Host], preflight.Runner] = _ssh_run,
                 collector: Collector | None = None):
        self.store = store
        self.host_id = host_id
        self.data_root = Path(data_root)
        self.client_for = client_for
        self.tunnel = tunnel or TunnelManager()
        self.run_for = run_for
        self.collector = collector or Collector(self.data_root)
        self._last_host: Host | None = None

    def tick(self) -> list[str]:
        log: list[str] = []
        host = get_host(self.store, self.host_id)
        if self._last_host is not None and self._last_host != host:
            owned = self.tunnel.owns(host.id)
            self.tunnel.stop(host.id)
            if not owned:
                log.append("host 配置变了,但当前隧道不是本进程起的,杀不掉 —— 请手工结束旧的 ssh -L 进程")
            else:
                log.append("host 配置变了,已停掉旧隧道,按新配置重建")
        self._last_host = host

        if not self.tunnel.ensure(host):
            log.append(f"隧道未通(本机端口 {host.local_port}),在途任务按连不上处理")
        client = self.client_for(host)

        inflight = self.store.conn.execute(
            f"SELECT * FROM task WHERE host_id = ? AND state IN ({','.join('?' * len(IN_FLIGHT))}) ORDER BY id",
            (host.id, *IN_FLIGHT)).fetchall()
        for task in inflight:
            out = self.collector.advance(self.store, client, task)
            log.append(f"task {task['id']}: {task['state']} → {out.state} {out.message}".rstrip())

        busy = self.store.conn.execute(
            f"SELECT COUNT(*) FROM task WHERE host_id = ? AND state IN ({','.join('?' * len(IN_FLIGHT))})",
            (host.id, *IN_FLIGHT)).fetchone()[0]
        if busy == 0:
            task = pick_next(self.store, host.id)
            if task is not None:
                out = dispatch(self.store, client, host, task, data_root=self.data_root, run=self.run_for(host))
                log.append(f"task {task['id']}: pending → {out.state} {out.message}".rstrip())
                log += [f"task {task['id']} 告警: {w}" for w in out.warnings]
        return log


def main(argv: list[str] | None = None) -> int:
    ap = argparse.ArgumentParser(description="AI 短剧生成 worker(一台云机一个)")
    ap.add_argument("--db", required=True)
    ap.add_argument("--host-id", type=int, required=True)
    ap.add_argument("--data-root", required=True, help="素材与产物的根目录;库里的路径都相对它")
    mode = ap.add_mutually_exclusive_group(required=True)
    mode.add_argument("--once", action="store_true", help="只跑一轮")
    mode.add_argument("--loop", action="store_true", help=f"常驻,每 {POLL_SEC}s 一轮")
    args = ap.parse_args(argv)

    data_root = Path(args.data_root)
    journal = data_root / "journal" / "worker.log"
    journal.parent.mkdir(parents=True, exist_ok=True)
    worker = Worker(Store(args.db), args.host_id, data_root=data_root)
    while True:
        stamp = datetime.now(timezone.utc).isoformat(timespec="seconds")
        lines = [f"{stamp} host={args.host_id} {line}" for line in worker.tick()]
        if lines:
            print("\n".join(lines), flush=True)
            with journal.open("a", encoding="utf-8") as f:
                f.write("\n".join(lines) + "\n")
        if args.once:
            return 0
        time.sleep(POLL_SEC)


if __name__ == "__main__":
    sys.exit(main())
```

        --scene outputs/gate0/0-3-scene_baseline_20260923-094310/0-3-scene_00001_.png \\
        --prop outputs/gate0/0-3-prop_baseline_20260923-094422/0-3-prop_00001_.png
"""
from __future__ import annotations

import argparse

# 溯源必须齐的字段。started_at 不在内:4 步出图可能在两次轮询之间就跑完,没被看到 running,
# 如实留空,而不是编一个时间。
PROVENANCE = ("model", "params_json", "seed", "prompt", "refs_json", "created_at", "host_id",
              "external_id", "result_path", "result_sha256", "finished_at")


def build_drama(store: Store, assets: dict[str, str]) -> int:
    d = repo.create_drama(store, "玉佩", "9:16")
    ep = repo.add_episode(store, d, 1, "script", script="林小雨在老宅庭院里回忆母亲。")
    lin = repo.add_character(store, d, "林小雨", turnaround_path=assets["character"], voice_desc="年轻女声,清亮,语速偏慢")
    yard = repo.add_scene(store, d, "老宅庭院", image_path=assets["scene"])
    jade = repo.add_prop(store, d, "玉佩", image_path=assets["prop"])
    seg = repo.add_segment(
        store, ep, 1, scene_id=yard, duration_sec=8.0, reference_mode="first",
        description="【镜头1】林小雨低头看着手里的玉佩。【镜头2】林小雨说：「这块玉佩，是我母亲留给我的。」",
        image_prompt="@林小雨 站在黄昏的 @老宅庭院 里,双手捧着 @玉佩,中景,竖构图",
        video_prompt="@林小雨 在 @老宅庭院 低头看 @玉佩,然后抬眼说话,固定机位,环境声,无配乐")
    repo.bind_characters(store, seg, [lin])
    repo.bind_props(store, seg, [jade])
    return seg


def run_until_final(worker: Worker, store: Store, task_id: int, *, timeout_sec: float,
                    sleep: Callable[[float], None] = time.sleep, clock: Callable[[], float] = time.monotonic) -> str:
    deadline = clock() + timeout_sec
    while True:
        for line in worker.tick():
            print(line, flush=True)
        row = repo.get(store, "task", task_id)
        if row["state"] in ("done", "failed"):
            return row["state"]
        if clock() > deadline:
            raise TimeoutError(f"task {task_id} {timeout_sec}s 内没结束,停在 {row['state']} —— "
                               f"它可能还在云机上跑,external_id={row['external_id']},不要重新提交")
        sleep(POLL_SEC)


def provenance(store: Store, task_id: int) -> dict:
    row = repo.get(store, "task", task_id)
    missing = [k for k in PROVENANCE if row[k] in (None, "")]
    if missing:
        raise AssertionError(f"task {task_id} 溯源字段缺失:{missing}")
    return {k: row[k] for k in row.keys()}


def run(data_root: Path, images: dict[str, Path], host: dict, *, seed: int, timeout_sec: float,
        make_worker: Callable[[Store, int, Path], Worker] = lambda s, h, root: Worker(s, h, data_root=root),
        sleep: Callable[[float], None] = time.sleep) -> dict:
    if data_root.exists() and any(data_root.iterdir()):
        raise FileExistsError(f"{data_root} 不是空目录 —— 验收要在全新目录里跑,不碰已有数据")
    (data_root / "assets").mkdir(parents=True)
    assets = {}
    for kind, src in images.items():
        dst = data_root / "assets" / f"{kind}{Path(src).suffix}"
        shutil.copyfile(src, dst)
        assets[kind] = dst.relative_to(data_root).as_posix()

    store = Store(data_root / "drama.db")
    seg = build_drama(store, assets)
    host_id = add_host(store, **host)
    worker = make_worker(store, host_id, data_root)

    report = {"segment": seg, "tasks": []}
    for plan, task_seed in ((plan_first_frame, seed), (plan_video, seed + 1)):
        tid = plan(store, seg, seed=task_seed)
        state = run_until_final(worker, store, tid, timeout_sec=timeout_sec, sleep=sleep)
        if state == "failed":
            raise RuntimeError(f"task {tid} 失败:{repo.get(store, 'task', tid)['error']}")
        report["tasks"].append(provenance(store, tid))
    s = repo.get(store, "segment", seg)
    report["first_frame_path"], report["video_path"] = s["first_frame_path"], s["video_path"]
    (data_root / "provenance.json").write_text(json.dumps(report, ensure_ascii=False, indent=2), encoding="utf-8")
    return report


def main(argv: list[str] | None = None) -> int:
    ap = argparse.ArgumentParser(description="P2 真链路验收:一个段落出首帧 + 视频")
    ap.add_argument("--data-root", required=True, type=Path)
    ap.add_argument("--ssh-host", required=True)
    ap.add_argument("--ssh-port", required=True, type=int)
    ap.add_argument("--ssh-user", default="root")
    ap.add_argument("--key-ref", required=True, help="私钥文件路径(只存路径,不读内容)")
    ap.add_argument("--local-port", type=int, default=8188)
    ap.add_argument("--vram-gb", type=float, default=24.0)
    ap.add_argument("--models-dir", required=True)
    ap.add_argument("--character", required=True, type=Path)
    ap.add_argument("--scene", required=True, type=Path)
    ap.add_argument("--prop", required=True, type=Path)
    ap.add_argument("--seed", type=int, default=20260923)
    ap.add_argument("--timeout-min", type=float, default=60)
    a = ap.parse_args(argv)

    host = {"label": "e2e", "ssh_host": a.ssh_host, "ssh_port": a.ssh_port, "ssh_user": a.ssh_user,
            "key_ref": a.key_ref, "local_port": a.local_port, "vram_gb": a.vram_gb, "models_dir": a.models_dir}
    report = run(a.data_root, {"character": a.character, "scene": a.scene, "prop": a.prop}, host,
                 seed=a.seed, timeout_sec=a.timeout_min * 60)
    print(f"首帧:{a.data_root / report['first_frame_path']}")
    print(f"视频:{a.data_root / report['video_path']}")
    print(f"溯源:{a.data_root / 'provenance.json'}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
