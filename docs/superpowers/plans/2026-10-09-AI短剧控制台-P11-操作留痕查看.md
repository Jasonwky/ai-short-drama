    more: int            # 符合条件、但超出 limit 没显示的条数


def read(data_root: Path, *, since: str = "", until: str = "", prefixes: list[str] | None = None,
         mentions: tuple[str, int] | None = None, limit: int = 500) -> Read:
    """**逐行流式读整个文件**再过滤 —— 只读最后若干行的话,更早的时间段永远筛不到。
    since / until 是 UTC ISO 时间(与行首时间戳同格式,按字符串比较);prefixes 匹配时间戳后面的前缀
    (`host=` / `console` / `workflows` / `runtime`);mentions=("task", 12) 只要提到这条的行(整词,`task 1` 不匹配 `task 12`)。"""
    path = Path(data_root) / LOG
    if not path.exists():
        return Read([], 0)
    hit = re.compile(rf"(?<![\w=]){mentions[0]} {mentions[1]}(?!\d)") if mentions else None
    keep: deque[str] = deque(maxlen=limit)
    total = 0
    with path.open(encoding="utf-8") as f:
        for line in f:
            line = line.rstrip("\n")
            stamp, _, rest = line.partition(" ")
            if (since and stamp < since) or (until and stamp > until):
                continue
            if prefixes and not any(rest.startswith(p) for p in prefixes):
                continue
            if hit and not hit.search(rest):
                continue
            keep.append(line)
            total += 1
    return Read(list(reversed(keep)), total - len(keep))


_LINK = re.compile(r"(?<![\w=])(task|llm_run|export) (\d+)(?!\d)")
_HREF = {"task": "/task/{}", "llm_run": "/llm/run/{}", "export": "/trace/export/{}"}


def links(line: str) -> list[tuple[str, str | None]]:
    """把一行切成 (文字, 链接或 None) 的片段:`task 12` / `llm_run 3` / `export 2` 变成链接。模板逐段渲染,不拼 HTML。"""
    out, pos = [], 0
    for m in _LINK.finditer(line):
        if m.start() > pos:
            out.append((line[pos:m.start()], None))
        out.append((m.group(0), _HREF[m.group(1)].format(m.group(2))))
        pos = m.end()
    if pos < len(line):
        out.append((line[pos:], None))
    return out
```

    """按「路径 + 修改时间 + 大小」缓存:倒查整集要算十几段视频(每段几十 MB);文件一变,三者跟着变,缓存自然失效。"""
    st = f.stat()
    key = (str(f.resolve()), st.st_mtime_ns, st.st_size)
    if key not in _SHA:
        _SHA[key] = sha256_file(f)
    return _SHA[key]


def node(store: Store, root: Path, path: str) -> dict:
    """一个产物往回追的树(导出的溯源报告与控制台的倒查页共用)。"""
    f = root / path
    out = {"path": path, "sha256": cached_sha256(f) if f.is_file() else None, "task": None, "inputs": []}
    t = _task_for(store, path)
    if t is None:
        return out
    if t["host_id"] is not None:
        host = repo.get(store, "host", t["host_id"])["label"]
    else:   # 本地确定性生成没有 external_id;有 external_id 却没有 host 的,是从别处导入的工程
        host = "本机" if t["external_id"] is None else "未登记的机器(导入的工程)"
    out["task"] = {"id": t["id"], "kind": t["kind"], "model": t["model"], "params": json.loads(t["params_json"]),
                   "seed": t["seed"], "prompt": t["prompt"], "host": host, "external_id": t["external_id"],
                   "created_at": t["created_at"], "finished_at": t["finished_at"],
                   "result_sha256": t["result_sha256"]}
    out["inputs"] = [node(store, root, r) for r in json.loads(t["refs_json"])]   # 参考素材都早于产物,链无环
    return out

# -*- coding: utf-8 -*-
"""留痕:时间线、事件归属、失败归类(设计 specs/2026-10-09-操作留痕查看-design.md §三)。**只读**。

时间线以库里四张表为主(task / llm_run / export / workflow_preset,都带时间戳与目标):
- **只用表里已有的时间戳,不编时间** —— 开跑时刻为空就不出「开跑」,LLM 失败没有单独时间戳就记在发起时刻;

"""
from __future__ import annotations

import json
import re
from dataclasses import dataclass, field

from tools.drama.db import Store

KINDS = {"task": "生成任务", "llm": "LLM", "export": "成片导出", "preset": "模板预设"}

PER_PAGE = 100


@dataclass(frozen=True)
class Event:
    at: str                  # UTC ISO,与库里一致
    kind: str                # task / llm / export / preset
    ref: int                 # 该表里的 id
    action: str
    label: str
    href: str
    drama_id: int | None = None
    episode_id: int | None = None
    host_id: int | None = None
    error: str = ""
    is_global: bool = False
    unattributed: bool = False

    @property
    def failed(self) -> bool:
        return self.action == "失败"


@dataclass(frozen=True)
class Filter:
    drama_id: int | None = None
    episode_id: int | None = None
    host_id: int | None = None
    kinds: tuple[str, ...] = ()      # 空 = 全部
    since: str = ""                  # UTC ISO
    until: str = ""

# ── 失败归类 ─────────────────────────────────────────────────────────

_WIN_PATH = re.compile(r"[A-Za-z]:[\\/][^\s,;)」]*")
_PATH = re.compile(r"(?:[\w.-]*/)+[\w.-]+")
_NUM = re.compile(r"\d+(?:\.\d+)?")


def failure_key(error: str | None) -> str:
    """**先**换掉路径(含 Windows 盘符路径,否则会在盘符的冒号处截断)、数字,**再**在第一个冒号或「——」处截断。"""
    text = (error or "").strip()
    if not text:
        return "(没有写原因)"
    text = _WIN_PATH.sub("<路径>", text)
    text = _PATH.sub("<路径>", text)
    text = _NUM.sub("#", text)
    return re.split(r"[:：]|——", text, maxsplit=1)[0].strip()


@dataclass(frozen=True)
class FailureGroup:
    key: str
    events: list[Event] = field(default_factory=list)     # 新的在前

    @property
    def count(self) -> int:
        return len(self.events)

    @property
    def latest(self) -> Event:
        return self.events[0]


def failures(store: Store, flt: Filter) -> list[FailureGroup]:
    """条数多的在前;同样多的,最近出过的在前。"""
    groups: dict[str, list[Event]] = {}
    for e in _events(store, Filter(**{**flt.__dict__, "failed_only": True})):
        groups.setdefault(failure_key(e.error), []).append(e)
    out = [FailureGroup(k, v) for k, v in groups.items()]
    out.sort(key=lambda g: g.key)
    out.sort(key=lambda g: g.latest.at, reverse=True)
    out.sort(key=lambda g: g.count, reverse=True)
    return out


# ── 归属 ───────────────────────────────────────────────────────────

_ASSET_PATHS = (("character", ("turnaround_path", "fourview_path")), ("scene", ("image_path",)), ("prop", ("image_path",)))


def _target_owner(store: Store, target_type: str, target_id: int) -> tuple[int | None, int | None] | None:
    """(剧, 集);目标不在了返回 None。"""
    if target_type == "drama":
        return (target_id, None) if store.conn.execute("SELECT 1 FROM drama WHERE id = ?", (target_id,)).fetchone() \
            else None
    if target_type == "episode":
        row = store.conn.execute("SELECT drama_id FROM episode WHERE id = ?", (target_id,)).fetchone()
        return (row[0], target_id) if row else None
    if target_type == "segment":
        row = store.conn.execute("SELECT e.drama_id, e.id FROM segment s JOIN episode e ON e.id = s.episode_id"
                                 " WHERE s.id = ?", (target_id,)).fetchone()
        return (row[0], row[1]) if row else None
    row = store.conn.execute(f"SELECT drama_id FROM {target_type} WHERE id = ?", (target_id,)).fetchone()
    return (row[0], None) if row else None


def _drama_of_path(store: Store, path: str, seen: set[str]) -> int | None:
    """一个参考素材属于哪部剧:是某个资产当前 / 历史的图,或由某条能定剧的任务产出。"""
    if path in seen:
        return None
    seen.add(path)
    for table, cols in _ASSET_PATHS:
        for col in cols:
            row = store.conn.execute(f"SELECT drama_id FROM {table} WHERE {col} = ?", (path,)).fetchone()
            if row:
                return row[0]
    for t in store.conn.execute("SELECT * FROM task WHERE result_path = ? ORDER BY id DESC", (path,)):
        owner = _target_owner(store, t["target_type"], t["target_id"])
        if owner:
            return owner[0]
        for ref in json.loads(t["refs_json"]):
            d = _drama_of_path(store, ref, seen)
            if d is not None:
                return d
    return None


def attribute(store: Store, kind: str, ref: int) -> tuple[int | None, int | None]:
    """事件属于哪部剧 / 哪一集。段落被删的任务顺着参考图定剧,集定不了;追不到 = (None, None)。"""
    if kind == "task":
        t = store.conn.execute("SELECT * FROM task WHERE id = ?", (ref,)).fetchone()
        owner = _target_owner(store, t["target_type"], t["target_id"])
        if owner:
            return owner
        seen: set[str] = set()
        for path in json.loads(t["refs_json"]):
            d = _drama_of_path(store, path, seen)
            if d is not None:
                return d, None
        return None, None
    if kind == "llm":
        r = store.conn.execute("SELECT target_type, target_id FROM llm_run WHERE id = ?", (ref,)).fetchone()
        return _target_owner(store, r["target_type"], r["target_id"]) or (None, None)
    if kind == "export":
        r = store.conn.execute("SELECT episode_id FROM export WHERE id = ?", (ref,)).fetchone()
        return _target_owner(store, "episode", r["episode_id"]) or (None, None)
    return None, None

def _target_label(store: Store, target_type: str, target_id: int) -> str:
    if target_type == "segment":
        row = store.conn.execute("SELECT s.number, e.number FROM segment s JOIN episode e ON e.id = s.episode_id"
                                 " WHERE s.id = ?", (target_id,)).fetchone()
        return f"第 {row[1]} 集段落 {row[0]}" if row else f"段落 #{target_id}(已删)"
    if target_type == "episode":
        row = store.conn.execute("SELECT number FROM episode WHERE id = ?", (target_id,)).fetchone()
        return f"第 {row[0]} 集" if row else f"集 #{target_id}"
    if target_type == "drama":
        row = store.conn.execute("SELECT title FROM drama WHERE id = ?", (target_id,)).fetchone()
        return f"《{row[0]}》" if row else f"剧 #{target_id}"
    names = {"character": "角色", "scene": "场景", "prop": "道具"}
    row = store.conn.execute(f"SELECT name FROM {target_type} WHERE id = ?", (target_id,)).fetchone()
    return f"{names[target_type]}「{row[0]}」" if row else f"{names[target_type]} #{target_id}"


def _events(store: Store, flt: Filter) -> list[Event]:
    out: list[Event] = []
    kinds = set(flt.kinds) or set(KINDS)
    if flt.host_id is not None:
        kinds &= {"task"}                     # 只有任务经云机
    owners: dict[tuple[str, int], tuple[int | None, int | None]] = {}

    def owner(kind: str, ref: int):
        if (kind, ref) not in owners:
            owners[(kind, ref)] = attribute(store, kind, ref)
        return owners[(kind, ref)]

    def add(at, kind, ref, action, label, href, error="", host_id=None, is_global=False):
        if not at:
            return
        d, ep = (None, None) if is_global else owner(kind, ref)
        out.append(Event(at, kind, ref, action, label, href, d, ep, host_id, error, is_global,
                         unattributed=not is_global and d is None))

    if "task" in kinds:
        for t in store.conn.execute("SELECT * FROM task"):
            label = f"task {t['id']} · {TASK_KIND[t['kind']]} · {_target_label(store, t['target_type'], t['target_id'])}"
            href, host = f"/task/{t['id']}", t["host_id"]
            if t["state"] == "done" and t["external_id"] is None and t["created_at"] == t["finished_at"]:
                upload = json.loads(t["params_json"]).get("workflow") == "upload"
                add(t["created_at"], "task", t["id"], "上传" if upload else "本机生成", label, href, host_id=host)
                continue
            add(t["created_at"], "task", t["id"], "创建", label, href, host_id=host)
            add(t["started_at"], "task", t["id"], "开跑", label, href, host_id=host)
            if t["state"] in ("done", "failed"):
                add(t["finished_at"], "task", t["id"], "完成" if t["state"] == "done" else "失败", label, href,
                    error=t["error"] or "", host_id=host)
    if "llm" in kinds:
        for r in store.conn.execute("SELECT * FROM llm_run"):
            label = f"llm_run {r['id']} · {r['task']} · {_target_label(store, r['target_type'], r['target_id'])}"
            href = f"/llm/run/{r['id']}"
            if r["state"] == "failed":       # 失败没有单独的时间戳:记在发起时刻,附原因
                add(r["created_at"], "llm", r["id"], "失败", label, href, error=r["error"] or "")
                continue
            add(r["created_at"], "llm", r["id"], "发起", label, href)
            if r["state"] in ("approved", "rejected"):
                add(r["decided_at"], "llm", r["id"], "批准" if r["state"] == "approved" else "驳回", label, href)
    if "export" in kinds:
        for x in store.conn.execute("SELECT * FROM export"):
            label = f"export {x['id']} · {_target_label(store, 'episode', x['episode_id'])} · {x['produce_id']}"
            href = f"/trace/export/{x['id']}"
            add(x["created_at"], "export", x["id"], "开始", label, href)
            if x["state"] in ("done", "failed"):
                add(x["finished_at"], "export", x["id"], "完成" if x["state"] == "done" else "失败", label, href,
                    error=x["error"] or "")
    if "preset" in kinds:
        for p in store.conn.execute("SELECT * FROM workflow_preset"):
            add(p["created_at"], "preset", p["id"], "保存", f"模板预设 {p['category']} v{p['version']}",
                "/workflows", is_global=True)

    def keep(e: Event) -> bool:
        if flt.failed_only and not e.failed:
            return False
        if flt.since and e.at < flt.since or flt.until and e.at > flt.until:
            return False
        if flt.host_id is not None and e.host_id != flt.host_id:
            return False
        if flt.episode_id is not None:
            return e.episode_id == flt.episode_id
        if flt.drama_id is not None:
            return e.is_global or e.drama_id == flt.drama_id
        return True

    out = [e for e in out if keep(e)]
    out.sort(key=lambda e: (e.at, e.kind, e.ref), reverse=True)
    return out


def timeline(store: Store, flt: Filter, page: int = 1) -> tuple[list[Event], int]:
    """按时间倒序,每页 100 条;返回 (这一页, 总条数)。"""
    events = _events(store, flt)
    start = (max(page, 1) - 1) * PER_PAGE
    return events[start:start + PER_PAGE], len(events)

# -*- coding: utf-8 -*-
"""留痕:倒查树、LLM 来源、同目标的其它尝试、任务详情、成片快照对比(设计 specs/2026-10-09-操作留痕查看-design.md)。**只读**。

- 倒查树复用溯源报告的 `provenance.node`(导出与控制台同一棵树),每个节点再标:内容核对、目标、LLM 来源、其它尝试;
- **能用数据证明的不推断**:视频提示词里逐字写着「<角色>的声音:<描述>」,批准时写回的是 `voices.compose(输出)` ——
  对上 = 「确证」;其余按时间推断(只取任务创建之前批准的),页面上标明;
- 成片不重新追:展示导出时的溯源报告(快照、备案材料),再标出之后换过的。
"""
from __future__ import annotations

import json
import sqlite3
from pathlib import Path

from tools.cut import provenance
from tools.drama.db import Store
from tools.llm import voices
from tools.serve import journal, trace
from tools.serve.dispatch import SUBMITTED

_SEGMENT_SOURCES = (("episode", ("storyboard", "script_synopsis", "script_adapt")), ("drama", ("outline",)))


def _inside(root: Path, rel: str) -> None:
    base = Path(root).resolve()
    if base not in (base / rel).resolve().parents:
        raise KeyError(f"{rel} 不在素材根里")


def _latest_approved(store: Store, task: str, target_type: str, target_id: int, before: str):
    return store.conn.execute(
        "SELECT * FROM llm_run WHERE task = ? AND target_type = ? AND target_id = ? AND state = 'approved'"
        " AND decided_at <= ? ORDER BY decided_at DESC, id DESC LIMIT 1",
        (task, target_type, target_id, before)).fetchone()


def _source(run: sqlite3.Row, how: str) -> dict:
    return {"run_id": run["id"], "task": run["task"], "how": how, "decided_at": run["decided_at"],
            "href": f"/llm/run/{run['id']}"}


def llm_sources(store: Store, task: sqlite3.Row) -> list[dict]:
    """这条任务用到的文字出自哪次 LLM 运行。只取任务创建之前批准的;驳回 / 失败的不算。"""
    before, out = task["created_at"], []
    if task["target_type"] == "segment":
        drama_id, episode_id = trace.attribute(store, "task", task["id"])
        for level, names in _SEGMENT_SOURCES:
            target = episode_id if level == "episode" else drama_id
            if target is None:
                continue                                   # 段落被删:集定不了,不猜
            for name in names:
                run = _latest_approved(store, name, level, target, before)
                if run:
                    out.append(_source(run, "按时间推断"))
        if task["kind"] == "video" and drama_id is not None:
            for ch in store.conn.execute("SELECT id, name FROM character WHERE drama_id = ?", (drama_id,)):
                if f"{ch['name']}的声音:" not in task["prompt"]:      # 提示词里写着谁的声音,就是谁(不看当前绑定)
                    continue
                run = _latest_approved(store, "voice", "character", ch["id"], before)
                if run:
                    proven = voices.compose(json.loads(run["output_json"])) in task["prompt"]
                    out.append(_source(run, "确证" if proven else "按时间推断"))
    else:                                                  # 图片:自己的反推描述(声音只进视频提示词,与图片无关)
        run = _latest_approved(store, f"describe_{task['target_type']}", task["target_type"], task["target_id"], before)
        if run:
            out.append(_source(run, "按时间推断"))
    return out


def siblings(store: Store, task: sqlite3.Row) -> list[dict]:
    """同一目标、同一写回字段的其它尝试:失败的、被换下的旧版本、在途的。"""
    writes = json.loads(task["params_json"]).get("writes")
    table = task["target_type"]
    current = store.conn.execute(f"SELECT {writes} FROM {table} WHERE id = ?", (task["target_id"],)).fetchone() \
        if writes else None
    out = []
    for t in store.conn.execute("SELECT * FROM task WHERE target_type = ? AND target_id = ? AND id != ? ORDER BY id DESC",
                                (table, task["target_id"], task["id"])):
        if json.loads(t["params_json"]).get("writes") != writes:
            continue
        if t["state"] == "failed":
            note = "失败"
        elif t["state"] != "done":
            note = "在途"
        elif current is not None and t["result_path"] == current[0]:
            note = "当前版"
        else:
            note = "旧版本"
        out.append({"id": t["id"], "note": note, "error": t["error"] or "", "result_path": t["result_path"],
                    "created_at": t["created_at"]})
    return out


def _annotate(store: Store, node: dict) -> dict:
    t = node["task"]
    if node["sha256"] is None:
        node["status"] = "missing"
    elif t is None:
        node["status"] = "manual"
    elif t["result_sha256"] and node["sha256"] != t["result_sha256"]:
        node["status"] = "mismatch"
    else:
        node["status"] = "ok"
    if t is not None:
        row = store.conn.execute("SELECT * FROM task WHERE id = ?", (t["id"],)).fetchone()
        node["target"] = trace._target_label(store, row["target_type"], row["target_id"])
        node["started_at"] = row["started_at"]
        node["llm"] = llm_sources(store, row)
        node["siblings"] = siblings(store, row)
    node["inputs"] = [_annotate(store, i) for i in node["inputs"]]
    return node


def product(store: Store, root: Path, path: str) -> dict:
    """一个产物往回追的树,每个节点标:ok / mismatch(文件被替换过)/ missing(不在本机)/ manual(没有生成记录)。"""
    _inside(root, path)
    return _annotate(store, provenance.node(store, Path(root), path))


def export_for_path(store: Store, path: str) -> int | None:
    row = store.conn.execute("SELECT id FROM export WHERE labeled_path = ? OR plain_path = ?", (path, path)).fetchone()
    return row[0] if row else None


def task_view(store: Store, root: Path, task_id: int) -> dict:
    row = store.conn.execute("SELECT * FROM task WHERE id = ?", (task_id,)).fetchone()
    if row is None:
        raise KeyError(f"task {task_id} 不存在")
    sub = Path(root) / "outputs" / "tasks" / str(task_id) / SUBMITTED
    return {"row": row, "params": json.loads(row["params_json"]),
            "refs": [{"path": r, "exists": (Path(root) / r).is_file()} for r in json.loads(row["refs_json"])],
            "submitted": sub.read_text(encoding="utf-8") if sub.is_file() else None,
            "target": trace._target_label(store, row["target_type"], row["target_id"]),
            "llm": llm_sources(store, row), "siblings": siblings(store, row),
            "journal": journal.read(root, mentions=("task", task_id))}


def _walk(node: dict):
    yield node
    for i in node["inputs"]:
        yield from _walk(i)


def export_view(store: Store, root: Path, export_id: int) -> dict:
    """导出时的溯源报告 + 之后换过的:①逐段比报告里的视频 / 字幕路径与段落当前的;②每个节点比报告记的 sha256 与现在的文件。"""
    row = store.conn.execute("SELECT * FROM export WHERE id = ?", (export_id,)).fetchone()
    if row is None:
        raise KeyError(f"export {export_id} 不存在")
    rep = Path(root) / row["report_path"] if row["report_path"] else None
    if rep is None or not rep.is_file():
        return {"row": row, "report": None, "changed": []}
    report = json.loads(rep.read_text(encoding="utf-8"))
    changed: list[str] = []
    for s in report["segments"]:
        seg = store.conn.execute("SELECT s.* FROM segment s JOIN export x ON x.episode_id = s.episode_id"
                                 " WHERE x.id = ? AND s.number = ?", (export_id, s["number"])).fetchone()
        for field, key, what in (("video_path", "video", "视频"), ("subtitle_path", "subtitles", "字幕")):
            then = s[key]["path"] if s.get(key) else None
            if seg is None:
                changed.append(f"段落 {s['number']} 已删")
                break
            if then and seg[field] != then:
                changed.append(f"段落 {s['number']} 的{what}之后换成了 {seg[field] or '(空)'}")
        for n in (n for k in ("video", "subtitles") if s.get(k) for n in _walk(s[k])):
            f = Path(root) / n["path"]
            if n["sha256"] and (not f.is_file() or provenance.cached_sha256(f) != n["sha256"]):
                changed.append(f"{n['path']} " + ("不在本机了" if not f.is_file() else "的内容与导出时不同"))
    return {"row": row, "report": report, "changed": list(dict.fromkeys(changed))}
```

# -*- coding: utf-8 -*-
"""留痕:失败归类 · 时间线 · 原始流水(面板),从产物倒查,任务详情(设计 specs/2026-10-09-操作留痕查看-design.md)。

全部只读。库与日志里是 UTC,页面按北京时间显示、悬停看原始字符串;筛选框填北京时间,这里换回 UTC 再查。
"""
from __future__ import annotations

from datetime import datetime, timedelta, timezone
from urllib.parse import urlencode

from fastapi import APIRouter, Request
from fastapi.responses import RedirectResponse

from tools.serve import journal, trace, trace_views
from tools.serve.panels.common import TEMPLATES, render, rt

router = APIRouter()
BJ = timezone(timedelta(hours=8))
PREFIXES = {"": "全部", "host=": "worker(各云机)", "console ": "控制台", "workflows ": "工作流", "runtime ": "后台"}


def bj(utc_iso: str | None) -> str:
    if not utc_iso:
        return ""
    return datetime.fromisoformat(utc_iso).astimezone(BJ).strftime("%m-%d %H:%M:%S")


def _utc(local: str) -> str:
    """筛选框里的北京时间(`2026-10-09T12:00`)→ 与库里同格式的 UTC 字符串;空的原样。"""
    if not local:
        return ""
    return datetime.fromisoformat(local).replace(tzinfo=BJ).astimezone(timezone.utc).isoformat(timespec="seconds")


TEMPLATES.env.filters["bj"] = bj
TEMPLATES.env.globals["journal_links"] = journal.links


def _int(v: str | None) -> int | None:
    return int(v) if v and v.isdigit() else None


@router.get("/trace")
def page(request: Request):
    q = request.query_params
    runtime = rt(request)
    flt = trace.Filter(drama_id=_int(q.get("d")), episode_id=_int(q.get("ep")), host_id=_int(q.get("host")),
                       kinds=tuple(k for k in q.getlist("kind") if k in trace.KINDS), since=_utc(q.get("since", "")),
                       until=_utc(q.get("until", "")), failed_only=q.get("failed") == "1")
    page_no = _int(q.get("page")) or 1
    events, total = trace.timeline(runtime.store, flt, page=page_no)
    prefix = q.get("prefix", "")
    raw = journal.read(runtime.data_root, since=flt.since, until=flt.until,
                       prefixes=[prefix] if prefix in PREFIXES and prefix else None)
    hosts = runtime.store.conn.execute("SELECT id, label FROM host ORDER BY id").fetchall()
    return render(request, "trace.html", "trace", groups=trace.failures(runtime.store, flt), events=events,
                  total=total, page=page_no, pages=max(1, -(-total // trace.PER_PAGE)), q=q, flt=flt,
                  qs=urlencode([(k, v) for k, v in q.multi_items() if k != "page"]),
                  kinds=trace.KINDS, hosts=hosts, raw=raw, prefixes=PREFIXES, prefix=prefix)


@router.get("/trace/product")
def product(request: Request, path: str):
    runtime = rt(request)
    eid = trace_views.export_for_path(runtime.store, path)
    if eid is not None:                    # 成片:看导出时的快照,不重新追
        return RedirectResponse(f"/trace/export/{eid}", status_code=303)
    return render(request, "trace_product.html", "trace", tree=trace_views.product(runtime.store, runtime.data_root, path))
