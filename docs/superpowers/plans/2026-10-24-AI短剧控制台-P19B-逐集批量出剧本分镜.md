`tools/llm/llm_queue.py`:

```python
# -*- coding: utf-8 -*-
"""持久的 LLM 任务队列(spec 2026-10-24-逐集批量出剧本分镜-design.md 第 4、8 节)。

只管「排了什么、排到哪」这份数据;什么时候开下一条由 `serve/runtime.Runtime.pump_queue` 决定(加锁,一次只跑一条)。
- 重启后 `waiting` 的接着跑;重启时正在跑的那条由 `runs.fail_stale` 标失败,队列项保持 `started`,不自动重试;
- 暂停原因存 `llm_queue_state`(单行);暂停时不出队,人点「继续」才清掉。
"""
from __future__ import annotations

import json
import sqlite3

from tools.drama.db import Store, now

BREAK_AFTER = 3        # 连续几条队列项因 LLM 本身(不可用 / 连不上 / 截断 / 拒答)失败就暂停整个队列


def enqueue(store: Store, task: str, target_type: str, target_id: int, params: dict) -> int:
    with store.write() as c:
        return c.execute("INSERT INTO llm_queue (task, target_type, target_id, params_json, created_at)"
                         " VALUES (?, ?, ?, ?, ?)",
                         (task, target_type, target_id, json.dumps(params, ensure_ascii=False), now())).lastrowid


def params(row: sqlite3.Row) -> dict:
    return json.loads(row["params_json"])


def next_waiting(store: Store) -> sqlite3.Row | None:
    return store.conn.execute("SELECT * FROM llm_queue WHERE state = 'waiting' ORDER BY id LIMIT 1").fetchone()


def running_item(store: Store) -> sqlite3.Row | None:
    """已开始、而且它的 LLM 运行还在跑的那一条(一次最多一条)。"""
    return store.conn.execute("SELECT q.* FROM llm_queue q JOIN llm_run r ON r.id = q.run_id"
                              " WHERE q.state = 'started' AND r.state = 'running' LIMIT 1").fetchone()


def item_for_run(store: Store, run_id: int) -> sqlite3.Row | None:
    return store.conn.execute("SELECT * FROM llm_queue WHERE run_id = ?", (run_id,)).fetchone()


def mark_started(store: Store, item_id: int, run_id: int | None, reason: str = "") -> None:
    """开了运行(run_id),或当场开不了(run_id 为 None,reason 写原因)。"""
    with store.write() as c:
        c.execute("UPDATE llm_queue SET state = 'started', run_id = ?, reason = ?, started_at = ?"
                  " WHERE id = ? AND state = 'waiting'", (run_id, reason, now(), item_id))


def position(store: Store, item_id: int) -> int:
    return store.conn.execute("SELECT COUNT(*) FROM llm_queue WHERE state = 'waiting' AND id <= ?",
                              (item_id,)).fetchone()[0]


def waiting_count(store: Store, episode_ids: list[int]) -> int:
    if not episode_ids:
        return 0
    marks = ",".join("?" * len(episode_ids))
    return store.conn.execute(f"SELECT COUNT(*) FROM llm_queue WHERE state = 'waiting' AND target_type = 'episode'"
                              f" AND target_id IN ({marks})", episode_ids).fetchone()[0]


def cancel_waiting(store: Store, episode_ids: list[int]) -> int:
    """停掉这些集还没开始的;正在跑的那条不打断。"""
    if not episode_ids:
        return 0
    marks = ",".join("?" * len(episode_ids))
    with store.write() as c:
        return c.execute(f"UPDATE llm_queue SET state = 'cancelled' WHERE state = 'waiting' AND target_type = 'episode'"
                         f" AND target_id IN ({marks})", episode_ids).rowcount


def paused(store: Store) -> str:
    return store.conn.execute("SELECT paused_reason FROM llm_queue_state WHERE id = 1").fetchone()[0]


def pause(store: Store, reason: str) -> None:
    with store.write() as c:
        c.execute("UPDATE llm_queue_state SET paused_reason = ? WHERE id = 1", (reason,))


def resume(store: Store) -> None:
    pause(store, "")
```

# -*- coding: utf-8 -*-
"""逐集批量出剧本 / 分镜(spec 2026-10-24-逐集批量出剧本分镜-design.md 第 2、5、8、9 节)。

- 每集用哪个入口由代码定:小说集(原文 + 大纲)→ `script_adapt`;只有大纲 → `script_outline`;都没有 → blocked;
- 两态(照搬 `gen/fill.py`):`plan` 只算不改;`confirm` 只认确认页列出的集,逐集再判一次,状态变了的跳过;
- 去重:队列里已有、正在生成、有待审的都算在途,不重复排;
- 剧本批准后:这集没段落、剧本里的角色都在资产里 → 排「拆分镜」;缺角色就不排,面板写明缺什么
  (场景只在一个都没有时拦;地名与资产名对不上只提示 —— 拆分镜会从已有场景里挑最接近的)。
"""
from __future__ import annotations

import math
import re
from dataclasses import dataclass

from tools.drama import repo, rules
from tools.drama.db import Store
from tools.llm import llm_queue, runs, script_writer, service

SCRIPT_TASKS = ("script_synopsis", "script_adapt", "script_outline")
EST_SEC = (30, 80)                 # 每集一次 LLM 运行的大致耗时(秒),确认页按区间报
_PLACE = re.compile(r"^【(.+?) · .+?】\s*$", re.M)


@dataclass(frozen=True)
class Item:
    episode_id: int
    number: int
    status: str            # todo / has / in_flight / blocked
    task: str | None
    reason: str


def script_task(ep) -> str | None:
    if ep["source_type"] == "novel":
        return "script_adapt" if ep["source_text"].strip() and ep["outline"].strip() else None
    return "script_outline" if ep["outline"].strip() else None


def busy(store: Store, episode_id: int, tasks: tuple[str, ...]) -> str:
    """在途的原因:队列里已有 / 正在生成 / 有待审的;不在途为空串。"""
    marks = ",".join("?" * len(tasks))
    if store.conn.execute(f"SELECT 1 FROM llm_queue WHERE state = 'waiting' AND target_type = 'episode'"
                          f" AND target_id = ? AND task IN ({marks})", (episode_id, *tasks)).fetchone():
        return "已在队列里"
    r = store.conn.execute(f"SELECT state FROM llm_run WHERE target_type = 'episode' AND target_id = ?"
                           f" AND task IN ({marks}) AND state IN ('running', 'proposed') ORDER BY id DESC LIMIT 1",
                           (episode_id, *tasks)).fetchone()
    return {"running": "正在生成", "proposed": "有待你审的"}[r[0]] if r else ""


def classify(store: Store, ep) -> Item:
    if ep["script"].strip():
        return Item(ep["id"], ep["number"], "has", None, "已有剧本")
    why = busy(store, ep["id"], SCRIPT_TASKS)
    if why:
        return Item(ep["id"], ep["number"], "in_flight", None, why)
    task = script_task(ep)
    if task is None:
        return Item(ep["id"], ep["number"], "blocked", None, "没有大纲(小说集还要有原文)—— 先去「大纲」面板写")
    return Item(ep["id"], ep["number"], "todo", task, "按大纲改写原文" if task == "script_adapt" else "按大纲写(带前后集大纲)")


def _episodes(store: Store, drama_id: int):
    return store.conn.execute("SELECT * FROM episode WHERE drama_id = ? ORDER BY number", (drama_id,)).fetchall()


def plan(store: Store, drama_id: int) -> list[Item]:
    repo.get(store, "drama", drama_id)
    return [classify(store, ep) for ep in _episodes(store, drama_id)]


def estimate_minutes(n: int) -> tuple[int, int]:
    return math.ceil(n * EST_SEC[0] / 60), math.ceil(n * EST_SEC[1] / 60)


def confirm(store: Store, drama_id: int, episode_ids: list[int], minutes: int) -> list[int]:
    """只认确认页列出的集;逐集再判一次,还是 todo 的才入队。返回新入队的队列项 id。"""
    if minutes not in script_writer.MINUTES:
        raise ValueError(f"目标时长只能选 {script_writer.MINUTES} 分钟")
    out = []
    for eid in episode_ids:
        ep = store.conn.execute("SELECT * FROM episode WHERE id = ? AND drama_id = ?", (eid, drama_id)).fetchone()
        if ep is None:
            continue
        it = classify(store, ep)
        if it.status == "todo":
            out.append(llm_queue.enqueue(store, it.task, "episode", eid, {"minutes": minutes}))
    return out


def _names(store: Store, table: str, drama_id: int) -> list[str]:
    return [r[0] for r in store.conn.execute(f"SELECT name FROM {table} WHERE drama_id = ? ORDER BY id", (drama_id,))]


def storyboard_gaps(store: Store, episode_id: int) -> list[str]:
    """拆分镜前缺的资产(拦):没有场景 / 没有角色 / 剧本里的说话人不在角色表。"""
    ep = repo.get(store, "episode", episode_id)
    chars, scenes = _names(store, "character", ep["drama_id"]), _names(store, "scene", ep["drama_id"])
    out = [] if scenes else ["这部剧还没有场景"]
    out += [] if chars else ["这部剧还没有角色"]
    return out + [f"角色「{w}」" for w in sorted({w for w, _ in rules.dialogue_lines(ep["script"], chars)} - set(chars))]


def place_hints(store: Store, episode_id: int) -> list[str]:
    """剧本里的地点没有同名场景(只提示:拆分镜会从已有场景里挑最接近的)。"""
    ep = repo.get(store, "episode", episode_id)
    scenes = set(_names(store, "scene", ep["drama_id"]))
    return sorted({p.strip() for p in _PLACE.findall(ep["script"])} - scenes)


def after_script_approved(store: Store, episode_id: int) -> str:
    """整集剧本批准后:没段落、不在途、资产够 → 排「拆分镜」。返回给人看的一句话(不需要说明时为空串)。"""
    if store.conn.execute("SELECT COUNT(*) FROM segment WHERE episode_id = ?", (episode_id,)).fetchone()[0]:
        return ""
    if busy(store, episode_id, ("storyboard",)):
        return ""
    gaps = storyboard_gaps(store, episode_id)
    if gaps:
        return "没排分镜 —— 缺资产:" + "、".join(gaps)
    llm_queue.enqueue(store, "storyboard", "episode", episode_id, {})
    return "已排分镜"


def _cell(store: Store, episode_id: int, tasks: tuple[str, ...], have: str, blocked: str = "") -> dict:
    marks = ",".join("?" * len(tasks))
    q = store.conn.execute(f"SELECT id FROM llm_queue WHERE state = 'waiting' AND target_type = 'episode'"
                           f" AND target_id = ? AND task IN ({marks}) ORDER BY id LIMIT 1", (episode_id, *tasks)).fetchone()
    if q:
        return {"state": "queued", "label": f"排队中(第 {llm_queue.position(store, q[0])} 位)"}
    r = store.conn.execute(f"SELECT * FROM llm_run WHERE target_type = 'episode' AND target_id = ? AND task IN ({marks})"
                           f" ORDER BY id DESC LIMIT 1", (episode_id, *tasks)).fetchone()
    if r is not None and r["state"] == "running":
        return {"state": "running", "label": "生成中", "run": r["id"]}
    if r is not None and r["state"] == "proposed":
        try:
            stale = service.prepare(store, r["task"], episode_id, runs.params(r)).sha != r["input_sha"]
        except ValueError as e:              # 前提没了(如大纲被清空):这一格写原因,不让整页 500
            return {"state": "stale", "label": f"前提变了:{e}", "run": r["id"]}
        if stale:
            return {"state": "stale", "label": "输入已变,要重跑", "run": r["id"]}
        return {"state": "proposed", "label": "待你审", "run": r["id"]}
    if have:
        return {"state": "has", "label": have}
    if r is not None and r["state"] == "failed":
        return {"state": "failed", "label": f"失败:{(r['error'] or '')[:60]}", "run": r["id"]}
    if blocked:
        return {"state": "blocked", "label": blocked}
    return {"state": "none", "label": "—"}


def episode_rows(store: Store, drama_id: int) -> list[dict]:
    rows = []
    for ep in _episodes(store, drama_id):
        n_seg = store.conn.execute("SELECT COUNT(*) FROM segment WHERE episode_id = ?", (ep["id"],)).fetchone()[0]
        has_script = bool(ep["script"].strip())
        gaps = storyboard_gaps(store, ep["id"]) if has_script and not n_seg else []
        rows.append({
            "id": ep["id"], "number": ep["number"], "outline": bool(ep["outline"].strip()),
            "script": _cell(store, ep["id"], SCRIPT_TASKS, f"{rules.count_chars(ep['script'])} 字" if has_script else ""),
            "storyboard": _cell(store, ep["id"], ("storyboard",), f"{n_seg} 段" if n_seg else "",
                                f"缺资产:{'、'.join(gaps)}—— 先去资产里建" if gaps else ""),
        })
    return rows
```

    def _finish_llm(self, rid: int, llm: LLMBackend) -> None:
        kind: str | None = "error"
        try:
            kind = service.finish(self.store, rid, llm, self.data_root / "journal",   # → journal/llm/*.json
                                  data_root=self.data_root)
        except Exception as e:     # 不吞:落成 failed 带原因、进 journal(service 自己处理不了的意外)
            runs.fail(self.store, rid, f"{type(e).__name__}: {e}")
            journal.append(self.data_root, [f"llm_run {rid} 意外失败:{type(e).__name__}: {e}"], prefix="runtime ")
        self._after_llm(rid, kind)

    def _after_llm(self, rid: int, kind: str | None) -> None:
        """一条 LLM 运行结束:是队列项就给断路器计数,然后取下一条。手动发起的运行不计入、也不清零(spec 8.7)。"""
        with self._queue_lock:
            if llm_queue.item_for_run(self.store, rid) is not None:
                self._queue_fails = self._queue_fails + 1 if kind == "llm" else 0
                if self._queue_fails >= llm_queue.BREAK_AFTER:
                    err = runs.get(self.store, rid)["error"] or ""
                    reason = (f"连续 {self._queue_fails} 条 LLM 调用失败,队列已暂停 —— 最后一条:{err[:200]};"
                              "修好后点「继续」")
                    llm_queue.pause(self.store, reason)
                    journal.append(self.data_root, [reason], prefix="runtime ")
                    self._queue_fails = 0
        self.pump_queue()

    def pump_queue(self) -> None:
        """取下一条:没暂停、没有在跑的队列项,就开最早的 waiting。开运行当场失败(前提不满足)记下原因、接着取下一条;
        命中锁定版(不调 LLM)也接着取;真开了一条就停,等它结束时再来(spec 第 4 节)。"""
        with self._queue_lock:
            while not llm_queue.paused(self.store) and llm_queue.running_item(self.store) is None:
                item = llm_queue.next_waiting(self.store)
                if item is None:
                    return
                try:
                    rid, reused = self.start_llm(item["task"], item["target_id"], llm_queue.params(item))
                except LLMUnavailable as e:
                    llm_queue.pause(self.store, f"LLM 不可用,队列已暂停 —— {e};修好后点「继续」")
                    return
                except (ValueError, KeyError) as e:        # 单集问题(前提不满足 / 集被删了):不卡队列
                    llm_queue.mark_started(self.store, item["id"], None, str(e))
                    continue
                llm_queue.mark_started(self.store, item["id"], rid)
                if not reused:
                    return
```

"""逐集:一部剧每集的大纲 / 剧本 / 分镜到了哪一步;批量出剧本(两态)、停掉 / 继续队列、新建空集

from fastapi.responses import RedirectResponse

from tools.drama import repo
from tools.llm import batch, llm_queue, script_writer
from tools.serve.panels.common import render, rt

router = APIRouter()
LABELS = {"todo": "要出", "has": "跳过 · 已有剧本", "in_flight": "跳过 · 在途", "blocked": "跳过 · 缺内容"}
MAX_NEW = 200


def _ids(store, drama_id: int) -> list[int]:
    return [r[0] for r in store.conn.execute("SELECT id FROM episode WHERE drama_id = ?", (drama_id,))]


@router.get("/drama/{drama_id}/episodes")
def page(request: Request, drama_id: int):
    store = rt(request).store
    repo.get(store, "drama", drama_id)
    rows = batch.episode_rows(store, drama_id)
    cells = [r[k] for r in rows for k in ("script", "storyboard")]
    return render(request, "episodes.html", "episodes", drama_id=drama_id, rows=rows,
                  review=sum(c["state"] == "proposed" for c in cells),
                  inflight=any(c["state"] in ("queued", "running") for c in cells),
                  waiting=llm_queue.waiting_count(store, _ids(store, drama_id)), paused=llm_queue.paused(store))


@router.post("/drama/{drama_id}/episodes/batch")
async def batch_scripts(request: Request, drama_id: int):
    """两态:第一次出确认页(逐集归类 + 预计耗时);带 confirm=1 才入队,只认确认页列出的集。"""
    runtime = rt(request)
    form = await request.form()
    if form.get("confirm") == "1":
        raw = str(form.get("minutes", ""))
        if not raw.isdigit():
            raise ValueError(f"目标时长要填整数分钟,填的是 {raw!r}")
        ids = [int(x) for x in str(form.get("ids", "")).split(",") if x.strip().isdigit()]
        batch.confirm(runtime.store, drama_id, ids, int(raw))
        runtime.pump_queue()
        return RedirectResponse(f"/drama/{drama_id}/episodes", status_code=303)
    items = batch.plan(runtime.store, drama_id)
    todo = [i for i in items if i.status == "todo"]
    return render(request, "episodes_batch.html", "episodes", drama_id=drama_id, items=items, todo=todo,
                  est=batch.estimate_minutes(len(todo)), minutes=script_writer.MINUTES, labels=LABELS)


@router.post("/drama/{drama_id}/episodes/cancel")
def cancel(request: Request, drama_id: int):
    store = rt(request).store
    llm_queue.cancel_waiting(store, _ids(store, drama_id))
    return RedirectResponse(f"/drama/{drama_id}/episodes", status_code=303)


@router.post("/drama/{drama_id}/episodes/resume")
def resume(request: Request, drama_id: int):
    runtime = rt(request)
    llm_queue.resume(runtime.store)
    runtime.pump_queue()
    return RedirectResponse(f"/drama/{drama_id}/episodes", status_code=303)


@router.post("/drama/{drama_id}/episodes/new")
def new_episodes(request: Request, drama_id: int, n: int = Form(...)):
    """一次建 n 个空集(原创剧手写大纲用),集号接在已有最后一集之后。"""
    store = rt(request).store
    repo.get(store, "drama", drama_id)
    if not 1 <= n <= MAX_NEW:
        raise ValueError(f"一次建 1–{MAX_NEW} 集,填的是 {n}")
    last = store.conn.execute("SELECT COALESCE(MAX(number), 0) FROM episode WHERE drama_id = ?", (drama_id,)).fetchone()[0]
    for k in range(1, n + 1):
        repo.add_episode(store, drama_id, last + k, "script")
    return RedirectResponse(f"/drama/{drama_id}/episodes", status_code=303)
